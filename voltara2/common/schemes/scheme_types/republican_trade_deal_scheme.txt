	# On Actions
	on_phase_completed = {
		# Grab our scopes.
		save_scope_as = scheme
		scheme_target_character = { save_scope_as = target }
		scheme_owner = { save_scope_as = owner }
		# Roll our result.
		random = {
			chance = scope:scheme.scheme_success_chance
			save_scope_value_as = {
				name = scheme_successful
				value = yes
			}
		}
		# How'd we do?
		scope:owner = {
			if = {
				limit = { exists = scope:scheme_successful }
				trigger_event = { on_action = trade_deal_scheme_success }
			}
			else = {
				trigger_event = { on_action = trade_deal_scheme_failure }
			}
		}
	}

	on_monthly = {
		save_scope_as = scheme
		scheme_target_character = { save_scope_as = target }
		scheme_owner = {
			save_scope_as = owner
			trigger_event = { on_action = trade_deal_scheme_ongoing }
		}
	}
	on_invalidated = {
		scheme_target_character = {
			save_scope_as = target
		}
		scheme_owner = {
			save_scope_as = owner
		}
		if = {
			limit = {
				scope:target = { is_alive = no }
			}
			scope:owner = {
				send_interface_toast = {
					title = trade_deal_scheme_invalidated_title
					left_icon = scope:target
					custom_tooltip = trade_deal_scheme_invalidated_dead
				}
			}
		}
		else_if = {	
			limit = {
				scope:owner = {
					is_at_war_with = scope:target
				}
			}
			scope:owner = {
				send_interface_toast = {
					title = trade_deal_scheme_invalidated_title
					left_icon = scope:target
					custom_tooltip = trade_deal_scheme_invalidated_war
				}
			}
		}

		if = {
			limit = {
				scope:target = { 
					NOT = { in_diplomatic_range = scope:owner } 
				}
			}
			scope:owner = {
				send_interface_toast = {
					title = trade_deal_scheme_invalidated_title
					left_icon = scope:target
					custom_description_no_bullet = {
						object = scope:target
						text = scheme_target_not_in_diplomatic_range
					}
				}
			}
		}
		if = {
			limit = {
				scope:target = { 
					is_independent_ruler = no
				}
			}
			scope:owner = {
				send_interface_toast = {

					custom_description_no_bullet = {
						object = scope:target
						text = trade_deal_scheme_invalidated_vassal
					}
				}
			}
		}
