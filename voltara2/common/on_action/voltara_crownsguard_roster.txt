on_game_start = { on_actions = { build_kingsguard_menu_trait_list } }
build_kingsguard_menu_trait_list = {
    effect = {
        every_trait = {
            limit = {
              OR = {
                 this = trait:squire
                this = trait:dragonrider
              }
            }
            add_to_global_variable_list = {
                name = kingsguard_relevant_traits
                target = this
            }
        }
        every_trait = {
            limit = {
                OR = {
                    
                    this = trait:education_martial_prowess_1
                    this = trait:education_martial_prowess_2
                    this = trait:education_martial_prowess_3
                    this = trait:education_martial_prowess_4
                    this = trait:education_martial_prowess_5
                }
            }
            add_to_global_variable_list = {
                name = kingsguard_relevant_traits
                target = this
            }
        }
        
        every_trait = {
            limit = {
                OR = {
                    
                    this = trait:education_martial_1
                    this = trait:education_martial_2
                    this = trait:education_martial_3
                    this = trait:education_martial_4
                    this = trait:education_martial_5
                }
            }
            add_to_global_variable_list = {
                name = kingsguard_relevant_traits
                target = this
            }
        }
         every_trait = {
            limit = {
                OR = {
                    
                    this = {
                        has_trait_category = lifestyle
                    }
                }
            }
            add_to_global_variable_list = {
                name = kingsguard_relevant_traits
                target = this
            }
        }
        every_trait = {
            limit = {
                OR = {
                    
                    this = {
                        has_trait_category = commander
                    }
                }
            }
            add_to_global_variable_list = {
                name = kingsguard_relevant_traits
                target = this
            }
        }
        every_trait = {
            limit = {
                OR = {
                    this = trait:physique_good_1
                    this = trait:physique_good_2    
                    this = trait:physique_good_3
                    this = trait:giant
                    this = trait:strong
                }
            }
            add_to_global_variable_list = {
                name = kingsguard_relevant_traits
                target = this
            }
        }
        every_trait = {
            limit = {
                this = trait:refusing_marriage
            }
            add_to_global_variable_list = {
                name = kingsguard_relevant_traits
                target = this
            }
        }
    }
}
