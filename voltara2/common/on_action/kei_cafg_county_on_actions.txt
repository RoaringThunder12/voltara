# County changes rite. Not called when county converts to different faith
# Root is the county title
# scope:old_rite is the rite it used to have
on_county_rite_change = {
	 on_actions = {
        kCAFG_on_county_rite_change
    }
}

# County changes faith.
# Root is the county title
# scope:old_faith is the faith it used to have
on_county_faith_change = {
	on_actions = {
        kCAFG_on_county_faith_change
    }
}

# Root = County title
# scope:old_culture = old culture
on_county_culture_change = {
	on_actions = {
        kCAFG_on_county_culture_change
    }
}

# County changes rite. Not called when county converts to different faith
# Root is the county title
# scope:old_rite is the rite it used to have

    effect = {
        E_kCAFG_on_county_rite_changed = yes
    }

trigger = { has_global_variable = voltara_kei_population_ready }
}
