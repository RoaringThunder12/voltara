  if = {

   set_global_variable = vt_crownsguard_initialized
   character:19060001 = {
    save_scope_as = vt_crownsguard_sovereign

    agot_join_kingsguard_effect = { KINGSGUARD = character:19060039 KING = scope:vt_crownsguard_sovereign }
    agot_join_kingsguard_effect = { KINGSGUARD = character:19060040 KING = scope:vt_crownsguard_sovereign }
    while = {
     count = 4
     create_character = {
      age = { 24 40 } gender = male culture = scope:vt_crownsguard_sovereign.culture faith = scope:vt_crownsguard_sovereign.faith
      employer = scope:vt_crownsguard_sovereign
      trait = brave trait = loyal trait = education_martial_3
      prowess = { 18 26 } martial = { 12 18 }
      save_scope_as = vt_new_crownsguard
     }
     agot_join_kingsguard_effect = { KINGSGUARD = scope:vt_new_crownsguard KING = scope:vt_crownsguard_sovereign }
    }
    if = {
 limit = {
  exists = cp:kingsguard_lord_commander
  cp:kingsguard_lord_commander = character:19060034
  exists = cp:kingsguard_1 exists = cp:kingsguard_2 exists = cp:kingsguard_3

 }
 debug_log = "VT_CROWNSGUARD_VALIDATED: Lucius commands seven guards and holds only the Keep; Roland holds all three Blackwood titles."
}
else = { debug_log = "VT_CROWNSGUARD_FAILED: Check initial roster or Blackwood titles." }

   }
  }
 }
