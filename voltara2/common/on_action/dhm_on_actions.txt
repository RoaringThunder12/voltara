on_betrothal_broken = {
    on_actions = {
        dhm_on_betrothal_broken
    }
}

dhm_on_betrothal_broken = {
	# A broken betrothal may be caused by death. CK3 keeps the dead character as
	# scope:second, but character variables are no longer readable on that scope.
	# Only living partners are inspected; the survivor's lock is still cleared.
	effect = {
		if = {
			limit = { is_alive = yes }
			if = {
				limit = {
					has_variable = dhm_arranged_union_spouse
					var:dhm_arranged_union_spouse = scope:second
				}
				dhm_clear_pending_arranged_union_state_effect = yes
			}
		}
		if = {
			limit = { scope:second = { is_alive = yes } }
			scope:second = {
				if = {
					limit = {
						has_variable = dhm_arranged_union_spouse
						var:dhm_arranged_union_spouse = root
					}
					dhm_clear_pending_arranged_union_state_effect = yes
				}
			}
		}
	}
}
