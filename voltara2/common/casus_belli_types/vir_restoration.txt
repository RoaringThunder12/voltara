 icon = peasant_war group = civil_war target_titles = none
 cb_name = vir_restoration war_name = vir_restoration_name
 allowed_for_character = { has_character_flag = vir_restoration_rebel }
 allowed_against_character = { has_title = title:h_voltara_severos }

 should_invalidate = { OR = { scope:attacker = { is_alive = no } scope:defender = { NOT = { has_title = title:h_voltara_severos } } NOT = { scope:attacker.var:vir_disputed_title.holder = scope:attacker } } }

 on_defeat_desc = { desc = vir_restoration_defeat }

 on_invalidated_desc = { desc = vir_restoration_invalid }

 on_white_peace = { scope:attacker = { capital_county = { add_county_modifier = { modifier = vir_unsettled years = 5 } } } vir_end_restoration = yes }
 on_invalidated = { vir_end_restoration = yes }
}
