 icon = independence_faction_war group = independence target_titles = none ai = no

 allowed_for_character = { has_variable = vic_warrant_status }
 allowed_against_character = { this = character:19060003 }
 valid_to_start = { scope:attacker = { vic_warrant_pending = yes } scope:defender = { this = character:19060003 has_title = title:h_voltara_severos } }
 should_invalidate = { OR = { scope:attacker = { OR = { is_alive = no is_independent_ruler = yes } } NOT = { exists = title:h_voltara_severos.holder } } }

 attacker_capital_gives_war_score = yes max_attacker_score_from_battles = 100 max_defender_score_from_battles = 100

  resolve_title_and_vassal_change = scope:vic_warrant_freedom
 }
 on_defeat = { scope:defender = { if = { limit = { scope:attacker = { is_alive = yes is_imprisoned = no } } imprison = { target = scope:attacker type = dungeon } } } scope:attacker = { set_variable = { name = vic_warrant_status value = 5 } } }
 on_white_peace = { scope:attacker = { set_variable = { name = vic_warrant_status value = 6 } } }
}
