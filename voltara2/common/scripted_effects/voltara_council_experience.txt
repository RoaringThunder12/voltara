 if = { limit = { NOT = { has_game_rule = vt_council_xp_off } }

 if = { limit = { is_alive = yes }

  add_trait_xp = { trait = lifestyle_councillor track = $TRACK$ value = { value = 2 if = { limit = { OR = { has_trait = intellect_good_2 has_trait = intellect_good_3 } } add = 1 } if = { limit = { has_trait = diligent } add = 1 } if = { limit = { has_game_rule = vt_council_xp_double } multiply = 2 } } }
 }

 }
}
