dhm_v081_breathing_room_effect = {
	add_character_flag = { flag = dhm081_aftermath_pause months = 6 }
}
dhm_v081_capture_officer_role_effect = {
	if = { limit = { scope:dhm_story_officer = { has_character_flag = dhm_office_steward } } save_scope_value_as = { name = dhm081_officer_role value = flag:steward } }
	else_if = { limit = { scope:dhm_story_officer = { has_character_flag = dhm_office_matchmaker } } save_scope_value_as = { name = dhm081_officer_role value = flag:matchmaker } }
	else_if = { limit = { scope:dhm_story_officer = { has_character_flag = dhm_office_tutor } } save_scope_value_as = { name = dhm081_officer_role value = flag:tutor } }
	else_if = { limit = { scope:dhm_story_officer = { has_character_flag = dhm_office_chronicler } } save_scope_value_as = { name = dhm081_officer_role value = flag:chronicler } }
}
dhm_v081_close_pair_effect = {
	$FIRST$ = { add_opinion = { target = $SECOND$ modifier = dhm082_settled_marriage_opinion } }
	$SECOND$ = { add_opinion = { target = $FIRST$ modifier = dhm082_settled_marriage_opinion } }
	dhm_v081_breathing_room_effect = yes
}
dhm_v081_close_dispute_effect = {
	scope:dhm_dispute_first = { add_character_flag = { flag = dhm081_dispute_settled years = 8 } }
	scope:dhm_dispute_second = { add_character_flag = { flag = dhm081_dispute_settled years = 8 } }
	dhm_v081_breathing_room_effect = yes
}
dhm_v081_couple_settles_effect = {
	if = {
		limit = { dhm_v081_couple_self_reliant_trigger = yes }
		scope:dhm_crisis_relative = { add_stress = minor_stress_loss }
		scope:dhm_crisis_spouse = { add_stress = minor_stress_loss }
		dhm_v081_close_pair_effect = { FIRST = scope:dhm_crisis_relative SECOND = scope:dhm_crisis_spouse }
		send_interface_message = { type = event_diplomacy_good_with_text title = dhm081_couple_settled_title desc = dhm081_couple_settled_desc left_icon = scope:dhm_crisis_relative right_icon = scope:dhm_crisis_spouse }
	}
}
dhm_v081_school_callback_effect = {
	if = {
		limit = { dhm_v081_learned_to_listen_trigger = yes }
		dhm_v090_record_lesson_context_effect = yes
		scope:dhm_story_child = {
			add_character_flag = { flag = dhm081_used_lesson years = 10 }
			add_stress = miniscule_stress_loss
			add_opinion = { target = scope:dhm_story_tutor modifier = dhm_quiet_encouragement_opinion }
		}
		scope:dhm_story_tutor = { add_opinion = { target = scope:dhm_story_child modifier = dhm_quiet_encouragement_opinion } }
		send_interface_message = { type = event_stewardship_good_with_text title = dhm081_callback_title desc = dhm081_callback_desc left_icon = scope:dhm_story_child right_icon = scope:dhm_story_tutor }
		dhm_v081_breathing_room_effect = yes
	}
}
