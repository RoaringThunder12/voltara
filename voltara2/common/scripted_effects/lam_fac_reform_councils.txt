lam_cr_start_effect = {
	if = {
		limit = {
			is_alive = yes
			is_adult = yes
			is_imprisoned = no
			has_variable = lam_bloc_lead
			exists = liege
			NOT = {
				has_variable = lam_fac_disband
			}
			NOT = {
				has_variable = lam_cr_active
			}
			NOT = {
				has_variable = lam_cr_contract_pending
			}
			NOT = {
				has_variable = lam_cr_retry_$ACTION$
			}
			NOT = {
				liege = {
					has_variable = lam_fac_viewonly
				}
			}
			NOT = {
				has_variable = lam_pd_pending
			}
		}
		save_scope_as = lam_cr_head
		liege = {
			save_scope_as = lam_cr_lord
		}
		if = {
			limit = {
				NOT = {
					has_global_variable = lam_cr_serial
				}
			}
			set_global_variable = {
				name = lam_cr_serial
				value = 0
			}
		}
		change_global_variable = {
			name = lam_cr_serial
			add = 1
		}
		set_variable = {
			name = lam_cr_id
			value = global_var:lam_cr_serial
		}
		save_scope_value_as = {
			name = lam_cr_token
			value = {
				value = 0
				if = {
					limit = {
						has_global_variable = lam_cr_serial
					}
					add = global_var:lam_cr_serial
				}
			}
		}
		set_variable = {
			name = lam_cr_active
			value = yes
		}
		set_variable = {
			name = lam_cr_issue
			value = flag:$ISSUE$
		}
		set_variable = {
			name = lam_cr_action
			value = flag:$ACTION$
		}
		set_variable = {
			name = lam_cr_lord
			value = liege
		}
		set_variable = {
			name = lam_cr_target
			value = $TARGET$
		}
		if = {
			limit = {
				has_variable = lam_fac_rec
			}
			set_variable = {
				name = lam_cr_record
				value = var:lam_fac_rec
			}
		}
		else_if = {
			limit = {
				has_variable = lam_cr_record
			}
			remove_variable = lam_cr_record
		}
		if = {
			limit = {
				has_variable = lam_fac_heir
			}
			set_variable = {
				name = lam_cr_old
				value = var:lam_fac_heir
			}
		}
		else_if = {
			limit = {
				has_variable = lam_cr_old
			}
			remove_variable = lam_cr_old
		}
		set_variable = {
			name = lam_cr_retry_$ACTION$
			value = yes
			months = 6
		}
		lam_cs_merge_retry_effect = yes
		set_variable = {
			name = lam_cr_deadline_m
			value = {
				value = lam_now_m
				add = 1
			}
		}
		clear_variable_list = lam_cr_voters
		if = {
			limit = {
				var:lam_cr_issue ?= flag:confidence
				var:lam_cr_target ?= this
			}
			ordered_in_list = {
				variable = lam_fac_members
				max = 1
				check_range_bounds = no
				limit = {
					lam_cr_member_trigger = yes
					is_ruler = yes
					NOT = {
						has_variable = lam_bloc_lead
					}
				}
				order_by = lam_cr_influence_value
				set_variable = {
					name = lam_cr_candidate_tmp
					value = yes
					days = 1
				}
				scope:lam_cr_head = {
					set_variable = {
						name = lam_cr_target
						value = prev
					}
				}
			}
		}
		lam_wc_plan_rise_effect = yes
		if = {
			limit = {
				lam_cr_action_valid_trigger = yes
			}
			if = {
				limit = {
					var:lam_cr_action ?= flag:confidence
				}
				set_variable = {
					name = lam_fac_conf_cd
					value = yes
					years = 5
				}
			}
			if = {
				limit = {
					NOT = {
						var:lam_cr_issue ?= flag:confidence
					}
				}
				add_to_variable_list = {
					name = lam_cr_voters
					target = this
				}
			}
			every_in_list = {
				variable = lam_fac_members
				limit = {
					lam_cr_member_trigger = yes
					NOT = {
						this = scope:lam_cr_head
					}
				}
				scope:lam_cr_head = {
					if = {
						limit = {
							NOT = {
								is_target_in_variable_list = {
									name = lam_cr_voters
									target = prev
								}
							}
						}
						add_to_variable_list = {
							name = lam_cr_voters
							target = prev
						}
					}
				}
			}
			lam_cr_restore_context_effect = yes
			lam_bp_meeting_snapshot_effect = yes
			every_in_list = {
				variable = lam_cr_voters
				lam_cr_refresh_person_effect = yes
				set_variable = {
					name = lam_cr_ballot_head
					value = scope:lam_cr_head
				}
				set_variable = {
					name = lam_cr_ballot_id
					value = scope:lam_cr_token
				}
				set_variable = {
					name = lam_cr_weight
					value = lam_cr_vote_weight_value
				}
				lam_ux_ballot_snapshot_effect = yes
				set_variable = {
					name = lam_cr_choice
					value = flag:pending
				}
				if = {
					limit = {
						has_variable = lam_cr_requested
					}
					remove_variable = lam_cr_requested
				}
				if = {
					limit = {
						is_ai = yes
					}
					lam_cr_ai_vote_$ISSUE$_effect = yes
				}
			}
			lam_fac_stat_add_effect = {
				KEY = votes
			}
			lam_cr_recount_effect = yes
			every_in_list = {
				variable = lam_cr_voters
				limit = {
					is_ai = no
					lam_cr_saved_voter_trigger = yes
					NOT = {
						var:lam_cr_view ?= scope:lam_cr_head
					}
				}
				trigger_event = lam_cr.0001
			}
			if = {
				limit = {
					var:lam_cr_pending = 0
					NOT = {
						lam_cv_player_stake_trigger = yes
					}
				}
				lam_cr_resolve_effect = yes
			}
			else = {
				trigger_event = {
					id = lam_cr.0002
					days = 15
				}
			}
		}
		else = {
			lam_cr_cancel_effect = yes
		}
	}
}

lam_cr_timeout_effect = {
	if = {
		limit = {
			has_variable = lam_cr_active
			var:lam_cr_id = scope:lam_cr_token
		}
		if = {
			limit = {
				lam_cr_owner_valid_trigger = yes
			}
			lam_cr_restore_context_effect = yes
			every_in_list = {
				variable = lam_cr_voters
				limit = {
					lam_cr_saved_voter_trigger = yes
					var:lam_cr_choice ?= flag:pending
				}
				set_variable = {
					name = lam_cr_choice
					value = flag:abstain
				}
			}
			lam_cr_recount_effect = yes
			lam_cr_resolve_effect = yes
		}
		else = {
			lam_cr_cancel_effect = yes
		}
	}
}
lam_cr_resolve_effect = {
	if = {
		limit = {
			has_variable = lam_cr_active
		}
		lam_cr_restore_context_effect = yes
		if = {
			limit = {
				lam_cr_owner_valid_trigger = yes
				lam_cr_action_valid_trigger = yes
			}
			lam_cr_recount_effect = yes
			if = {
				limit = {
					var:lam_cr_pending = 0
					var:lam_cr_total > 0
				}
				save_scope_value_as = {
					name = lam_cr_result
					value = flag:rejected
				}
				if = {
					limit = {
						lam_cr_supermajority_trigger = yes
					}
					if = {
						limit = {
							var:lam_cr_yes >= {
								value = var:lam_cr_total
								multiply = 2
								divide = 3
							}
						}
						save_scope_value_as = {
							name = lam_cr_result
							value = flag:passed
						}
					}
				}
				else_if = {
					limit = {
						var:lam_cr_yes > {
							value = var:lam_cr_total
							multiply = 0.5
						}
					}
					save_scope_value_as = {
						name = lam_cr_result
						value = flag:passed
					}
				}
				remove_variable = lam_cr_active
				lam_cr_notify_result_effect = yes
				if = {
					limit = {
						scope:lam_cr_result = flag:passed
					}
					lam_cr_memory_effect = yes
					lam_cr_commit_effect = yes
					lam_cr_dissent_effect = yes
				}
				else = {
					lam_fac_cohesion_add_effect = {
						ADD = -2
					}
					lam_cr_rejected_effect = yes
				}
				lam_cr_clear_ballots_effect = yes
			}
			else_if = {
				limit = {
					var:lam_cr_total = 0
				}
				lam_cr_cancel_effect = yes
			}
		}
		else = {
			lam_cr_cancel_effect = yes
		}
	}
}
