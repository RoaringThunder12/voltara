lam_pd_grievance_add_effect = {
	if = {
		limit = {
			NOT = {
				has_variable = lam_pd_gr_$KIND$
			}
		}
		set_variable = {
			name = lam_pd_gr_$KIND$
			value = 0
		}
	}
	if = {
		limit = {
			NOT = {
				var:lam_pd_gr_$KIND$_last ?= lam_now_m
			}
		}
		change_variable = {
			name = lam_pd_gr_$KIND$
			add = $AMOUNT$
		}
		clamp_variable = {
			name = lam_pd_gr_$KIND$
			min = 0
			max = 30
		}
		set_variable = {
			name = lam_pd_gr_$KIND$_last
			value = lam_now_m
		}
	}
	set_variable = {
		name = lam_pd_gr_$KIND$_until
		value = {
			value = lam_now_m
			add = $MONTHS$
		}
	}
	if = {
		limit = {
			exists = liege
		}
		set_variable = {
			name = lam_pd_gr_$KIND$_actor
			value = liege
		}
	}
	set_variable = {
		name = lam_pd_gr_$KIND$_victim
		value = this
	}
	lam_pd_grievance_sync_effect = yes
}

lam_pd_grievance_migrate_effect = {
	if = {
		limit = {
			has_variable = lam_fac_grievance
			NOT = {
				has_variable = lam_pd_gr_office
			}
			NOT = {
				has_variable = lam_pd_gr_wealth
			}
			NOT = {
				has_variable = lam_pd_gr_promise
			}
			NOT = {
				has_variable = lam_pd_gr_repression
			}
			NOT = {
				has_variable = lam_pd_gr_succession
			}
			NOT = {
				has_variable = lam_pd_gr_policy
			}
			NOT = {
				has_variable = lam_pd_gr_legacy
			}
		}
		lam_pd_grievance_add_effect = {
			KIND = legacy
			AMOUNT = 10
			MONTHS = 24
		}
	}
	lam_pd_grievance_sync_effect = yes
}
