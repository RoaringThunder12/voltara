game_rule_randomize_faith_randomization_effect = {
	if = {
		limit = {
			NOT = { faith.religious_head ?= this }
			trigger_if = {
				limit = { top_liege != this }
				faith != liege.faith #We don't want to override faiths which "trickled down" via conversion effect
			}
		}



		#Randomize faith
		random_in_list = {
			list = faiths_list
			save_temporary_scope_as = randomized_faith
		}

		#Randomize rite within that faith
		scope:randomized_faith = {
			random_faith_rite = {
				save_temporary_scope_as = randomized_rite
			}
		}

		if = {
			limit = {
				exists = scope:randomized_rite
			}

			#Covert all courtiers, guests, pool characters of same faith
			if = {
				limit = {
					exists = capital_province
				}
				save_temporary_scope_as = this_ruler
				every_pool_character = {
					province = scope:this_ruler.capital_province
					limit = { faith = scope:this_ruler.faith }
					set_character_rite = scope:randomized_rite
				}
			}
			every_courtier_or_guest = {
				limit = {  faith = prev.faith }
				set_character_rite = scope:randomized_rite
			}

			#Change faith/rite with conversion for chance of trickling down to vassals
			set_character_rite_with_conversion = scope:randomized_rite

			#Convert every held county
			every_held_title = {
				title_tier = county
				E_kCAFG_replace_county_rite = { RITE = scope:randomized_rite }
			}
		}
	}
}
