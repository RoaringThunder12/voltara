# same scopes are available in these events as in the CBs' on_declaration, this just fires for all CBs instead of a specific CB.
E_kCAFG_declared_war_rite_tension_effects = {
    if = {
        limit = {
            scope:war = { using_holy_war_cb_trigger = yes }
        }

        scope:war.casus_belli = {
            ordered_target_title = {
                order_by = tier
                max = 1
                save_temporary_scope_as = highest_tiered_title # used by V_kCAFG_rite_tension_on_religious_war_declaration
            }
        }

		scope:attacker = {
            E_kCAFG_increase_scaled_rite_tension_in_realm_provinces = {
                RITE = scope:defender.rite
                VALUE = V_kCAFG_rite_tension_on_religious_war_declaration
                SCALE_MIN = 0.1
            }
        }
    }
}

# scope is the joiner
# scope:war is the war
E_kCAFG_joined_war_rite_tension_effects = {
    if = {
        limit = {
            scope:war = {
                using_holy_war_cb_trigger = yes
                is_attacker = root
            }
        }

        scope:war.casus_belli = {
            ordered_target_title = {
                order_by = tier
                max = 1
                save_temporary_scope_as = highest_tiered_title # used by V_kCAFG_rite_tension_on_religious_war_join
            }
        }

        E_kCAFG_increase_scaled_rite_tension_in_realm_provinces = {
            RITE = scope:war.primary_defender.rite
            VALUE = V_kCAFG_rite_tension_on_religious_war_join
            SCALE_MIN = 0.1
        }
    }
}

# Scope : war
# warring_character
E_kCAFG_ongoing_war_rite_tension_quarterly_change = {
    casus_belli = {
        ordered_target_title = {
            order_by = tier
            max = 1
            save_temporary_scope_as = highest_tiered_title # used by V_kCAFG_rite_tension_on_religious_warring_quarterly
        }
    }

    scope:warring_character = {
        E_kCAFG_increase_scaled_rite_tension_in_realm_provinces = {
            RITE = scope:war.primary_defender.rite
            VALUE = V_kCAFG_rite_tension_on_religious_warring_quarterly
            SCALE_MIN = 0.1
        }
    }
}

# scope : character
# RITE : rite which has their tension increased
# VALUE : script value or fixed point value which will be scaled depending on the rite 
# SCALE_MIN : minimum scale for the increase
E_kCAFG_increase_scaled_rite_tension_in_realm_provinces = {
    $RITE$ = { save_temporary_scope_as = kCAFG_rite }
    every_realm_province = {
        limit = { T_kCAFG_does_province_contain_rite = { RITE = scope:kCAFG_rite } }
        change_variable = {
            name = kCAFG_rite_tension
            add = {
                # Base value
                value = $VALUE$
                # Scale
                multiply = {
                    value = $SCALE_MIN$
                    add = {
                        value = V_kCAFG_province_rite_factor
                        multiply = {
                            value = 1
                            subtract = $SCALE_MIN$
                        }
                    }
                }
            }
        }
        E_kCAFG_on_province_rite_tension_updated = yes
    }
}
