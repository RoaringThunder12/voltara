lam_lf_campaign_pick_effect = {
	lam_st_campaign_clear_effect = yes
	remove_variable = lam_lf_preparations
	lam_st_campaign_pick_effect = yes
	save_scope_as = lam_lf_head
	if = {
		limit = {
			lam_lf_office_pool_trigger = yes
			has_variable = lam_pd_gr_office
			var:lam_pd_gr_office > var:lam_st_campaign_score
		}
		set_variable = {
			name = lam_st_campaign_kind
			value = flag:council
		}
		set_variable = {
			name = lam_st_campaign_score
			value = var:lam_pd_gr_office
		}
	}
	if = {
		limit = {
			has_variable = lam_pd_gr_policy
			var:lam_pd_gr_policy > var:lam_st_campaign_score
			var:lam_fac_goal ?= 1
			liege = {
				OR = {
					has_realm_law = crown_authority_1
					has_realm_law = crown_authority_2
					has_realm_law = crown_authority_3
				}
			}
		}
		set_variable = {
			name = lam_st_campaign_kind
			value = flag:law
		}
		set_variable = {
			name = lam_st_campaign_score
			value = var:lam_pd_gr_policy
		}
	}
	if = {
		limit = {
			has_variable = lam_pd_gr_repression
			var:lam_pd_gr_repression >= 10
			any_in_list = {
				variable = lam_fac_members
				is_alive = yes
				var:lam_bloc ?= prev
				is_imprisoned_by = prev.liege
			}
		}
		set_variable = {
			name = lam_st_campaign_kind
			value = flag:release
		}
	}
	if = {
		limit = {
			var:lam_st_campaign_kind ?= flag:gold
			var:lam_pd_gr_wealth >= 20
			OR = {
				var:lam_fac_agenda ?= flag:parochial
				var:lam_fac_agenda ?= flag:barons_and_minor_landholders
			}
			liege = {
				lam_nc_county_available_trigger = {
					RECIPIENT = scope:lam_lf_head
				}
			}
		}
		set_variable = {
			name = lam_st_campaign_kind
			value = flag:land
		}
	}
	if = {
		limit = {
			lam_lf_pledge_current_trigger = yes
			var:lam_lf_pledged_member = {
				lam_st_office_available_trigger = yes
				NOT = {
					is_councillor_of = liege
				}
			}
		}
		set_variable = {
			name = lam_st_campaign_kind
			value = flag:council
		}
	}
	if = {
		limit = {
			lam_lf_exchange_current_trigger = yes
			var:lam_lf_exchange_role ?= flag:sponsor
		}
		if = {
			limit = {
				var:lam_lf_exchange_stage ?= 1
			}
			set_variable = {
				name = lam_st_campaign_kind
				value = flag:heir
			}
			set_variable = {
				name = lam_st_campaign_target
				value = var:lam_lf_exchange_heir
			}
		}
		else = {
			set_variable = {
				name = lam_st_campaign_kind
				value = flag:council
			}
		}
	}
	lam_bp_campaign_effect = yes
	if = {
		limit = {
			has_variable = lam_st_campaign_kind
		}
		set_variable = {
			name = lam_st_campaign_lord
			value = liege
		}
		set_variable = {
			name = lam_st_campaign_until
			value = {
				value = lam_now_m
				add = 36
			}
		}
		if = {
			limit = {
				has_variable = lam_fac_rec
			}
			set_variable = {
				name = lam_st_campaign_record
				value = var:lam_fac_rec
			}
		}
		set_variable = {
			name = lam_lf_preparations
			value = 0
		}
	}
}
lam_lf_campaign_plan_effect = {
	if = {
		limit = {
			has_variable = lam_lf_setback_lord
			NOT = {
				var:lam_lf_setback_lord ?= liege
			}
		}
		remove_variable = lam_lf_setback_lord
		remove_variable = lam_lf_setbacks
		remove_variable = lam_lf_setback_until
		remove_variable = lam_st_retry_after
	}
	if = {
		limit = {
			has_variable = lam_st_campaign_kind
			NOT = {
				lam_lf_campaign_current_trigger = yes
			}
		}
		lam_st_campaign_clear_effect = yes
	}
	if = {
		limit = {
			NOT = {
				has_variable = lam_st_campaign_kind
			}
		}
		lam_lf_campaign_pick_effect = yes
	}
	if = {
		limit = {
			lam_lf_campaign_current_trigger = yes
		}
		set_variable = {
			name = lam_pd_plan
			value = flag:wait
		}
		if = {
			limit = {
				NOT = {
					has_variable = lam_head_demand_cd
				}
				NOT = {
					has_variable = lam_fac_promise
				}
				OR = {
					NOT = {
						has_variable = lam_st_retry_after
					}
					var:lam_st_retry_after <= lam_now_m
				}
				OR = {
					NOT = {
						has_variable = lam_lf_setback_until
					}
					var:lam_lf_setback_until <= lam_now_m
				}
			}
			if = {
				limit = {
					NOT = {
						has_variable = lam_lf_preparations
					}
				}
				set_variable = {
					name = lam_lf_preparations
					value = 0
				}
			}
			if = {
				limit = {
					var:lam_lf_preparations < 2
					trigger_if = {
						limit = {
							has_variable = lam_fac_cohesion
						}
						var:lam_fac_cohesion < 45
					}
					trigger_else = {
						always = no
					}
					gold >= 150
					NOT = {
						has_variable = lam_pd_aid_cd
					}
					any_in_list = {
						variable = lam_fac_members
						is_alive = yes
						var:lam_bloc ?= prev
						liege ?= prev.liege
						gold < 0
					}
				}
				lam_pd_aid_member_effect = yes
				change_variable = {
					name = lam_lf_preparations
					add = 1
				}
				set_variable = {
					name = lam_pd_plan
					value = flag:aid
				}
			}
			else_if = {
				limit = {
					var:lam_lf_preparations < 2
					trigger_if = {
						limit = {
							has_variable = lam_fac_count
						}
						var:lam_fac_count < 3
					}
					trigger_else = {
						always = no
					}
					gold >= 100
					lam_pd_recruit_needed_trigger = yes
					NOT = {
						has_variable = lam_fac_recruit_cd
					}
				}
				lam_pd_recruit_search_effect = yes
				change_variable = {
					name = lam_lf_preparations
					add = 1
				}
				set_variable = {
					name = lam_pd_plan
					value = flag:recruit
				}
			}
			else_if = {
				limit = {
					var:lam_lf_preparations < 1
					trigger_if = {
						limit = {
							has_variable = lam_cr_share
						}
						var:lam_cr_share < 25
					}
					trigger_else = {
						always = no
					}
					NOT = {
						has_variable = lam_pd_pact_partner
					}
					NOT = {
						var:lam_st_campaign_kind ?= flag:gold
					}
					NOT = {
						var:lam_st_campaign_kind ?= flag:release
					}
				}
				lam_pd_pact_search_effect = yes
				change_variable = {
					name = lam_lf_preparations
					add = 1
				}
				set_variable = {
					name = lam_pd_plan
					value = flag:pact
				}
			}
			else_if = {
				limit = {
					gold < 0
					NOT = {
						var:lam_st_campaign_kind ?= flag:gold
					}
					NOT = {
						var:lam_st_campaign_kind ?= flag:release
					}
				}
				set_variable = {
					name = lam_pd_plan
					value = flag:wait
				}
			}
			else = {
				set_variable = {
					name = lam_pd_plan
					value = flag:petition
				}
				set_variable = {
					name = lam_st_retry_after
					value = {
						value = lam_now_m
						add = 12
					}
				}
				if = {
					limit = {
						var:lam_st_campaign_kind ?= flag:council
					}
					lam_pd_begin_effect = {
						KIND = council
					}
				}
				else_if = {
					limit = {
						var:lam_st_campaign_kind ?= flag:heir
					}
					lam_pd_begin_effect = {
						KIND = heir
					}
				}
				else_if = {
					limit = {
						var:lam_st_campaign_kind ?= flag:gold
					}
					lam_pd_begin_effect = {
						KIND = gold
					}
				}
				else_if = {
					limit = {
						var:lam_st_campaign_kind ?= flag:land
					}
					lam_pd_begin_effect = {
						KIND = land
					}
				}
				else_if = {
					limit = {
						var:lam_st_campaign_kind ?= flag:release
					}
					lam_pd_begin_effect = {
						KIND = release
					}
				}
				else_if = {
					limit = {
						var:lam_st_campaign_kind ?= flag:law
					}
					lam_pd_begin_effect = {
						KIND = law
					}
				}
			}
		}
	}
	else_if = {
		limit = {
			OR = {
				NOT = {
					has_variable = lam_lf_setback_until
				}
				var:lam_lf_setback_until <= lam_now_m
			}
		}
		lam_st_opportunistic_plan_effect = yes
	}
}
lam_lf_campaign_refused_effect = {
	if = {
		limit = {
			has_variable = lam_st_campaign_kind
		}
		if = {
			limit = {
				NOT = {
					var:lam_lf_setback_lord ?= liege
				}
			}
			set_variable = {
				name = lam_lf_setbacks
				value = 0
			}
		}
		if = {
			limit = {
				NOT = {
					has_variable = lam_lf_setbacks
				}
			}
			set_variable = {
				name = lam_lf_setbacks
				value = 0
			}
		}
		set_variable = {
			name = lam_lf_setback_lord
			value = liege
		}
		change_variable = {
			name = lam_lf_setbacks
			add = 1
		}
		clamp_variable = {
			name = lam_lf_setbacks
			min = 0
			max = 3
		}
		set_variable = {
			name = lam_lf_setback_until
			value = {
				value = lam_now_m
				add = {
					value = var:lam_lf_setbacks
					multiply = 6
					add = 6
				}
			}
		}
		set_variable = {
			name = lam_lf_preparations
			value = 0
		}
	}
}
