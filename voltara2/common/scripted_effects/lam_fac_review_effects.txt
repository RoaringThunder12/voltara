	}
}

# Copy a faction-owned value with its original expiry, and consume the old copy.
# Old saves did not record expiry for these states: use one season, never a fresh
# multi-year term. Native timed flags still expire normally before a succession.
lam_fac_review_timed_pass_effect = {
	if = {
		limit = { $FROM$ = { has_variable = $NAME$ } }
		$TO$ = {
			set_variable = { name = $NAME$ value = $FROM$.var:$NAME$ }
			if = { limit = { $FROM$ = { has_variable = $NAME$_m } } set_variable = { name = $NAME$_m value = $FROM$.var:$NAME$_m } }
			else = { set_variable = { name = $NAME$_m value = { value = lam_now_m add = $LEGACY_MONTHS$ } } }
			lam_fac_timer_expire_effect = { NAME = $NAME$ }
		}
		$FROM$ = { lam_fac_timer_drop_effect = { NAME = $NAME$ } }
	}
}
lam_fac_review_state_pass_effect = {
	if = {
		limit = { NOT = { $FROM$ = $TO$ } }
		if = {
			limit = { $FROM$ = { has_variable = lam_fac_inf } }
			$TO$ = { set_variable = { name = lam_fac_inf value = $FROM$.var:lam_fac_inf } }
			$FROM$ = { remove_variable = lam_fac_inf }
		}
		lam_fac_review_timed_pass_effect = { NAME = lam_fac_content FROM = $FROM$ TO = $TO$ LEGACY_MONTHS = 3 }
		lam_fac_review_timed_pass_effect = { NAME = lam_fac_conceded FROM = $FROM$ TO = $TO$ LEGACY_MONTHS = 3 }
		lam_fac_review_timed_pass_effect = { NAME = lam_fac_rallied_war FROM = $FROM$ TO = $TO$ LEGACY_MONTHS = 120 }
		# Includes legacy candidates that were only recorded on members.
		$FROM$ = {
			every_in_list = { variable = lam_fac_members limit = { var:lam_fac_backed_by ?= $FROM$ } $TO$ = { add_to_variable_list = { name = lam_fac_backed_candidates target = prev } } }
			every_in_list = { variable = lam_fac_backed_candidates limit = { var:lam_fac_backed_by ?= $FROM$ } $TO$ = { add_to_variable_list = { name = lam_fac_backed_candidates target = prev } } }
			clear_variable_list = lam_fac_backed_candidates
		}
		$TO$ = {
			every_in_list = {
				variable = lam_fac_backed_candidates
				limit = { var:lam_fac_backed_by ?= $FROM$ }
				set_variable = { name = lam_fac_backed_by value = $TO$ }
			}
			every_in_list = {
				variable = lam_fac_members
				limit = { var:lam_fac_sulk ?= $FROM$ }
				set_variable = { name = lam_fac_sulk value = $TO$ }
			}
		}
		lam_fac_truce_pass_effect = { FROM = $FROM$ TO = $TO$ }
	}
}
lam_fac_review_state_drop_effect = {
	save_scope_as = lam_review_drop_head
	every_in_list = {
		variable = lam_fac_backed_candidates
		limit = { var:lam_fac_backed_by ?= scope:lam_review_drop_head }
		remove_variable = lam_fac_backed_by
		if = { limit = { has_variable = lam_fac_backed_for } remove_variable = lam_fac_backed_for }
	}
	if = { limit = { has_variable = lam_fac_inf } remove_variable = lam_fac_inf }
	lam_fac_timer_drop_effect = { NAME = lam_fac_content }
	lam_fac_timer_drop_effect = { NAME = lam_fac_conceded }
	lam_fac_timer_drop_effect = { NAME = lam_fac_rallied_war }
	clear_variable_list = lam_fac_backed_candidates
	lam_fac_truce_drop_effect = yes
}
lam_fac_review_state_expire_effect = {
