vic_issue_warrants = {
 if = { limit = { this = character:19060003 is_alive = yes has_title = title:h_voltara_severos global_var:vic_war_result = 1 NOT = { has_global_variable = vic_warrants_issued } }
  set_global_variable = { name = vic_warrants_issued value = yes }
  every_vassal_or_below = { limit = { vic_warrant_recipient = yes NOT = { has_variable = vic_warrant_status } }
   set_variable = { name = vic_warrant_status value = 1 }
   trigger_event = vic_warrants.2
  }
 }
}
vic_accept_warrant = {
 if = { limit = { vic_warrant_pending = yes }
  save_scope_as = vic_arrest_target
  character:19060003 = { imprison = { target = scope:vic_arrest_target type = dungeon } }
  set_variable = { name = vic_warrant_status value = 2 }
 }
}
vic_refuse_warrant = {
 if = { limit = { vic_warrant_pending = yes }
  # The demanded emperor fights their own war; the rest of the hegemony stays intact.
  start_war = { casus_belli = vic_warrant_independence target = character:19060003 }
  if = { limit = { any_character_war = { using_cb = vic_warrant_independence is_attacker = prev } }
   set_variable = { name = vic_warrant_status value = 3 }
  }
  else = { trigger_event = { id = vic_warrants.2 days = 3 } }
 }
}
