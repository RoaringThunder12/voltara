vic_fixed_war_side = {
 save_scope_as = vic_fixed_member
 set_variable = { name = vic_fixed_war_side value = $SIDE$ }
 set_variable = { name = vip_alignment value = $SIDE$ }
 remove_character_flag = vic_neutral_house
 remove_character_flag = vip_wartime_neutral
 if = { limit = { var:vic_fixed_war_side = 2 }
  add_character_flag = vic_pledged_octavian remove_character_flag = vic_pledged_cassian
  add_character_flag = vic_wartime_opposition remove_character_flag = vic_wartime_crown
 } else = {
  add_character_flag = vic_pledged_cassian remove_character_flag = vic_pledged_octavian
  add_character_flag = vic_wartime_crown remove_character_flag = vic_wartime_opposition
 }
 if = { limit = { is_landed = yes highest_held_title_tier > tier_barony is_independent_ruler = no }
  voltara_mw_detach = yes
 }
 if = { limit = { is_landed = yes }
  primary_title = {
   set_variable = { name = voltara_mw_home value = title:h_voltara_severos }
   set_variable = { name = voltara_mw_war value = scope:war }
  }
  if = { limit = { var:vic_fixed_war_side = 2 }
   scope:war = { if = { limit = { NOT = { is_attacker = scope:vic_fixed_member } } add_attacker = scope:vic_fixed_member } }
  } else = {
   scope:war = { if = { limit = { NOT = { is_defender = scope:vic_fixed_member } } add_defender = scope:vic_fixed_member } }
  }
 }
}
