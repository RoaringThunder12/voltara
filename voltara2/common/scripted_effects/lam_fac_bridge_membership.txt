	save_scope_as = lam_pb_detached_member
	var:lam_bloc ?= {

			limit = {
				is_alive = yes
			}
			remove_list_variable = {
				name = lam_fac_members
				target = scope:lam_pb_detached_member
			}
			remove_list_variable = {
				name = lam_fac_prev_members
				target = scope:lam_pb_detached_member
			}
		}
	}
	remove_variable = lam_bloc
	remove_variable = lam_fac_tier
	remove_variable = lam_fac_pinned
	remove_variable = lam_cr_service_head
	remove_variable = lam_cr_since_m
	remove_variable = lam_cr_service
	if = {
		limit = {
			has_variable = lam_fac_blocked
		}
		lam_fac_unblock_effect = yes
	}
	lam_ss_clear_member_effect = yes
}
lam_pb_validate_member_effect = {
	if = {
		limit = {
			is_alive = yes
		}
		if = {
			limit = {
				has_variable = lam_bloc
				NOT = {
					has_variable = lam_bloc_lead
				}
			}
			if = {
				limit = {
					lam_pb_member_current_trigger = yes
				}
				save_scope_as = lam_pb_verified_member
				var:lam_bloc = {
					if = {
						limit = {
							NOT = {
								is_target_in_variable_list = {
									name = lam_fac_members
									target = scope:lam_pb_verified_member
								}
							}
						}
						add_to_variable_list = {
							name = lam_fac_members
							target = scope:lam_pb_verified_member
						}
					}
				}
			}
			else = {
				lam_pb_clear_membership_effect = yes
			}
		}
	}
