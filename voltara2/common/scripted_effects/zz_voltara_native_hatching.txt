agot_spawn_owned_hatchling_from_egg_effect = {
 $OWNER$ = { save_scope_as = owner }
 $EGG$ = { save_scope_as = egg }
 clear_saved_scope = dragon
 if = { limit = { exists = scope:egg scope:owner = { is_alive = yes } scope:egg = { artifact_owner = scope:owner } }
 create_character = {
  location = scope:owner.location
  template = voltara_agot_hatchling_template
  faith = scope:owner.faith
  random_traits = no
  save_scope_as = dragon
 }
 if = { limit = { exists = scope:dragon }
  scope:dragon = {
   voltara_register_dragon_effect = yes
   set_variable = { name = voltara_hatched_by value = scope:owner }
   if = { limit = { exists = scope:egg.var:egg_color } set_variable = { name = egg_color value = scope:egg.var:egg_color } }
   if = { limit = { exists = scope:egg.var:dragon_mother } set_mother = scope:egg.var:dragon_mother }
   else_if = { limit = { exists = scope:egg.var:dragon_parent }
    if = { limit = { scope:egg.var:dragon_parent = { is_female = yes } } set_mother = scope:egg.var:dragon_parent }
    else = { set_father = scope:egg.var:dragon_parent }
   }
  }
  scope:owner = { add_courtier = scope:dragon }
  scope:dragon = { add_character_flag = owned_dragon }
  if = { limit = { exists = scope:egg.var:dragon_parent_other }
   scope:dragon = { set_father = scope:egg.var:dragon_parent_other }
  }
  if = { limit = { NOT = { has_game_rule = agot_dragon_trait_inheritance_disabled } }
   agot_transfer_parent_congenital_traits_to_dragon = { DRAGON_EGG = scope:egg DRAGON = scope:dragon }
   agot_dragon_assign_parent_congenital_traits = { DRAGON = scope:dragon }
  }
  scope:egg = { remove_list_global_variable = { name = dragon_eggs target = this } }
  destroy_artifact = scope:egg
 }

 }
}
