# Living Factions - court issues (tools/claude_fixes/issue.py). This = the lord unless noted.
# Called from lam_fac_realm_pulse_effect right after lam_fac_heirs_effect.
lam_is_pulse_effect = {
	save_scope_as = lam_is_lord
	lam_is_type_pulse_effect = {
		T = heir
	}
	lam_is_type_pulse_effect = {
		T = crown
	}
	lam_is_type_pulse_effect = {
		T = council
	}
	lam_is_type_pulse_effect = {
		T = faith
	}
	lam_is_type_pulse_effect = {
		T = tax
	}
	lam_is_type_pulse_effect = {
		T = war
	}
	if = {
		limit = {
			lam_is_open_count_value < 3
		}
		lam_is_pick_new_effect = yes
	}
	if = {
		limit = {
			lam_is_open_count_value > 0
		}
		set_variable = {
			name = lam_is_any
			value = yes
		}
	}
	else = {
		remove_variable = lam_is_any
	}
	# A player vassal of an AI court gets the yearly brief (a player lord sees the issues in the court digest).
	if = {
		limit = {
			is_ai = yes
			has_variable = lam_is_any
		}
		every_vassal = {
			limit = {
				is_ai = no
				NOT = {
					has_variable = lam_is_brief_cd
				}
			}
			set_variable = {
				name = lam_is_brief_cd
				value = yes
				months = 11
			}
			trigger_event = lam_is.0002
		}
	}
}

lam_is_type_pulse_effect = {
	clear_saved_scope = lam_is_ca
	clear_saved_scope = lam_is_cb
	lam_is_$T$_scopes_effect = yes
	if = {
		limit = {
			has_variable = lam_is_$T$_open
		}
		lam_is_check_effect = {
			T = $T$
		}
	}
	if = {
		limit = {
			has_variable = lam_is_$T$_open
		}
		lam_is_$T$_lock_effect = yes
		lam_is_sum_effect = {
			T = $T$
		}
		if = {
			limit = {
				lam_is_held_trigger = {
					T = $T$
				}
			}
			change_variable = {
				name = lam_is_$T$_since
				add = {
					value = 12
					if = {
						limit = {
							has_variable = lam_fac_fast
						}
						value = 3
					}
				}
			}
		}
		else = {
			lam_is_drift_effect = {
				T = $T$
			}
			lam_is_ai_effect = {
				T = $T$
			}
			lam_is_resolve_effect = {
				T = $T$
			}
		}
	}
	if = {
		limit = {
			has_variable = lam_is_$T$_open
		}
		lam_is_$T$_dissent_effect = yes
	}
}

# At most one new issue per court update: the hottest candidate - the weight on both sides plus the crown's stake.
# A candidate is faction against faction (see lam_is_open/oppose/split_share_value); a flashpoint (lam_is_T_hot_trigger:
# an ageing lord, a weak crown, an empty seat, a long war...) makes it the first to surface.
lam_is_pick_new_effect = {
	set_variable = {
		name = lam_is_best
		value = 0
	}
	lam_is_candidate_effect = {
		T = heir
	}
	lam_is_candidate_effect = {
		T = crown
	}
	lam_is_candidate_effect = {
		T = council
	}
	lam_is_candidate_effect = {
		T = faith
	}
	lam_is_candidate_effect = {
		T = tax
	}
	lam_is_candidate_effect = {
		T = war
	}
	if = {
		limit = {
			var:lam_is_best_t ?= flag:heir
		}
		lam_is_open_chosen_effect = {
			T = heir
		}
	}
	if = {
		limit = {
			var:lam_is_best_t ?= flag:crown
		}
		lam_is_open_chosen_effect = {
			T = crown
		}
	}
	if = {
		limit = {
			var:lam_is_best_t ?= flag:council
		}
		lam_is_open_chosen_effect = {
			T = council
		}
	}
	if = {
		limit = {
			var:lam_is_best_t ?= flag:faith
		}
		lam_is_open_chosen_effect = {
			T = faith
		}
	}
	if = {
		limit = {
			var:lam_is_best_t ?= flag:tax
		}
		lam_is_open_chosen_effect = {
			T = tax
		}
	}
	if = {
		limit = {
			var:lam_is_best_t ?= flag:war
		}
		lam_is_open_chosen_effect = {
			T = war
		}
	}
	remove_variable = lam_is_best
	remove_variable = lam_is_best_t
}
lam_is_candidate_effect = {
	if = {
		limit = {
			NOT = {
				has_variable = lam_is_$T$_open
			}
			NOT = {
				has_variable = lam_is_$T$_cd
			}
			lam_is_$T$_open_trigger = yes
		}
		clear_saved_scope = lam_is_ca
		clear_saved_scope = lam_is_cb
		lam_is_$T$_prepare_effect = yes
		lam_is_sum_effect = {
			T = $T$
		}
		save_scope_value_as = {
			name = lam_is_heat
			value = {
				value = var:lam_is_$T$_sa
				add = var:lam_is_$T$_sb
				add = lam_is_$T$_lw_value
				if = {
					limit = {
						lam_is_$T$_hot_trigger = yes
					}
					add = 20
				}
				if = {
					limit = {
						has_variable = lam_is_$T$_mem
					}
					add = 10
				}
			}
		}
		if = {
			limit = {
				var:lam_is_$T$_sb >= lam_is_open_share_value
				var:lam_is_$T$_sa >= lam_is_oppose_share_value
				lam_is_$T$_split_value >= lam_is_split_share_value
				scope:lam_is_heat > var:lam_is_best
			}
			set_variable = {
				name = lam_is_best
				value = scope:lam_is_heat
			}
			set_variable = {
				name = lam_is_best_t
				value = flag:$T$
			}
		}
		remove_variable = lam_is_$T$_sa
		remove_variable = lam_is_$T$_sb
		remove_variable = lam_is_$T$_lw
		remove_variable = lam_is_$T$_xp
		remove_variable = lam_is_$T$_yp
		remove_variable = lam_is_$T$_ua
		remove_variable = lam_is_$T$_ub
		remove_variable = lam_is_$T$_cz
		remove_variable = lam_is_$T$_cza
		every_in_list = {
			variable = lam_bloc_leads
			remove_variable = lam_is_st_$T$
		}
		lam_is_$T$_unprepare_effect = yes
	}
}
lam_is_open_chosen_effect = {
	clear_saved_scope = lam_is_ca
	clear_saved_scope = lam_is_cb
	lam_is_$T$_prepare_effect = yes
	lam_is_sum_effect = {
		T = $T$
	}
	lam_is_open_effect = {
		T = $T$
	}
}

# Validity: side Y already got its way (win), the issue lost its subject (void), side Y has no faction and no unaligned
# backer left (side X wins if it held its ground, otherwise the debate fades), or ten years on (the side ahead takes it).
lam_is_check_effect = {
	if = {
		limit = {
			lam_is_$T$_y_done_trigger = yes
		}
		lam_fac_stat_add_effect = {
			KEY = is_end_done_y
		}
		lam_is_win_y_effect = {
			T = $T$
		}
	}
	else_if = {
		limit = {
			lam_is_$T$_x_done_trigger = yes
		}
		lam_fac_stat_add_effect = {
			KEY = is_end_done_x
		}
		lam_is_win_x_effect = {
			T = $T$
		}
	}
	else_if = {
		limit = {
			lam_is_$T$_void_trigger = yes
		}
		lam_fac_stat_add_effect = {
			KEY = is_end_void
		}
		lam_is_close_effect = {
			T = $T$
			KEY = lam_is_msg_$T$_void
		}
	}
	else_if = {
		limit = {
			NOT = {
				any_in_list = {
					variable = lam_bloc_leads
					is_alive = yes
					lam_is_side_$T$_y_trigger = yes
				}
			}
			NOT = {
				any_vassal = {
					lam_is_backer_trigger = {
						T = $T$
						SIDE = y
					}
				}
			}
			NOT = {
				any_in_list = {
					variable = lam_bloc_leads
					is_alive = yes
					lam_wc_committed_trigger = yes
				}
			}
		}
		# side Y fell apart (its factions dissolved, merged, were banned or turned): side X carries the day
		lam_fac_stat_add_effect = {
			KEY = is_end_fall_y
		}
		lam_fac_stat_add_effect = {
			KEY = is_end_fall_y_$T$
		}
		lam_is_win_x_key_effect = {
			T = $T$
			KEY = lam_is_msg_$T$_fall_y
		}
	}
	else_if = {
		limit = {
			NOT = {
				any_in_list = {
					variable = lam_bloc_leads
					is_alive = yes
					lam_is_side_$T$_x_trigger = yes
				}
			}
			NOT = {
				any_vassal = {
					lam_is_backer_trigger = {
						T = $T$
						SIDE = x
					}
				}
			}
			lam_is_$T$_lw_value < 1
			NOT = {
				any_in_list = {
					variable = lam_bloc_leads
					is_alive = yes
					lam_wc_committed_trigger = yes
				}
			}
		}
		# side X fell apart and the crown has no stake: side Y has carried the court and the lord must decide
		lam_fac_stat_add_effect = {
			KEY = is_end_fall_x
		}
		set_variable = {
			name = lam_is_$T$_bar
			value = 100
		}
		set_variable = {
			name = lam_is_$T$_disp
			value = 0
		}
		set_variable = {
			name = lam_is_$T$_abs
			value = 100
		}
		lam_is_resolve_effect = {
			T = $T$
		}
	}
	# Ten years on, the side ahead takes it: clearly ahead for X settles for X, clearly ahead for Y goes to the lord
	# as if the bar had reached the end; a dead heat fades. Not while a faction with a stand on it is off in a rising
	# or a war: the issue waits for that outcome (its clock is moved on below; GPT review 2026-10-21).
	else_if = {
		limit = {
			var:lam_is_$T$_since <= {
				value = lam_now_m
				subtract = 120
			}
			NOT = {
				lam_is_held_trigger = {
					T = $T$
				}
			}
		}
		if = {
			limit = {
				var:lam_is_$T$_bar <= -25
			}
			lam_fac_stat_add_effect = {
				KEY = is_end_late_x
			}
			lam_is_win_x_effect = {
				T = $T$
			}
		}
		else_if = {
			limit = {
				var:lam_is_$T$_bar >= 25
			}
			lam_fac_stat_add_effect = {
				KEY = is_end_late_y
			}
			set_variable = {
				name = lam_is_$T$_bar
				value = 100
			}
			set_variable = {
				name = lam_is_$T$_disp
				value = 0
			}
			set_variable = {
				name = lam_is_$T$_abs
				value = 100
			}
			lam_is_resolve_effect = {
				T = $T$
			}
		}
		else = {
			lam_fac_stat_add_effect = {
				KEY = is_end_stale
			}
			lam_is_close_effect = {
				T = $T$
				KEY = lam_is_msg_$T$_stale
			}
		}
	}
}

lam_is_sum_effect = {
	scope:lam_is_lord = {
		set_variable = {
			name = lam_is_$T$_sa
			value = 0
		}
		set_variable = {
			name = lam_is_$T$_sb
			value = 0
		}
		set_variable = {
			name = lam_is_$T$_lw
			value = lam_is_$T$_lw_value
		}
		# the issue's cause, read from the realm (> 0 for Y), and its size for the card
		set_variable = {
			name = lam_is_$T$_cz
			value = lam_is_$T$_cause_value
		}
		set_variable = {
			name = lam_is_$T$_cza
			value = {
				value = var:lam_is_$T$_cz
				abs = yes
			}
		}
		lam_is_$T$_portraits_effect = yes
		every_in_list = {
			variable = lam_bloc_leads
			limit = {
				is_alive = yes
			}
			remove_variable = lam_is_st_$T$
			if = {
				limit = {
					lam_is_side_$T$_x_trigger = yes
				}
				set_variable = {
					name = lam_is_st_$T$
					value = flag:x
				}
				lam_is_st_mode_effect = yes
			}
			else_if = {
				limit = {
					lam_is_side_$T$_y_trigger = yes
				}
				set_variable = {
					name = lam_is_st_$T$
					value = flag:y
				}
				lam_is_st_mode_effect = yes
			}
		}
		every_in_list = {
			variable = lam_bloc_leads
			limit = {
				is_alive = yes
				has_variable = lam_fac_share
				lam_is_side_$T$_x_trigger = yes
			}
			save_scope_as = lam_is_backer
			scope:lam_is_lord = {
				change_variable = {
					name = lam_is_$T$_sa
					add = scope:lam_is_backer.var:lam_fac_share
				}
			}
		}
		every_in_list = {
			variable = lam_bloc_leads
			limit = {
				is_alive = yes
				has_variable = lam_fac_share
				lam_is_side_$T$_y_trigger = yes
			}
			save_scope_as = lam_is_backer
			scope:lam_is_lord = {
				change_variable = {
					name = lam_is_$T$_sb
					add = scope:lam_is_backer.var:lam_fac_share
				}
			}
		}
		# Unaligned vassals who took a side add their own share of the court's power (lam_is_T_ua / _ub).
		set_variable = {
			name = lam_is_$T$_ua
			value = 0
		}
		set_variable = {
			name = lam_is_$T$_ub
			value = 0
		}
		if = {
			limit = {
				has_variable = lam_bloc_total
				var:lam_bloc_total > 0
			}
			every_vassal = {
				limit = {
					lam_is_backer_trigger = {
						T = $T$
						SIDE = x
					}
				OR = {
					is_ai = no
					scope:lam_is_lord = {
						is_target_in_variable_list = {
							name = lam_fac_pool
							target = prev
						}
					}
				}
				}
				save_scope_value_as = {
					name = lam_is_uw
					value = {
						value = max_military_strength
						multiply = 100
						divide = scope:lam_is_lord.var:lam_bloc_total
					}
				}
				scope:lam_is_lord = {
					change_variable = {
						name = lam_is_$T$_ua
						add = scope:lam_is_uw
					}
				}
			}
			every_vassal = {
				limit = {
					lam_is_backer_trigger = {
						T = $T$
						SIDE = y
					}
				OR = {
					is_ai = no
					scope:lam_is_lord = {
						is_target_in_variable_list = {
							name = lam_fac_pool
							target = prev
						}
					}
				}
				}
				save_scope_value_as = {
					name = lam_is_uw
					value = {
						value = max_military_strength
						multiply = 100
						divide = scope:lam_is_lord.var:lam_bloc_total
					}
				}
				scope:lam_is_lord = {
					change_variable = {
						name = lam_is_$T$_ub
						add = scope:lam_is_uw
					}
				}
			}
		}
	}
}

# This = the lord, in the win effects: did the winners lead on power and cause when it was settled? (diagnostics:
# lam_stat_is_win_ahead / _upset). SIGN 1 = side Y won, -1 = side X won.
lam_is_fair_stat_effect = {
	if = {
		limit = {
			has_variable = lam_is_$T$_sa
			has_variable = lam_is_$T$_sb
			has_variable = lam_is_$T$_cz
		}
		save_scope_value_as = {
			name = lam_is_sc
			value = {
				value = var:lam_is_$T$_sb
				subtract = var:lam_is_$T$_sa
				add = var:lam_is_$T$_cz
				subtract = lam_is_$T$_lw_value
				multiply = $SIGN$
			}
		}
		if = {
			limit = {
				has_variable = lam_is_$T$_ub
				has_variable = lam_is_$T$_ua
			}
			save_scope_value_as = {
				name = lam_is_sc
				value = {
					value = var:lam_is_$T$_ub
					subtract = var:lam_is_$T$_ua
					multiply = $SIGN$
					add = scope:lam_is_sc
				}
			}
		}
		if = {
			limit = {
				scope:lam_is_sc > 0
			}
			lam_fac_stat_add_effect = {
				KEY = is_win_ahead
			}
		}
		else = {
			lam_fac_stat_add_effect = {
				KEY = is_win_upset
			}
		}
	}
}

# Yearly movement: power (factions and unaligned backers on Y minus those on X), the cause (CAUSE) and the crown's
# stake, capped at 20 a year; a court updated every season moves a quarter of that each time.
lam_is_drift_effect = {
	save_scope_value_as = {
		name = lam_is_d
		value = {
			value = var:lam_is_$T$_sb
			add = var:lam_is_$T$_ub
			subtract = var:lam_is_$T$_sa
			subtract = var:lam_is_$T$_ua
			subtract = lam_is_$T$_lw_value
			add = var:lam_is_$T$_cz
			multiply = 0.8
			min = -20
			max = 20
			if = {
				limit = {
					has_variable = lam_fac_fast
				}
				multiply = 0.25
			}
		}
	}
	if = {
		limit = {
			OR = {
				scope:lam_is_d >= 1
				scope:lam_is_d <= -1
			}
		}
		save_scope_as = lam_is_who
		lam_is_move_effect = {
			T = $T$
			AMOUNT = scope:lam_is_d
			KIND = drift
			WHO = scope:lam_is_who
		}
	}
	else = {
		lam_is_nudge_effect = {
			T = $T$
			AMOUNT = scope:lam_is_d
		}
	}
}
# A move too small to list: the bar still takes it.
lam_is_nudge_effect = {
	scope:lam_is_lord = {
		change_variable = {
			name = lam_is_$T$_bar
			add = $AMOUNT$
		}
		clamp_variable = {
			name = lam_is_$T$_bar
			min = -100
			max = 100
		}
		set_variable = {
			name = lam_is_$T$_disp
			value = {
				value = 100
				subtract = var:lam_is_$T$_bar
			}
		}
		set_variable = {
			name = lam_is_$T$_abs

lam_is_resolve_effect = {
	if = {
		limit = {
			var:lam_is_$T$_bar >= 100
		}
		if = {
			limit = {
				is_ai = no
			}
			if = {
				limit = {
					NOT = {
						has_variable = lam_is_$T$_pending
					}
					NOT = {
						has_variable = lam_is_evt_lock
					}
				}
				set_variable = {
					name = lam_is_$T$_pending
					value = yes
				}
				set_variable = {
					name = lam_is_evt_lock
					value = yes
				}
				set_variable = {
					name = lam_is_evt
					value = flag:$T$
				}
				trigger_event = lam_is.0001
			}
		}
		# An AI lord has the last word too: the stronger the crown, the likelier they refuse (crown 0: always gives
		# way, crown 100: half the time).
		else_if = {
			limit = {
				lam_is_$T$_can_concede_trigger = yes
			}
			random = {
				chance = {
					value = 100
					if = {
						limit = {
							has_variable = lam_crown
						}
						subtract = {
							value = var:lam_crown
							multiply = 0.5
						}
					}
				}
				set_variable = {
					name = lam_is_gave
					value = yes
				}
				lam_fac_stat_add_effect = {
					KEY = is_end_bar_y
				}
				lam_is_$T$_concede_effect = yes
			}
			if = {
				limit = {
					has_variable = lam_is_gave
				}
				remove_variable = lam_is_gave
			}
			else = {
				lam_is_refused_effect = {
					T = $T$
				}
			}
		}
		else = {
			lam_fac_stat_add_effect = {
				KEY = is_end_noconcede
			}
			lam_is_close_effect = {
				T = $T$
				KEY = lam_is_msg_$T$_void
			}
		}
	}
	else_if = {
		limit = {
			var:lam_is_$T$_bar <= -100
		}
		lam_fac_stat_add_effect = {
			KEY = is_end_bar_x
		}
		lam_is_$T$_x_bonus_effect = yes
		lam_is_win_x_effect = {
			T = $T$
		}
	}
}

	if = {
			limit = {
				any_character_war = { OR = { primary_attacker = scope:lam_is_lord primary_defender = scope:lam_is_lord } }
			}
			set_variable = { name = lam_is_war_mode value = flag:peace }
			random_character_war = {
				limit = { OR = { primary_attacker = scope:lam_is_lord primary_defender = scope:lam_is_lord } OR = { war_days >= 180 attacker_war_score >= 25 attacker_war_score <= -25 } }
				save_scope_as = lam_is_wr
			}
			if = {
				limit = { NOT = { exists = scope:lam_is_wr } }
				random_character_war = {
					limit = { OR = { primary_attacker = scope:lam_is_lord primary_defender = scope:lam_is_lord } }
					save_scope_as = lam_is_wr
				}
			}
			set_variable = { name = lam_is_war_ref value = scope:lam_is_wr }
			if = {
				limit = { scope:lam_is_wr = { primary_attacker = scope:lam_is_lord } }
				set_variable = { name = lam_is_war_side value = flag:attacker }
				set_variable = { name = lam_is_war_foe value = scope:lam_is_wr.primary_defender }
			}
			else = {
				set_variable = { name = lam_is_war_side value = flag:defender }
				set_variable = { name = lam_is_war_foe value = scope:lam_is_wr.primary_attacker }
			}
		}

		}

							name = lam_is_lean_for

					}

					value = flag:x

					KEY = is_stance_call
				}

					ACTION = stance
					TARGET = this
				}
			}
