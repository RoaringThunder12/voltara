lam_hi_update_memory_effect = {
	if = {
		limit = {
			has_variable = lam_gg_source
		}
		if = {
			limit = {
				lam_hi_memory_trigger = yes
			}
			if = {
				limit = {
					lam_gg_current_cause_trigger = yes
				}
				if = {
					limit = {
						has_variable = lam_hi_paused
					}
					if = {
						limit = {
							var:lam_hi_paused > {
								value = lam_now_m
								subtract = 6
							}
							has_variable = lam_ev_since
						}
						change_variable = {
							name = lam_ev_since
							add = {
								value = lam_now_m
								subtract = var:lam_hi_paused
							}
						}
						remove_variable = lam_hi_paused
					}
					else = {
						lam_ev_clear_person_effect = yes
					}
				}
			}
			else_if = {
				limit = {
					lam_hi_soft_pause_trigger = yes
				}
				if = {
					limit = {
						NOT = {
							has_variable = lam_hi_paused
						}
					}
					set_variable = {
						name = lam_hi_paused
						value = lam_now_m
					}
				}
				else_if = {
					limit = {
						var:lam_hi_paused <= {
							value = lam_now_m
							subtract = 6
						}
					}
					lam_ev_clear_person_effect = yes
				}
			}
			else = {
				lam_ev_clear_person_effect = yes
			}
		}
		else = {
			lam_ev_clear_person_effect = yes
		}
	}
}

			}
		}
		lam_er_observe_member_effect = yes
	}
}
lam_hi_shift_one_effect = {
	if = {
		limit = {
			has_variable = $FIELD$
		}
		if = {
			limit = {
				var:$FIELD$ = scope:lam_hi_old_record
			}
			if = {
				limit = {
					scope:lam_hi_old_record = scope:lam_gn_cut
				}
				remove_variable = $FIELD$
			}
			else_if = {
				limit = {
					scope:lam_hi_old_record > scope:lam_gn_cut
				}
				change_variable = {
					name = $FIELD$
					add = -1
				}
			}
		}
	}
}
lam_hi_watch_offer_effect = {
	set_variable = {
		name = lam_hi_offer_old_head
		value = scope:lam_cr_original_head
	}
	set_variable = {
		name = lam_hi_offer_new_head
		value = scope:lam_cr_split_head
	}
	scope:lam_cr_original_head = {
		if = {
			limit = {
				NOT = {
					is_target_in_variable_list = {
						name = lam_hi_record_watchers
						target = prev
					}
				}
			}
			add_to_variable_list = {
				name = lam_hi_record_watchers
				target = prev
			}
		}
	}
	scope:lam_cr_split_head = {
		if = {
			limit = {
				NOT = {
					is_target_in_variable_list = {
						name = lam_hi_record_watchers
						target = prev
					}
				}
			}
			add_to_variable_list = {
				name = lam_hi_record_watchers
				target = prev
			}
		}
	}
}
