voltara_assign_regional_government = {
 if = {
  limit = { is_alive = yes is_landed = yes highest_held_title_tier >= tier_county NOT = { voltara_is_ruins_placeholder = yes } NOT = { voltara_keeper_realm = yes } NOT = { has_trait = dragon } }
  if = { limit = { voltara_elven_government_culture = yes } change_government = celestial_government }
  else_if = { limit = { voltara_orc_government_culture = yes } change_government = mandala_government }
  else_if = {
   limit = {
    OR = { has_government = feudal_government has_government = lp_feudal_government }
    capital_county = { target_is_de_jure_liege_or_above = title:h_voltara_southorion }
   }
   change_government = tribal_government
  }
 }
}
voltara_assign_all_regional_governments = {
 every_living_character = { voltara_assign_regional_government = yes }
 set_global_variable = { name = voltara_regional_governments_initialized value = yes }
}
