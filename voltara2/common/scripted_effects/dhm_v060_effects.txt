# Current scope = marriage subject. Candidate and manager scopes are supplied by
# the existing ranked matcher. Only adult close-family matches open the connected
# courtship chain; every other legal match keeps the proven direct execution path.
dhm_v060_begin_match_effect = {
	save_scope_as = dhm_marriage_subject
	if = {
		limit = {
			exists = scope:dhm_manager
			exists = scope:dhm_marriage_candidate
			has_global_variable = dhm_setting_marriage_stories_all
			is_adult = yes
			dhm_v060_courtship_eligible_trigger = yes
			scope:dhm_marriage_candidate = { is_alive = yes is_adult = yes NOT = { has_character_flag = dhm_courtship_recent } }
			scope:dhm_manager = {
				is_alive = yes
				is_ai = no
				NOT = { has_character_flag = dhm_courtship_story_cooldown }
			}
			NOT = { has_global_variable = dhm_setting_marriage_stories_off }
		}
		set_variable = { name = dhm_courtship_reserved_for value = scope:dhm_marriage_candidate months = 6 }
		scope:dhm_marriage_candidate = { set_variable = { name = dhm_courtship_reserved_for value = scope:dhm_marriage_subject months = 6 } }
		save_scope_value_as = { name = dhm071_courtship_chain value = yes }
		add_character_flag = { flag = dhm_courtship_recent years = 5 }
		scope:dhm_marriage_candidate = { add_character_flag = { flag = dhm_courtship_recent years = 5 } }
		scope:dhm_manager = {
			add_character_flag = { flag = dhm_courtship_story_cooldown years = 2 }
			add_character_flag = dhm_v080_visible_story_scheduled_this_pass
			trigger_event = { id = dhm_saga.5000 days = { 3 8 } }
		}
	}
	else = { dhm_execute_match_effect = yes }
}

dhm_v060_initialize_character_effect = {
	add_character_flag = dhm_v060_character_initialized
	if = { limit = { NOT = { has_variable = dhm_heir_journey_stage } } set_variable = { name = dhm_heir_journey_stage value = 0 } }
}

dhm_v060_migration_effect = {
	# Additive migration: all v0.5 settings, counters, memories, relations and
	# cooldowns remain untouched.
	if = { limit = { NOT = { has_global_variable = dhm_chapter_named_count } } set_global_variable = { name = dhm_chapter_named_count value = 0 } }
	set_global_variable = { name = dhm_migration_0600 value = yes }
	dhm_v060_initialize_character_effect = yes
}

# A light identity echo makes established house values react to later stories.
# Every benefit carries an explicit burden from the same identity.
dhm_v060_legacy_echo_effect = {
	if = {
		limit = {
			NOT = { has_global_variable = dhm_setting_legacy_mechanics_disabled }
			OR = {
				has_global_variable = dhm_legacy_identity_scholarly
				has_global_variable = dhm_legacy_identity_honorable
				has_global_variable = dhm_legacy_identity_protective
				has_global_variable = dhm_legacy_identity_ambitious
				has_global_variable = dhm_legacy_identity_traditional
				has_global_variable = dhm_legacy_identity_severe
			}
		}
		custom_tooltip = dhm_v060_legacy_echo_tt
		if = {
			limit = { has_global_variable = dhm_legacy_identity_scholarly }
			add_prestige = minor_prestige_gain
			add_stress = miniscule_stress_gain
		}
		else_if = {
			limit = { has_global_variable = dhm_legacy_identity_honorable }
			dhm_v050_cohesion_gain_small_effect = yes
			add_prestige = minor_prestige_loss
		}
		else_if = {
			limit = { has_global_variable = dhm_legacy_identity_protective }
			add_prestige = minor_prestige_gain
			add_stress = miniscule_stress_gain
		}
		else_if = {
			limit = { has_global_variable = dhm_legacy_identity_ambitious }
			add_prestige = minor_prestige_gain
			dhm_v050_cohesion_loss_small_effect = yes
		}
		else_if = {
			limit = { has_global_variable = dhm_legacy_identity_traditional }
			add_piety = minor_piety_gain
			add_stress = miniscule_stress_gain
		}
		else_if = {
			limit = { has_global_variable = dhm_legacy_identity_severe }
			add_prestige = minor_prestige_gain
			dhm_v050_cohesion_loss_small_effect = yes
		}
	}
}

dhm_v060_maintain_household_effect = {
	save_scope_as = dhm_manager
	dhm_v050_cleanup_officers_effect = yes

	if = {
		limit = { NOT = { has_character_flag = dhm_v060_character_initialized } }
		dhm_v060_initialize_character_effect = yes
	}

	# Succession or a changed designated heir starts a fresh, save-safe journey.
	if = {
		limit = {
			exists = player_heir
			OR = {
				NOT = { has_variable = dhm_heir_journey_subject }
				NOT = { var:dhm_heir_journey_subject ?= player_heir }
			}
		}
		set_variable = { name = dhm_heir_journey_subject value = player_heir }
		set_variable = { name = dhm_heir_journey_stage value = 0 }
		remove_character_flag = dhm_heir_journey_completed
		remove_character_flag = dhm_heir_reassessment_pending
	}

	if = {
		limit = { NOT = { has_character_flag = dhm_v070_institution_maintenance_cooldown } }
		add_character_flag = { flag = dhm_v070_institution_maintenance_cooldown years = 1 }
		dhm_v050_maintain_household_effect = yes

		# Existing officers receive a neutral loyalty baseline. Personal treatment,
		# traits, friendships and rivalries move it thereafter.
		every_courtier = {
			limit = { dhm_v060_officer_trigger = yes NOT = { has_variable = dhm_officer_loyalty } }
			dhm_v071_initialize_officer_loyalty_effect = yes
		}
		every_courtier = {
			limit = { dhm_v060_officer_trigger = yes has_variable = dhm_officer_loyalty }
			dhm_v080_clamp_officer_loyalty_effect = yes
		}
		# Office loyalty is now earned through continuous service, while personal
		# career plans advance only for the bounded members of this court.
		dhm_v080_maintain_officer_tenures_effect = yes
		dhm_v080_maintain_career_plans_effect = yes

		# The death bridge is written only to close family and spouses. No world scan.
		every_close_family_member = {
			limit = { is_alive = yes }
			set_variable = { name = dhm_v060_household_manager value = scope:dhm_manager }
		}
		every_spouse = {
			limit = { is_alive = yes }
			set_variable = { name = dhm_v060_household_manager value = scope:dhm_manager }
		}
	}
}
