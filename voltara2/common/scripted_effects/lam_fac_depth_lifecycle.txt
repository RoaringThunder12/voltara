lam_pd_promise_result_effect = {
	set_variable = {
		name = lam_pd_last_result
		value = flag:$STATE$
	}
	set_variable = {
		name = lam_pd_last_m
		value = lam_now_m
	}
	if = {
		limit = {
			has_variable = lam_pd_promised_title
		}
		remove_variable = lam_pd_promised_title
	}
	if = {
		limit = {
			has_variable = lam_pd_promised_owner
		}
		remove_variable = lam_pd_promised_owner
	}
}

lam_pd_identity_pass_effect = {
	$FROM$ = {
		if = {
			limit = {
				has_variable = lam_pd_active
			}
			lam_pd_cancel_effect = yes
		}
		if = {
			limit = {
				has_variable = lam_pd_recruit_target
			}
			var:lam_pd_recruit_target ?= {
				lam_pd_recruit_clear_effect = yes
			}
		}
	}
	$TO$ = {
		if = {
			limit = {
				has_variable = lam_pd_goal_until
			}
			remove_variable = lam_pd_goal_until
		}
		if = {
			limit = {
				has_variable = lam_pd_plan_next
			}
			remove_variable = lam_pd_plan_next
		}
	}
	if = {
		limit = {
			$FROM$ = {
				has_variable = lam_pd_promised_title
			}
		}
		$TO$ = {
			set_variable = {
				name = lam_pd_promised_title
				value = $FROM$.var:lam_pd_promised_title
			}
			set_variable = {
				name = lam_pd_promised_owner
				value = $FROM$.var:lam_pd_promised_owner
			}
		}
	}
	$TO$ = {
		lam_pd_grievance_sync_effect = yes
	}
	lam_st_identity_pass_effect = {
		FROM = $FROM$
		TO = $TO$
	}
	lam_lf_identity_pass_effect = {
		FROM = $FROM$
		TO = $TO$
	}
	lam_bp_origin_pass_effect = {
		FROM = $FROM$
		TO = $TO$
	}
}
