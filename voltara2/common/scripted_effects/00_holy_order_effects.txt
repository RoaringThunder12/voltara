

send_child_to_holy_order_tooltip_effect = {
	scope:recipient = {
		add_trait = order_member
		add_piety_level = 1
		custom_tooltip = character_will_leave_court_holy_order
		if = {
			limit = {
				is_married = yes
			}
			every_spouse = {
				divorce = scope:recipient
			}
		}
		if = {
			limit = {
				exists = betrothed
			}
			break_betrothal = betrothed
			if = {
				limit = {
					has_been_promised_grand_wedding = yes
				}
				break_grand_wedding_betrothal_effect = yes
			}
		}
		if = {
			limit = {
				is_concubine = yes
			}
			this.concubinist = {
				remove_concubine = scope:recipient
			}
		}
		if = {
			limit = {
				number_of_concubines > 0
			}
			every_concubine = {
				scope:recipient = {
					remove_concubine = prev
				}
			}
		}
	}
}

send_child_to_holy_order_effect = {
	$GRANDMASTER$ = {
		add_courtier = scope:recipient

                scope:recipient = {
                    is_close_family_of = scope:actor
                    NOT = {
                    	has_trait = bastard
                    }
                }
            }
            add_opinion = {
                target = scope:actor
                modifier = pleased_opinion
                opinion = 25
            }
        }
        else_if = {
            limit = {
                scope:recipient = {
                    is_close_family_of = scope:actor
                }
            }
            add_opinion = {
                target = scope:actor
                modifier = pleased_opinion
                opinion = 15
            }
        }
	}
	scope:recipient = {
		add_trait = order_member
		add_piety_level = 1
		if = {
			limit = { is_ruler = yes }
			depose = yes
		}

				faith != $GRANDMASTER$.faith
			}
			set_character_faith = $GRANDMASTER$.faith
			set_character_rite = $GRANDMASTER$.rite
		}
		if = {
			limit = {
				is_married = yes
			}
			every_spouse = {
				divorce = scope:recipient
			}
		}
		if = {
			limit = {
				exists = betrothed
			}
			break_betrothal = betrothed
			if = {
				limit = {
					has_been_promised_grand_wedding = yes
				}
				break_grand_wedding_betrothal_effect = yes
			}
		}
		if = {
			limit = {
				is_concubine = yes
			}
			this.concubinist = {
				remove_concubine = scope:recipient
			}
		}
		if = {
			limit = {
				number_of_concubines > 0
			}
			every_concubine = {
				scope:recipient = {
					remove_concubine = prev
				}
			}
		}
	}
	take_vows_faith_opinion_effect = yes
	if = {
		limit = {
			$GRANDMASTER$ = {
				has_realm_law = cultural_holy_order_succession_law
				NOT = {
					culture = {
						has_same_culture_heritage = scope:recipient.culture
					}
				}
			}
		}

	}
}

send_child_to_clergy_effect = {
	scope:recipient = {
		if = {
			limit = {
				NOT = { has_character_flag = pam_simony_paid_vows }
			}
			if = {
				limit = {
					OR = {
						scope:demand ?= yes
						AND = {
							is_imprisoned_by = scope:puppet_or_actor
							OR = {
								is_heir_of = scope:puppet_or_actor
								any_claim = {
									exists = this
								}
							}
						}
					}
				}
				add_opinion = {
					modifier = forced_taking_vows
					target = scope:puppet_or_actor
				}
			}
			else = {
				add_opinion = {
					modifier = demanded_taking_vows
					target = scope:puppet_or_actor
				}
			}
		}
		simplified_clergy_effect = { RECIPIENT = scope:recipient }
	}
	scope:puppet_or_actor = {
		#Spiritual fulfillment gain
   		if = {
   			limit = {
   				scope:recipient = {
    	            is_child_of = scope:puppet_or_actor
    	            NOT = {
    	        		has_trait = bastard
    	        	}
    	        }
   			}
   			change_spiritual_fulfillment = {
   				value = medium_spiritual_fulfillment_value
   				if = {
   					limit = {
   						scope:recipient = {
   							is_imprisoned = yes
   						}
   					}
   					multiply = 0.5
   				}
   			}
   		}
   		else_if = {
   			limit = {
   				scope:recipient = {
    	            trigger_if = {
    	            	limit = {
    	            		has_trait = bastard
    	            	}
    	            	is_child_of = scope:puppet_or_actor
    	            }
    	            OR = {
    	            	house ?= scope:puppet_or_actor.house
    	            	is_close_family_of = scope:puppet_or_actor
    	            }
    	        }
   			}
   			change_spiritual_fulfillment = {
   				value = minor_spiritual_fulfillment_value
   				if = {
   					limit = {
   						scope:recipient = {
   							is_imprisoned = yes
   						}
   					}
   					multiply = 0.5
   				}
   			}
   		}
   		else_if = {
   			limit = {
   				scope:recipient = {
    	            trigger_if = {
    	            	limit = {
    	            		has_trait = bastard
    	            	}
    	            	OR = {
    	            		house ?= scope:puppet_or_actor.house
    	            		is_close_family_of = scope:puppet_or_actor
    	           		}
    	            }
    	            trigger_else = {
    	            	OR = {
    	            		dynasty ?= scope:puppet_or_actor.dynasty
    	            		is_close_or_extended_family_of = scope:puppet_or_actor
    	            	}
    	            }
    	        }
   			}
   			change_spiritual_fulfillment = {
   				value = miniscule_spiritual_fulfillment_value
   				if = {
   					limit = {
   						scope:recipient = {
   							is_imprisoned = yes
   						}
   					}
   					multiply = 0.5
   				}
   			}
   		}
   		else = {
   			change_spiritual_fulfillment = {
   				value = miniscule_spiritual_fulfillment_value
   				multiply = 0.5
   				if = {
   					limit = {
   						scope:recipient = {
   							is_imprisoned = yes
   						}
   					}
   					multiply = 0.5
   				}
   			}
   		}
	}
}

create_monastic_decision_effect = {
	save_scope_as = root_scope
	custom_tooltip = create_monastic_order_decision_effect_message_tenet
	custom_tooltip = create_monastic_order_decision_effect_message
	if = {
		limit = {
			OR = {
				has_realm_law = church_authority_1
				has_realm_law = church_authority_2
				has_realm_law = church_authority_3
			}
		}
		custom_tooltip = create_monastic_order_gain_hook
	}
	custom_tooltip = create_monastic_order_decision_effect_message_warning

	change_spiritual_fulfillment = medium_spiritual_fulfillment_value
	stress_impact = {
		zealous = minor_stress_impact_loss
		impatient = miniscule_stress_impact_gain
		patient = miniscule_stress_impact_loss
		calm = miniscule_stress_impact_loss
	}

	if = { # If you have no valid barony this will throw errors when you open the decision - we add this check to avoid that 
		limit = {
			exists = scope:title
		}
		hidden_effect = {
			scope:title = {
				ordered_county_province = {
					order_by = {
						value = 0
						if = {
							limit = {
								OR = {
									terrain = hills
									terrain = mountains
									terrain = desert_mountains
									terrain = desert
								}
							}
							add = 2
						}
						if = {
							limit = {
								OR = {
									terrain = forest
									terrain = drylands
									terrain = wetlands
									terrain = jungle
									terrain = terraced_hills
								}
							}
							add = 1
						}
					}
					limit = {
						has_holding = no
						has_ongoing_construction = no
					}
					begin_create_holding = church_holding
					save_scope_as = province
					set_variable = {
						name = monastic_order_capital
						value = root.faith
						years = 50
					}
				}
			}
		}
	}
	trigger_event = holy_order.2009
}

create_monastic_order_leader_and_notification_effect = {
	if = {
		limit = {
			exists = scope:chosen_leader
		}
		scope:chosen_leader = {
			save_scope_as = new_monastic_leader
		}
	}
	else = {
		scope:leader = { save_scope_as = new_monastic_leader }
	}
	hidden_effect = {
		if = { # set up currencies for the HO leader, has to be saved as scope:leader
			limit = {
				exists = scope:new_monastic_leader
			}
			scope:new_monastic_leader = {
				set_variable = founding_monastic
			}
			scope:new_holy_order.leader.primary_title = {
				set_variable = {
					name = founder
					value = scope:founder
				}
			}
			#If we have to transfer Order from default leader to someone else
			if = {
				limit = {
					scope:new_monastic_leader = {
						has_character_flag = created_char
					}
				}
				scope:new_monastic_leader = {
					remove_character_flag = created_char
				}
			}
			if = {
				limit = {
					NOT = {
						scope:new_holy_order.leader ?= scope:new_monastic_leader 
					}
				}
				scope:new_holy_order.leader.primary_title = {
					create_title_and_vassal_change = {
						type = granted
						save_scope_as = change
						add_claim_on_loss = yes
					}
					change_title_holder_include_vassals = {
						holder = scope:new_monastic_leader
						change = scope:change
					}
					resolve_title_and_vassal_change = scope:change
				}
				scope:new_monastic_leader = {
					every_courtier = {
						add_trait = devoted
					}
				}
				
			}
			
			scope:new_monastic_leader = {
				add_gold = 100 #So that they have some money to lend out
				add_piety_level = 2
				add_gold = holy_order_starting_gold
			}
			every_in_list = {
				list = grabbed_monks
				limit = {
					NOR = {
						this = scope:new_monastic_leader
						is_courtier_of = scope:new_monastic_leader
					}
				}
				scope:new_monastic_leader = { add_courtier = prev }
				remove_character_flag = created_char
			}
			
			# send messages
			send_interface_toast = {
				type = monastic_holy_order_founded_message
				desc = i_created_monastic_holy_order_message
				custom_tooltip = monastic_order_tenet_advocated_tt
				left_icon = scope:new_monastic_leader
				right_icon = scope:new_holy_order.title
			}
		}
		# send messages to relevant characters that HO got created
		save_scope_as = holy_order_creator
		every_player = {
			limit = {
				faith = scope:holy_order_creator.faith
				this != scope:holy_order_creator
			}
			send_interface_message = {
				type = monastic_holy_order_founded_message
				desc = someone_created_monastic_holy_order_message
				custom_tooltip = monastic_order_tenet_advocated_tt
				left_icon = scope:holy_order_creator
				right_icon = scope:new_holy_order.title
			}
		}
	}
}

create_monastic_order_founder_effect = {
	# gain opinion with your HoF, if you have one and it's not you
	if = {
		limit = {
			exists = faith.religious_head
			faith.religious_head != root
		}
		faith.religious_head = {
			add_opinion = {
				target = root
				modifier = founded_monastic_order_opinion
				opinion = 20
			}
		}
	}
	# get a nice modifier
	if = {
		limit = {
			NOT = { has_character_modifier = founded_monastic_order_modifier }
		}
		add_character_modifier = {
			modifier = founded_monastic_order_modifier
			years = 5
		}
	}
	# boost the fervor
	faith = {
		change_fervor = {
			value = notable_fervor_gain
			desc = fervor_gain_monastic_order_founded
		}
	}

	change_spiritual_fulfillment = minor_spiritual_fulfillment_value

	scope:leader = {
		save_scope_as = order_master
	}

	create_character_memory = {
		type = founded_monastic_order_memory
		participants = {
			order_master = scope:leader
		}
	}

	# PAM - Catalysts
	if = {
		limit = {
			has_pam_dlc_trigger = yes
			religion = religion:christianity_religion
			situation:the_christian_church ?= {
				situation_top_has_catalyst = catalyst_the_christian_church_founded_monastic_order
			}
		}
		situation:the_christian_church = {
			trigger_situation_catalyst = catalyst_the_christian_church_founded_monastic_order
		}
	}
}

				NOT = {
					has_relation_rival = scope:puppet_or_actor
				}
				OR = {
					ai_zeal <= low_negative_ai_value
					ai_greed >= low_positive_ai_value
					ai_vengefulness >= low_positive_ai_value
				}
				NOR = {
					has_trait = forgiving
					has_trait = craven
					has_trait = zealous
					has_trait = content
					has_trait = humble
					has_trait = bastard
				}
			}
			set_relation_rival = {
				target = scope:puppet_or_actor
				reason = rival_forced_to_take_vows
			}
		}
		else = {
			if = {
				limit = {
					#Bastards with decently matching personality don't see this as betrayal
					NAND = {
						has_trait = bastard
						ai_vengefulness < low_positive_ai_value
						ai_greed < low_positive_ai_value
						ai_zeal > low_negative_ai_value
					}
				}
				progress_towards_rival_effect = {
					CHARACTER = scope:puppet_or_actor
					REASON = rival_forced_to_take_vows
					OPINION = 0
				}
			}
		}
		every_parent = {
			limit = {
				NOR = {
					this = scope:puppet_or_actor
					#Doesn't have other kid in root's line succession
					any_child = {
						NOT = {
							this = scope:recipient
						}
						is_heir_of = scope:puppet_or_actor
					}
					has_trait = zealous
				}
			}
			add_opinion = {
				modifier = forced_child_take_vows
				target = scope:puppet_or_actor
			}
		}
	}
