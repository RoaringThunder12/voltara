VIET_huge_piety_gain_effect = {
	if = {
		limit = {
			has_game_rule = VIET_resource_twice
		}
		add_piety = 100
	}
	else_if = {
		limit = {
			has_game_rule = VIET_resource_one_and_a_half
		}
		add_piety = 75
	}
	else_if = {
		limit = {
			has_game_rule = VIET_resource_quadruple
		}
		add_piety = 200

	else = {
		add_piety = 50

}
