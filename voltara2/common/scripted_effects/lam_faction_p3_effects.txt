lam_fac_heirs_effect = {
	save_scope_as = lam_fh_lord
	clear_variable_list = lam_fh_cands
	every_child = {
		limit = {
			is_alive = yes
			NOT = {
				has_trait = disinherited
			}
		}
		scope:lam_fh_lord = {
			add_to_variable_list = {
				name = lam_fh_cands
				target = prev
			}
		}
	}
	primary_heir ?= {
		if = {
			limit = {
				NOT = {
					scope:lam_fh_lord = {
						is_target_in_variable_list = {
							name = lam_fh_cands
							target = prev
						}
					}
				}
			}
			scope:lam_fh_lord = {
				add_to_variable_list = {
					name = lam_fh_cands
					target = prev
				}
			}
		}
	}
	if = {
		limit = {
			any_in_list = {
				variable = lam_fh_cands
			}
		}
		every_in_list = {
			variable = lam_bloc_leads
			limit = {
				is_alive = yes
				NOT = {
					lam_wc_committed_trigger = yes
				}
			}
			save_scope_as = lam_fh_lead
   if = { limit = { vip_octavian_self_backing = yes }
    set_variable = { name = lam_fac_heir value = this }
    remove_variable = lam_fac_heir_lawful
   }
   else_if = {
				limit = {
					OR = {
						AND = {
							has_variable = lam_fac_heir_locked
							var:lam_fac_heir ?= {
								is_alive = yes
							}
						}
						AND = {
							is_ai = no
							has_variable = lam_fac_heir_pick
							var:lam_fac_heir ?= {
								is_alive = yes
								OR = {
									scope:lam_fh_lord = {
										is_target_in_variable_list = {
											name = lam_fh_cands
											target = prev
										}
									}
									has_claim_on = scope:lam_fh_lord.primary_title
								}
							}
						}
						AND = {
							has_variable = lam_fac_spurned
							var:lam_fac_heir ?= {
								is_alive = yes
								has_claim_on = scope:lam_fh_lord.primary_title
							}
						}
					}
				}
			}
			else = {
				lam_fac_pick_heir_effect = yes
			}
			if = {
				limit = {
					exists = scope:lam_fh_lord.primary_heir
					var:lam_fac_heir ?= scope:lam_fh_lord.primary_heir
				}
				set_variable = {
					name = lam_fac_heir_lawful
					value = yes
				}
			}
			else_if = {
				limit = {
					has_variable = lam_fac_heir_lawful
				}
				remove_variable = lam_fac_heir_lawful
			}
		}
		every_in_list = {
			variable = lam_fh_cands
			limit = {
				is_alive = yes
				has_variable = lam_fh_score
			}
			remove_variable = lam_fh_score
		}
	}
	else = {
		every_in_list = {
			variable = lam_bloc_leads
			limit = {
				is_alive = yes
				has_variable = lam_fac_heir
				NOT = {
					lam_wc_committed_trigger = yes
				}
			}
			remove_variable = lam_fac_heir
		}
	}
	clear_variable_list = lam_fh_cands
}
lam_fac_pick_heir_effect = {
	scope:lam_fh_lord = {
		every_in_list = {
			variable = lam_fh_cands
			limit = {
				is_alive = yes
			}
			save_scope_as = lam_fh_c
			set_variable = {
				name = lam_fh_score
				value = lam_fac_heir_score_value
			}
		}
		ordered_in_list = {
			variable = lam_fh_cands
			limit = {
				is_alive = yes
				has_variable = lam_fh_score
			}
			order_by = var:lam_fh_score
			save_scope_as = lam_fh_best
			max = 1
			check_range_bounds = no
		}
	}
	if = {
		limit = {
			var:lam_fac_heir ?= {
				is_alive = yes
				scope:lam_fh_lord = {
					is_target_in_variable_list = {
						name = lam_fh_cands
						target = prev
					}
				}
			}
		}
		var:lam_fac_heir = {
			save_scope_as = lam_fh_cur
		}
		if = {
			limit = {
				scope:lam_fh_cur = {
					has_variable = lam_fh_score
				}
				exists = scope:lam_fh_best
				scope:lam_fh_best = {
					has_variable = lam_fh_score
				}
			}
			save_scope_value_as = {
				name = lam_fh_bar
				value = {
					value = scope:lam_fh_cur.var:lam_fh_score
					add = lam_tune_heir_switch_bar
				}
			}
			if = {
				limit = {
					scope:lam_fh_best.var:lam_fh_score > scope:lam_fh_bar
					is_ai = yes
					NOT = {
						has_variable = lam_fac_heir_hold
					}
				}
				scope:lam_fh_cur = {
					save_scope_as = lam_vt_old
				}
				scope:lam_fh_best = {
					save_scope_as = lam_vt_new
				}
				lam_cr_start_effect = {
					ISSUE = heir
					ACTION = heir
					TARGET = scope:lam_fh_best
				}
			}
		}
	}
	else_if = {
		limit = {
			exists = scope:lam_fh_best
		}
		lam_cr_start_effect = {
			ISSUE = heir
			ACTION = heir
			TARGET = scope:lam_fh_best
		}
	}
}

					}

					name = lam_bloc

				}

					name = lam_fac_tier
					value = 3

lam_fac_blood_feud_effect = {
	$A$ = {
		save_scope_as = lam_bf_a
	}
	$B$ = {
		save_scope_as = lam_bf_b
	}
	if = {
		limit = {
			scope:lam_bf_a = {
				NOT = {
					has_variable = lam_fac_bf_cd
				}
			}
			scope:lam_bf_b = {
				NOT = {
					has_variable = lam_fac_bf_cd
				}
			}
		}
		scope:lam_bf_a = {
			set_variable = {
				name = lam_fac_bf_cd
				value = yes
				years = 5
			}
		}
		scope:lam_bf_b = {
			set_variable = {
				name = lam_fac_bf_cd
				value = yes
				years = 5
			}
		}
		scope:lam_bf_a = {
			if = {
				limit = {
					is_ai = no
				}
				scope:lam_bf_b = {
					save_scope_as = lam_bf_foe
				}
				trigger_event = lam_faction.3010
			}
			else_if = {
				limit = {
					can_start_scheme = {
						type = murder
						target_character = scope:lam_bf_b
					}
				}
				random = {
					chance = lam_fac_feud_murder_chance_value
					start_scheme = {
						type = murder
						target_character = scope:lam_bf_b
					}
				}
			}
		}
		scope:lam_bf_b = {
			if = {
				limit = {
					is_ai = no
				}
				scope:lam_bf_a = {
					save_scope_as = lam_bf_foe
				}
				trigger_event = lam_faction.3010
			}
			else_if = {
				limit = {
					can_start_scheme = {
						type = murder
						target_character = scope:lam_bf_a
					}
				}
				random = {
					chance = lam_fac_feud_murder_chance_value
					start_scheme = {
						type = murder
						target_character = scope:lam_bf_a
					}
				}
			}
		}
	}
