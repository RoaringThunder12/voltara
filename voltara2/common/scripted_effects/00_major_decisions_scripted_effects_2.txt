make_temporal_head_of_faith_title = {
	# Set up the remaining bits'n'bobs.
	if = {
		limit = {
			NOT = { has_realm_law = same_faith_succession_law }
		}
		add_realm_law_skip_effects = same_faith_succession_law
	}
	$TITLE$ = { add_title_law = temporal_head_of_faith_succession_law }
	# Make it the HoF head.
	faith = { set_religious_head_title = $TITLE$ }
	save_scope_as = hof_assigned
}

give_or_make_temporal_head_of_faith_title = {
	if = {
		limit = { NOT = { exists = scope:hof_assigned } }
		if = {
			limit = { has_title = $TITLE$ }
			make_temporal_head_of_faith_title = { TITLE = $TITLE$ }
		}
		else_if = {
			limit = {
				rite = $RITE$
				$TITLE$ = { is_title_created = no }
			}
			# Transfer the title.
			create_title_and_vassal_change = {
				type = created
				save_scope_as = change
				add_claim_on_loss = no
			}
			$TITLE$ = {
				change_title_holder = {
					holder = scope:founder
					change = scope:change
				}
			}
			resolve_title_and_vassal_change = scope:change
			make_temporal_head_of_faith_title = { TITLE = $TITLE$ }
		}
	}
}
