# Deliberate one-person offer, prepared once on event arrival, never scanned by a tooltip.

 }
}
dhm_v090_accept_mediator_effect = {
 if = {
  limit = { dhm_v090_mediator_offer_current_trigger = yes }
  add_stress = miniscule_stress_loss
  add_prestige = minor_prestige_loss
  scope:dhm090_mediator = {
   add_character_flag = { flag = dhm090_mediation_recent years = 5 }
   add_stress = minor_stress_gain
   add_opinion = { target = root modifier = dhm090_responsibility_opinion }
  }
  random_list = {
   60 = {
    modifier = { add = 20 scope:dhm090_mediator = { diplomacy >= 12 } }
    modifier = { add = 10 scope:dhm090_mediator = { has_trait = patient } }
    modifier = { add = -15 scope:dhm090_mediator = { has_trait = wrathful } }
    save_scope_value_as = { name = dhm072_dispute_result value = flag:reconciled }
    scope:dhm_dispute_first = {
     add_opinion = { target = scope:dhm_dispute_second modifier = dhm_mediated_family_dispute_opinion }
     add_opinion = { target = scope:dhm090_mediator modifier = dhm090_mediator_peace_opinion }
     create_character_memory = { type = dhm_family_reconciliation_memory participants = { reconciled_relative = scope:dhm_dispute_second } }
    }
    scope:dhm_dispute_second = {
     add_opinion = { target = scope:dhm_dispute_first modifier = dhm_mediated_family_dispute_opinion }
     add_opinion = { target = scope:dhm090_mediator modifier = dhm090_mediator_peace_opinion }
     create_character_memory = { type = dhm_family_reconciliation_memory participants = { reconciled_relative = scope:dhm_dispute_first } }
    }
    save_scope_value_as = { name = dhm090_mediator_result value = flag:peace }
    dhm_v090_rc2_record_mediated_peace_effect = yes
   }
   40 = {
    save_scope_value_as = { name = dhm072_dispute_result value = flag:grievance_both }
    scope:dhm_dispute_first = { add_opinion = { target = scope:dhm090_mediator modifier = dhm090_mediator_blame_opinion } }
    scope:dhm_dispute_second = { add_opinion = { target = scope:dhm090_mediator modifier = dhm090_mediator_blame_opinion } }
    save_scope_value_as = { name = dhm090_mediator_result value = flag:blame }
   }
  }
  trigger_event = { id = dhm_living_family.0410 years = 2 }
 }
 else = { custom_tooltip = dhm090_offer_lapsed_tt }
}

# A spouse proposes help for somebody they actually like, not an invented victim.
dhm_v090_prepare_spouse_plan_effect = {
 clear_saved_scope = dhm090_overworked_relative
 if = {
  limit = { exists = scope:dhm_partner_spouse scope:dhm_partner_spouse = { is_alive = yes is_spouse_of = root is_in_the_same_court_as = root is_imprisoned = no is_hostage = no NOT = { has_trait = incapable } } }
  random_close_family_member = {
   limit = {
    is_alive = yes is_adult = yes is_in_the_same_court_as = root
    is_imprisoned = no is_hostage = no NOT = { has_trait = incapable }
    this != root this != scope:dhm_partner_spouse
    stress_level >= 1
    scope:dhm_partner_spouse = { opinion = { target = prev value >= 20 } }
   }
   save_scope_as = dhm090_overworked_relative
  }
 }
}
dhm_v090_apply_spouse_plan_effect = {
 save_scope_value_as = { name = dhm090_plan_approach value = $APPROACH$ }
 if = {
  limit = {
   exists = scope:dhm090_overworked_relative exists = scope:dhm_partner_spouse
   scope:dhm_partner_spouse = { is_alive = yes is_spouse_of = root is_in_the_same_court_as = root is_imprisoned = no is_hostage = no NOT = { has_trait = incapable } }
   scope:dhm090_overworked_relative = { is_alive = yes is_in_the_same_court_as = root is_imprisoned = no is_hostage = no NOT = { has_trait = incapable } }
  }
  scope:dhm090_overworked_relative = {
   if = {
    limit = { scope:dhm090_plan_approach = flag:command }
    add_opinion = { target = root modifier = dhm_parent_demanded_duty_opinion }
    add_stress = miniscule_stress_gain
   }
   else = {
    add_stress = minor_stress_loss
    dhm_v090_remember_help_effect = yes
    if = { limit = { scope:dhm090_plan_approach = flag:yield } add_opinion = { target = scope:dhm_partner_spouse modifier = dhm_spouse_consulted_opinion } }
    else = { add_opinion = { target = root modifier = dhm_shared_family_burden_opinion } }
   }
  }
 }
}

# Exactly one new keepsake in this single-player campaign. Native inventory handles
# equipment, gifting, damage and succession; changing ruler never resets eligibility.

    name = dhm090_school_keepsake_name description = dhm090_school_keepsake_desc
    type = miscellaneous visuals = small_box
    modifier = artifact_monthly_prestige_1_modifier
    save_scope_as = dhm090_keepsake
   }
  }
  else_if = {
   limit = { dhm_v090_private_triumph_trigger = yes }
   create_artifact = {
    name = dhm090_peace_keepsake_name description = dhm090_peace_keepsake_desc
    type = miscellaneous visuals = small_box
    modifier = artifact_monthly_prestige_1_modifier
    save_scope_as = dhm090_keepsake
   }
  }
  else = {
   create_artifact = {
    name = dhm090_chronicle_keepsake_name description = dhm090_chronicle_keepsake_desc
    type = miscellaneous visuals = small_box
    modifier = artifact_monthly_prestige_1_modifier
    save_scope_as = dhm090_keepsake
   }
  }
  if = {
   limit = { exists = scope:dhm090_keepsake }
   set_global_variable = { name = dhm090_keepsake_awarded value = yes }
   custom_tooltip = dhm090_keepsake_awarded_tt
