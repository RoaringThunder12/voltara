voltara_elven_immortality_effect = {
 if = {
  limit = { is_immortal = yes NOT = { has_trait = voltara_lifespan } NOT = { has_character_flag = voltara_artificial_lifespan } }
  add_character_flag = voltara_natural_lifespan_0
  add_character_flag = voltara_artificial_lifespan
 }

 if = {
  limit = { is_immortal = yes has_trait = voltara_lifespan }
  if = { limit = { has_trait = voltara_lifespan_1 } add_character_flag = voltara_natural_lifespan_1 remove_trait = voltara_lifespan_1 }
  if = { limit = { has_trait = voltara_lifespan_2 } add_character_flag = voltara_natural_lifespan_2 remove_trait = voltara_lifespan_2 }
  if = { limit = { has_trait = voltara_lifespan_3 } add_character_flag = voltara_natural_lifespan_3 remove_trait = voltara_lifespan_3 }
  if = { limit = { has_trait = voltara_lifespan_4 } add_character_flag = voltara_natural_lifespan_4 remove_trait = voltara_lifespan_4 }
  add_character_flag = voltara_artificial_lifespan
 }
}
