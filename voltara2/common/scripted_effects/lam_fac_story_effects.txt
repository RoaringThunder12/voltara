lam_st_campaign_pick_effect = {
	lam_st_campaign_clear_effect = yes
	set_variable = {
		name = lam_st_campaign_score
		value = 9
	}
	if = {
		limit = {
			has_variable = lam_pd_gr_office
			var:lam_pd_gr_office > var:lam_st_campaign_score
			NOT = {
				is_councillor_of = liege
			}
			lam_st_office_available_trigger = yes
		}
		set_variable = {
			name = lam_st_campaign_kind
			value = flag:council
		}
		set_variable = {
			name = lam_st_campaign_score
			value = var:lam_pd_gr_office
		}
	}
	if = {
		limit = {
			has_variable = lam_pd_gr_wealth
			var:lam_pd_gr_wealth > var:lam_st_campaign_score
		}
		set_variable = {
			name = lam_st_campaign_kind
			value = flag:gold
		}
		set_variable = {
			name = lam_st_campaign_score
			value = var:lam_pd_gr_wealth
		}
	}
	save_scope_as = lam_st_candidate_head
	if = {
		limit = {
			has_variable = lam_pd_gr_succession
			var:lam_pd_gr_succession > var:lam_st_campaign_score
			has_variable = lam_fac_heir
			liege = {
				lam_nc_heir_trigger = {
					CANDIDATE = scope:lam_st_candidate_head.var:lam_fac_heir
				}
			}
		}
		set_variable = {
			name = lam_st_campaign_kind
			value = flag:heir
		}
		set_variable = {
			name = lam_st_campaign_target
			value = var:lam_fac_heir
		}
		set_variable = {
			name = lam_st_campaign_score
			value = var:lam_pd_gr_succession
		}
	}
	if = {
		limit = {
			has_variable = lam_st_campaign_kind
		}
		set_variable = {
			name = lam_st_campaign_lord
			value = liege
		}
		set_variable = {
			name = lam_st_campaign_until
			value = {
				value = lam_now_m
				add = 24
			}
		}
		if = {
			limit = {
				has_variable = lam_fac_rec
			}
			set_variable = {
				name = lam_st_campaign_record
				value = var:lam_fac_rec
			}
		}
	}
}
lam_st_campaign_plan_effect = {
	if = {
		limit = {
			has_variable = lam_st_campaign_kind
			NOT = {
				lam_st_campaign_current_trigger = yes
			}
		}
		lam_st_campaign_clear_effect = yes
	}
	if = {
		limit = {
			NOT = {
				has_variable = lam_st_campaign_kind
			}
		}
		lam_st_campaign_pick_effect = yes
	}
	if = {
		limit = {
			lam_st_campaign_current_trigger = yes
		}
		set_variable = {
			name = lam_pd_plan
			value = flag:wait
		}
		if = {
			limit = {
				NOT = {
					has_variable = lam_head_demand_cd
				}
				NOT = {
					has_variable = lam_fac_promise
				}
				OR = {
					NOT = {
						has_variable = lam_st_retry_after
					}
					var:lam_st_retry_after <= lam_now_m
				}
			}
			if = {
				limit = {
					NOT = {
						has_variable = lam_st_campaign_prepared
					}
					NOT = {
						var:lam_st_campaign_kind ?= flag:gold
					}
				}
				set_variable = {
					name = lam_st_campaign_prepared
					value = yes
				}
				if = {
					limit = {
						gold >= 150
						var:lam_fac_cohesion < 45
						NOT = {
							has_variable = lam_pd_aid_cd
						}
						any_in_list = {
							variable = lam_fac_members
							is_alive = yes
							var:lam_bloc ?= prev
							gold < 0
						}
					}
					set_variable = {
						name = lam_pd_plan
						value = flag:aid
					}
					lam_pd_aid_member_effect = yes
				}
				else_if = {
					limit = {
						var:lam_fac_count < 3
						gold >= 100
						lam_pd_recruit_needed_trigger = yes
						NOT = {
							has_variable = lam_fac_recruit_cd
						}
					}
					set_variable = {
						name = lam_pd_plan
						value = flag:recruit
					}
					lam_pd_recruit_search_effect = yes
				}
				else_if = {
					limit = {
						NOT = {
							has_variable = lam_pd_pact_partner
						}
					}
					set_variable = {
						name = lam_pd_plan
						value = flag:pact
					}
					lam_pd_pact_search_effect = yes
				}
			}
			else = {
				set_variable = {
					name = lam_pd_plan
					value = flag:petition
				}
				set_variable = {
					name = lam_st_retry_after
					value = {
						value = lam_now_m
						add = 12
					}
				}
				if = {
					limit = {
						var:lam_st_campaign_kind ?= flag:council
					}
					lam_pd_begin_effect = {
						KIND = council
					}
				}
				else_if = {
					limit = {
						var:lam_st_campaign_kind ?= flag:heir
					}
					lam_pd_begin_effect = {
						KIND = heir
					}
				}
				else_if = {
					limit = {
						var:lam_st_campaign_kind ?= flag:gold
					}
					lam_pd_begin_effect = {
						KIND = gold
					}
				}
			}
		}
	}
	else = {
		lam_st_opportunistic_plan_effect = yes
	}
}
lam_st_share_rewards_effect = {
	save_scope_as = lam_sg_lead
	save_scope_value_as = {
		name = lam_st_award_original
		value = $GOLD$
	}
	save_scope_value_as = {
		name = lam_st_award_pool
		value = {
			value = $GOLD$
			min = 0
			max = {
				value = gold
				min = 0
			}
			multiply = lam_st_award_fraction_value
		}
	}
	set_variable = {
		name = lam_st_award_weight
		value = 0
	}
	every_in_list = {
		variable = lam_fac_members
		limit = {
			lam_st_reward_member_trigger = yes
		}
		scope:lam_sg_lead = {
			change_variable = {
				name = lam_st_award_weight
				add = prev.lam_st_reward_weight_value
			}
		}
	}
	if = {
		limit = {
			scope:lam_st_award_pool > 0
			AND = {
				trigger_if = {
					limit = {
						has_variable = lam_st_award_weight
					}
					var:lam_st_award_weight > 0
				}
				trigger_else = {
					always = no
				}
			}
		}
		every_in_list = {
			variable = lam_fac_members
			limit = {
				lam_st_reward_member_trigger = yes
			}
			save_scope_value_as = {
				name = lam_st_member_reward
				value = {
					value = scope:lam_st_award_pool
					multiply = lam_st_reward_weight_value
					divide = scope:lam_sg_lead.var:lam_st_award_weight
				}
			}
			scope:lam_sg_lead = {
				pay_short_term_gold = {
					target = prev
					gold = scope:lam_st_member_reward
				}
			}
			set_variable = {
				name = lam_st_reward_head
				value = scope:lam_sg_lead
			}
			if = {
				limit = {
					scope:lam_sg_lead = {
						lam_st_award_fraction_value >= 0.6
					}
				}
				remove_variable = lam_st_overlooked
				add_opinion = {
					target = scope:lam_sg_lead
					modifier = lam_fac_fair_share_opinion
				}
			}
			else = {
				if = {
					limit = {
						NOT = {
							has_variable = lam_st_overlooked
						}
					}
					set_variable = {
						name = lam_st_overlooked
						value = 0
					}
				}
				change_variable = {
					name = lam_st_overlooked
					add = 1
				}
				set_variable = {
					name = lam_st_overlooked
					value = var:lam_st_overlooked
					years = 3
				}
				clamp_variable = {
					name = lam_st_overlooked
					min = 0
					max = 3
				}
				add_opinion = {
					target = scope:lam_sg_lead
					modifier = lam_fac_passed_over_opinion
				}
			}
		}
		if = {
			limit = {
				lam_st_award_fraction_value >= 0.6
			}
			remove_variable = lam_fac_hoarded
			set_variable = {
				name = lam_fac_share_style
				value = flag:fair
			}
			lam_fac_cohesion_add_effect = {
				ADD = 10
			}
			if = {
				limit = {
					scope:lam_st_award_original >= medium_gold_value
					NOT = {
						has_variable = lam_st_reward_record_cd
					}
				}
				lam_st_record_effect = {
					KEY = spoils_shared
				}
				set_variable = {
					name = lam_st_reward_record_cd
					value = yes
					years = 2
				}
			}
		}
		else = {
			set_variable = {
				name = lam_fac_hoarded
				value = yes
				years = 3
			}
			set_variable = {
				name = lam_fac_share_style
				value = flag:hoard
			}
			lam_fac_cohesion_add_effect = {
				ADD = -10
			}
			if = {
				limit = {
					scope:lam_st_award_original >= medium_gold_value
					NOT = {
						has_variable = lam_st_reward_record_cd
					}
				}
				lam_st_record_effect = {
					KEY = spoils_hoarded
				}
				set_variable = {
					name = lam_st_reward_record_cd
					value = yes
					years = 2
				}
			}
		}
	}
	remove_variable = lam_st_award_weight
}

				lam_cr_saved_voter_trigger = yes

				var:lam_bloc ?= scope:lam_st_reconcile_head
				var:lam_cr_unrest_head ?= scope:lam_st_reconcile_head
				var:lam_cr_unrest >= 2
				NOT = {
					this = scope:lam_st_reconcile_head
				}
				NOT = {
					has_variable = lam_bloc_lead
				}
			}
			save_scope_as = lam_st_dissenter
		}
		if = {
			limit = {
				exists = scope:lam_st_dissenter
			}
			set_variable = {
				name = lam_st_reconcile_member
				value = scope:lam_st_dissenter
			}
			set_variable = {
				name = lam_st_reconcile_open
				value = yes
				months = 2
			}
			set_variable = {
				name = lam_st_reconcile_cd
				value = yes
				years = 2
			}
			set_variable = {
				name = lam_st_reconcile_lord
				value = liege
			}
			set_variable = {
				name = lam_st_reconcile_id
				value = var:lam_cr_id
			}
			if = {
				limit = {
					has_variable = lam_fac_rec
				}
				set_variable = {
					name = lam_st_reconcile_record
					value = var:lam_fac_rec
				}
			}
			save_scope_value_as = {
				name = lam_st_reconcile_token
				value = var:lam_st_reconcile_id
			}
			if = {
				limit = {
					is_ai = no
				}
				trigger_event = lam_st.0100
			}
			else_if = {
				limit = {
					prestige >= 150
					NOT = {
						has_trait = arrogant
					}
					OR = {
						var:lam_fac_cohesion < 50
						has_trait = humble
						has_trait = compassionate
					}
				}
				lam_st_reconcile_effect = yes
			}
			else = {
				remove_variable = lam_st_reconcile_open
			}
		}
	}
}
