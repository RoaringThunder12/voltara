## Very expensive
# Scope : county
# REASON : flag containing the update reason
E_kCAFG_update_county_rite = {
	hidden_effect = {
        save_temporary_scope_as = kCAFG_county

        # Set the rite to the highest total rite level
        ordered_in_list = {
            variable = kCAFG_county_rites
            order_by = {
                value = 0
                save_temporary_scope_as = kCAFG_rite # used by V_kCAFG_province_rite_amount
                scope:kCAFG_county = {
                    every_county_province = { add = V_kCAFG_province_rite_amount }
                }
            }
            position = 0

            if = {
                limit = {
                    scope:kCAFG_county.rite != this
                }

                scope:kCAFG_county = {
                    if = { 
                        limit = { NOT = { has_variable = kCAFG_county_faith_change_reason } }
                        set_variable = { name = kCAFG_county_faith_change_reason value = $REASON$ }
                    }
                    set_county_faith = prev.faith
                    if = {
                        limit = { NOT = { has_variable = kCAFG_county_rite_change_reason } }
                        set_variable = { name = kCAFG_county_rite_change_reason value = $REASON$ }
                    }
                    set_county_rite = prev
                }
            }
        }
    }
}


# Scope : county
E_kCAFG_update_county_rites_list = {
	hidden_effect = {
        save_temporary_scope_as = E_kCAFG_update_county_rites_list_COUNTY
        clear_variable_list = kCAFG_county_rites
        clear_variable_list = kCAFG_county_faiths
        every_county_province = {
            every_in_list = {
                variable = kCAFG_province_rites
                scope:E_kCAFG_update_county_rites_list_COUNTY = {
                    add_to_variable_list = {
                        name = kCAFG_county_rites
                        target = prev
                    }
                    if = {
                        limit = {
                            NOT = {
                                is_target_in_variable_list = {
                                    name = kCAFG_county_faiths
                                    target = prev.faith
                                }
                            }
                        }
                        add_to_variable_list = {
                            name = kCAFG_county_faiths
                            target = prev.faith
                        }
                    }
                }
            }
        }
    }
}

# Completely replace a county rite with 100% of the chosen rite
# Scope : county
# RITE 
E_kCAFG_replace_county_rite = {
    $RITE$ = { save_temporary_scope_as = E_kCAFG_replace_county_rite_target_rite } # used by localization
    custom_tooltip = {
		text = kCAFG_E_replace_county_rite_tt
        E_kCAFG_clear_all_county_rites = yes
        every_county_province = {
            E_kCAFG_set_maximum_province_rite_factor = {
                RITE = $RITE$
            }
        }
        add_to_variable_list = {
            name = kCAFG_county_rites
            target = $RITE$
        }
        add_to_variable_list = {
            name = kCAFG_county_faiths
            target = $RITE$.faith
        }
        set_variable = { name = kCAFG_county_faith_change_reason value = flag:kCAFG_replacement } 
        set_county_faith = $RITE$.faith
        set_variable = { name = kCAFG_county_rite_change_reason value = flag:kCAFG_replacement } 
        set_county_rite = $RITE$
    }
}

# Scope : county 
# FACTOR : base factor conversion
# VARIANCE : randomization added to the base factor for each province
# RITE : rite to convert to
# MAX : maximum factor in province (i.e. keep minorities)
E_kCAFG_convert_rite_in_all_county_provinces = {
    $RITE$ = { save_temporary_scope_as = E_kCAFG_convert_rite_in_all_county_provinces_converting_rite } # used by localization
    custom_tooltip = {
		text = kCAFG_E_convert_rite_in_all_county_provinces_tt
        every_county_province = {
            E_kCAFG_province_randomized_weighted_rite_conversion = {
                FACTOR = $FACTOR$
                VARIANCE = $VARIANCE$
                RITE = $RITE$
                MAX = $MAX$
            }
        }
        E_kCAFG_update_county_rites_list = yes
        E_kCAFG_update_county_rite = { REASON = flag:kCAFG_default_reason }
    }
}
