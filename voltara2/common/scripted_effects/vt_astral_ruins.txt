vt_forge_astral_weapon_effect = {
 if = {
  limit = { vt_can_forge_astral_trigger = yes }
  random_character_artifact = { limit = { has_variable = vt_astral_steel } save_scope_as = vt_forging_material }
if = {
 limit = { var:vt_astral_design = 1 }
 vt_create_astral_weapon_1_effect = yes
}
else_if = {
 limit = { var:vt_astral_design = 2 }
 vt_create_astral_weapon_2_effect = yes
}
else_if = {
 limit = { var:vt_astral_design = 3 }
 vt_create_astral_weapon_3_effect = yes
}
else_if = {
 limit = { var:vt_astral_design = 4 }
 vt_create_astral_weapon_4_effect = yes
}
else_if = {
 limit = { var:vt_astral_design = 5 }
 vt_create_astral_weapon_5_effect = yes
}
else_if = {
 limit = { var:vt_astral_design = 6 }
 vt_create_astral_weapon_6_effect = yes
}
else_if = {
 limit = { var:vt_astral_design = 7 }
 vt_create_astral_weapon_7_effect = yes
}
else_if = {
 limit = { var:vt_astral_design = 8 }
 vt_create_astral_weapon_8_effect = yes
}
else_if = {
 limit = { var:vt_astral_design = 9 }
 vt_create_astral_weapon_9_effect = yes
}
else_if = {
 limit = { var:vt_astral_design = 10 }
 vt_create_astral_weapon_10_effect = yes
}
else_if = {
 limit = { var:vt_astral_design = 11 }
 vt_create_astral_weapon_11_effect = yes
}
else_if = {
 limit = { var:vt_astral_design = 12 }
 vt_create_astral_weapon_12_effect = yes
}
else_if = {
 limit = { var:vt_astral_design = 13 }
 vt_create_astral_weapon_13_effect = yes
}
else_if = {
 limit = { var:vt_astral_design = 14 }
 vt_create_astral_weapon_14_effect = yes
}
  else = { vt_create_astral_weapon_1_effect = yes }
  if = {
   limit = { exists = scope:vt_new_astral_weapon }
   remove_short_term_gold = 150
   destroy_artifact = scope:vt_forging_material
   if = {
    limit = { has_character_flag = vt_astral_naming }
    if = {
     limit = { has_variable = new_artifact_name }
     scope:vt_new_astral_weapon = { set_artifact_name = rename_artifact_default_key }
    }
    else = { scope:vt_new_astral_weapon = { set_artifact_name = rename_artifact_entered_text } }
   }
  }
 }
}
vt_astral_menu_cleanup_effect = {
 remove_variable = vt_astral_design
 remove_variable = new_artifact_name
 remove_character_flag = vt_astral_naming
 remove_character_flag = vt_astral_menu_open
}
