lam_pb_council_mandate_effect = {
	save_scope_as = lam_pb_council_head
	clear_saved_scope = lam_pb_mandated_host
	if = {
		limit = {
			exists = var:lam_wc_plan_title
			exists = liege
			AND = {
				trigger_if = {
					limit = {
						exists = liege.primary_title
					}
					var:lam_wc_plan_title ?= liege.primary_title
				}
				trigger_else = {
					always = no
				}
			}
		}
		if = {
			limit = {
				var:lam_cr_choice ?= flag:yes
				lam_pb_can_propose_trigger = yes
			}
			save_scope_as = lam_pb_mandated_host
		}
		else = {
			ordered_in_list = {
				variable = lam_cr_voters
				order_by = lam_cr_influence_value
				max = 1
				check_range_bounds = no
				limit = {
					is_alive = yes
					var:lam_cr_choice ?= flag:yes
					lam_pb_can_propose_trigger = yes
				}
				save_scope_as = lam_pb_mandated_host
			}
		}
		if = {
			limit = {
				exists = scope:lam_pb_mandated_host
				has_variable = lam_pb_council_kind
			}
			scope:lam_pb_mandated_host = {
				set_variable = {
					name = lam_pb_origin_party
					value = scope:lam_pb_council_head
				}
				set_variable = {
					name = lam_pb_selected_kind
					value = scope:lam_pb_council_head.var:lam_pb_council_kind
				}
				lam_rv_kind_fallback_effect = yes
				remove_variable = lam_pb_claimant
				if = {
					limit = {
						scope:lam_pb_council_head = {
							exists = var:lam_wc_plan_candidate
						}
					}
					set_variable = {
						name = lam_pb_claimant
						value = scope:lam_pb_council_head.var:lam_wc_plan_candidate
					}
				}
				lam_pb_dispatch_effect = yes
				remove_variable = lam_pb_selected_kind
				if = {
					limit = {
						NOT = {
							has_variable = lam_pb_gathering
						}
					}
					lam_pb_request_cleanup_effect = yes
				}
			}
		}
	}
}
lam_pb_player_proposal_effect = {
	if = {
		limit = {
			lam_pb_can_propose_trigger = yes
		}
		if = {
			limit = {
				has_variable = lam_bloc_lead
			}
			set_variable = {
				name = lam_pb_council_kind
				value = flag:$KIND$
			}
			if = {
				limit = {
					exists = var:lam_pb_claimant
				}
				set_variable = {
					name = lam_pb_selected_claimant
					value = var:lam_pb_claimant
				}
			}
			lam_cr_start_effect = {
				ISSUE = rise
				ACTION = rise
				TARGET = this
			}
			remove_variable = lam_pb_selected_claimant
			if = {
				limit = {
					NOT = {
						has_variable = lam_cr_active
					}
				}
				remove_variable = lam_pb_council_kind
			}
		}
		else = {
			lam_pb_begin_effect = {
				KIND = $KIND$
			}
		}
	}
}
