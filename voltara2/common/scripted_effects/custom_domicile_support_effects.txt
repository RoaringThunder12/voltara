#ADVENTURER TITLES
#Scope: character
grant_domicile_title_effect = {
	save_scope_as = character
	create_adventurer_title = {
		name = voltara_civic_estate_title
		holder = scope:character
		#article = DEFAULT_TITLE_NAME_ARTICLE
		government = voltara_colonial_government
		save_scope_as = domicile_title
	}
	set_variable = {
		name = domicile_title
		value = scope:domicile_title
	}
	scope:domicile_title = {
		set_variable = {
			name = custom_domicile_title
			value = yes
		}
	}
}

#Scope: character
recreate_domicile_title_effect = {
	save_scope_as = character
	if = {
		limit = { has_variable = domicile_title }
		create_title_and_vassal_change = {
			type = created
			save_scope_as = title_change
			add_claim_on_loss = no
		}
		root.var:domicile_title = {
			change_title_holder = {
				holder = scope:character
				change = scope:title_change
			}
		}
		resolve_title_and_vassal_change = scope:title_change
	}
}

grant_domicile_title_nomad_effect = {

 save_scope_as = civic_estate
