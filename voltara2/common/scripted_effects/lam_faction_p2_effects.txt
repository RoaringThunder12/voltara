lam_fac_rally_effect = {
	if = {
		limit = {
			exists = scope:defender
			exists = scope:attacker
			exists = scope:war
		}
		if = {
			limit = {
				scope:defender = {
					is_alive = yes
				}
				scope:attacker = {
					is_alive = yes
				}
			}
			if = {
				limit = {
					exists = scope:defender
					exists = scope:attacker
					exists = scope:war
					scope:defender = {
						any_in_list = {
							variable = lam_bloc_leads
						}
					}
					OR = {
						scope:war = {
							using_civil_war_cb_trigger = yes
						}
						scope:attacker = {
							OR = {
								has_variable = lam_fac_armed
								var:lam_bloc ?= {
									var:lam_fac_stage ?= 4
								}
							}
						}
					}
				}
				scope:defender = {
					save_scope_as = lam_rally_lord
					every_in_list = {
						variable = lam_bloc_leads
						limit = {
							is_alive = yes
							is_ai = yes
							liege ?= scope:lam_rally_lord
							NOT = {
								this = scope:attacker
							}
							NOT = {
								has_variable = lam_fac_hide
							}
							NOT = {
								has_variable = lam_fac_armed
							}
							NOT = {
								is_at_war_with = scope:lam_rally_lord
							}
							NOT = {
								joined_faction ?= {
									faction_target = scope:lam_rally_lord
								}
							}
							OR = {
								NOT = {
									has_variable = lam_fac_favour
								}
								var:lam_fac_favour >= 0
							}
							OR = {
								has_variable = lam_fac_courtparty
								has_variable = lam_fac_kings_own
								var:lam_fac_stage ?= 5
								var:lam_fac_stage ?= 6
								var:lam_fac_stage ?= 8
							}
						}
						if = {
							limit = {
								OR = {
									has_variable = lam_fac_courtparty
									has_variable = lam_fac_kings_own
									var:lam_fac_stage ?= 5
									var:lam_fac_stage ?= 6
								}
								NOT = {
									scope:lam_rally_lord = {
										has_variable = lam_fac_deceived
									}
								}
							}
							lam_fac_rally_join_effect = yes
						}
						else = {
							random = {
								chance = {
									value = 50
									if = {
										limit = {
											has_variable = lam_fac_leg_arms
										}
										add = {
											value = var:lam_fac_leg_arms
											multiply = 10
										}
									}
								}
								lam_fac_rally_join_effect = yes
							}
						}
					}
				}
			}
		}
	}
}
