republican_succession_effect = {
	var:republican_government_story = { save_scope_as = story }
	save_scope_as = character
	if = {
		limit = { NOT = { this = scope:successor } }
		if = {
			limit = { is_ai = no }
			set_player_character = scope:successor
		}
		if = {
			limit = { is_merchant_family = yes }
			save_scope_value_as = {
				name = merchant_title_found
				value = yes
			}
			if = {
				limit = { NOT = { house = scope:successor.house } }
				every_held_title = {
					limit = { is_merchant_family_title = yes }
					save_scope_as = merchant_family_title
				}
				destroy_title = scope:merchant_family_title
			}
		}
		remove_trait = republican_legacy
		scope:story = {
			make_story_owner = scope:successor
		}
		set_designated_heir_unsafe = scope:successor
		force_step_down_landed_titles = yes
		#Transfer half of the gold back, if merchant family
		if = {
			limit = { is_merchant_family = yes }
			scope:successor = {
				pay_short_term_gold = {
					target = prev
					gold = {
						value = gold
						multiply = 0.5
					}
				}
			}
		}
		scope:successor = {
			set_variable = {
				name = republican_government_story
				value = scope:story
			}
			appoint_court_position = {
				recipient = prev
				court_position = republican_magistrate
			}
			republican_trigger_election_after_term = yes
			republican_government_elect_magistrates_effect = yes
			republican_legacy_apply_trait_effect = yes
		}
		if = {
			limit = { scope:merchant_title_found = yes }
			if = {
				limit = { NOT = { house = scope:successor.house } }
				#Return the merchant family title
				create_title_and_vassal_change = {
					type = returned
					save_scope_as = change
					add_claim_on_loss = no
				}
				scope:merchant_family_title = {
					change_title_holder = {
						holder = prev
						change = scope:change
					}
				}
				resolve_title_and_vassal_change = scope:change
				change_government = merchant_republic_government
			}
			else = {
				scope:story = {
					remove_list_variable = {
						name = merchant_families
						target = prev
					}
					add_to_variable_list = {
						name = merchant_families
						target = scope:successor
					}
				}
				remove_variable = republican_government_story
			}
		}
	}
	else = {
		republican_trigger_election_after_term = yes
		republican_government_elect_magistrates_effect = yes
	}
	scope:story = {
		set_variable = {
			name = succession_date
			value = current_date
		}
		set_variable = {
			name = election_date
			value = {
				value = current_date
				add = republican_election_term_in_days
			}
		}
	}
	remove_trade_variables = yes
	scope:story = {
		republican_government_populate_candidate_list_effect = yes
	}
	scope:successor = {
		if = {
			limit = {
				is_ai = no
				prev = { is_merchant_family = yes }
				NOT = { house = prev.house }
				scope:story.var:play_as_same_family ?= yes
			}
			set_player_character = prev
		}
		if = {
			limit = {
				any_vassal_or_below = {
					is_merchant_family = yes
					house = prev.house
				}
			}
			pass_merchant_family_title_to_ruler_of_same_house = yes
		}
	}
}

			set_variable = {

}

#Scope: character (republican ruler)
republican_government_elect_magistrates_effect = {
	if = {
		limit = { can_employ_court_position_type = republican_magistrate }
		save_scope_as = liege
		every_vassal = {
			limit = {
				is_eligible_magistrate_trigger = yes
				NOT = { has_court_position = republican_magistrate }
				OR = {
					government_has_flag = government_is_republic
					government_has_flag = government_is_active_republic
				}
			}
			add_to_list = candidate_pool
		}
		every_courtier = {
			limit = {
				is_eligible_magistrate_trigger = yes
				NOT = { has_court_position = republican_magistrate }
			}
			add_to_list = candidate_pool
		}

		while = {
			limit = {
				can_employ_court_position_type = republican_magistrate
			}
			if = {
				limit = { list_size:candidate_pool > 0 }
				ordered_in_list = {
					list = candidate_pool
					order_by = republican_magistrate_eligibility

					scope:liege = {
						appoint_court_position = {
							recipient = prev
							court_position = republican_magistrate
						}
					}
					remove_from_list = candidate_pool
				}
			}
			else = {
				create_character = {
					template = republican_magistrate_character
					faith = scope:liege.capital_county.faith
					culture = scope:liege.capital_county.culture
					location = scope:liege.capital_province
					gender_female_chance = scope:liege.republican_magistrate_gender_female_chance
					save_scope_as = magistrate_character
				}
				scope:liege = {
					add_courtier = scope:magistrate_character
					appoint_court_position = {
						recipient = scope:magistrate_character
						court_position = republican_magistrate
					}
				}
			}
		}
	}
}

republican_legacy_purchase_perk_effect = {
	if = {
		limit = { is_republican_government_story_owner = yes }
		var:republican_government_story = {
			if = { limit = { var:republican_prestige >= republican_legacy_cost }
				change_variable = {
					name = republican_prestige
					subtract = republican_legacy_cost
				}
				if = { limit = { NOT = { has_variable = legacy_level_$TRACK$ } }
					set_variable = {
						name = legacy_level_$TRACK$
						value = 0
					}
				}
				change_variable = {
					name = legacy_level_$TRACK$
					add = 1
				}
			}
		}
		republican_legacy_apply_trait_effect = yes
	}
}

republican_legacy_apply_trait_effect = {
	if = {
		limit = {
			has_trait = republican_legacy
		}
		remove_trait = republican_legacy
	}
	if = {
		limit = {
			OR = {
				var:republican_government_story.var:legacy_level_trade > 0
				var:republican_government_story.var:legacy_level_diplomacy > 0
				var:republican_government_story.var:legacy_level_civic > 0
				var:republican_government_story.var:legacy_level_militia > 0
				var:republican_government_story.var:legacy_level_prosperity > 0
			}
		}
		add_trait = republican_legacy
		add_trait_xp = {
			trait = republican_legacy
			track = trade
			value = { value = var:republican_government_story.var:legacy_level_trade multiply = 25 }
		}
		add_trait_xp = {
			trait = republican_legacy
			track = diplomacy
			value = { value = var:republican_government_story.var:legacy_level_diplomacy multiply = 25 }
		}
		add_trait_xp = {
			trait = republican_legacy
			track = civic
			value = { value = var:republican_government_story.var:legacy_level_civic multiply = 25 }
		}
		add_trait_xp = {
			trait = republican_legacy
			track = militia
			value = { value = var:republican_government_story.var:legacy_level_militia multiply = 25 }
		}
		add_trait_xp = {
			trait = republican_legacy
			track = prosperity
			value = { value = var:republican_government_story.var:legacy_level_prosperity multiply = 25 }
		}
	}
}

republican_drift_faction_support_effect = {
	save_scope_value_as = {
		name = support_difference
		value = {
			value = var:republican_government_story.var:$FACTION$_support
			subtract = republican_$FACTION$_support_target
		}
	}

	if = {
		limit = {
			scope:support_difference != 0
		}
		var:republican_government_story = {
			if = {
				limit = {
					scope:support_difference < 0.01
					scope:support_difference > -0.01
				}
				set_variable = {
					name = $FACTION$_support
					value = prev.republican_$FACTION$_support_target
				}
			}
			else = {
				set_variable = {
					name = $FACTION$_support
					value = {
						value = var:$FACTION$_support
						subtract = {
							value = scope:support_difference
							divide = 96
							if = { limit = { scope:support_difference < 0 } max = -0.0025 }
							else = { min = 0.0025 }
						}
					}
				}
			}
		}
	}
	var:republican_government_story = {
		set_variable = {
			name = republican_support
			value = prev.republican_support
		}
	}
}

handle_nested_republics_effect = {
	save_scope_as = liege
	every_vassal_or_below = { #We handle the Subject Merchant Families first
		limit = {
			is_nested_active_republic_character = yes
			is_merchant_family = yes
			is_republican_government_story_owner = no
		}
		if = { #We are a Merchant Republic, Integrate: We change their liege and assign their story variable
			limit = { scope:liege = { government_has_flag = government_is_merchant_republic } }
			create_title_and_vassal_change = {
				type = swear_fealty
				save_scope_as = change
			}
			change_liege = {
				liege = scope:liege
				change = scope:change
			}
			resolve_title_and_vassal_change = scope:change
			scope:liege.var:republican_government_story = {
				add_to_variable_list = {
					name = merchant_families
					target = prev
				}
			}
			set_variable = {
				name = republican_government_story
				value = scope:liege.var:republican_government_story
			}
		}
		else = { #We are not Merchant Republic, End Them: They must be destroyed!!!
			var:republican_government_story = { save_scope_as = republican_government_story }
			#remove_merchant_family_effect = yes
		}
	}
	every_vassal_or_below = { #We handle the Merchant Republic Leader(s)
		limit = {
			is_nested_active_republic_character = yes
			is_merchant_family = yes
			is_republican_government_story_owner = yes
		}
		if = { #We are a Merchant Republic, Integrate: We change their liege, end their story and assign their story variable
			limit = { scope:liege = { government_has_flag = government_is_merchant_republic } }
			if = {
				limit = { NOT = { liege = scope:liege } }
				create_title_and_vassal_change = {
					type = swear_fealty
					save_scope_as = change
				}
				change_liege = {
					liege = scope:liege
					change = scope:change
				}
				resolve_title_and_vassal_change = scope:change
			}
			var:republican_government_story = {
				every_in_list = {
					variable = merchant_families
					prev = {
						remove_list_variable = {
							name = merchant_families
							target = prev
						}
					}
				}
			}
			scope:liege.var:republican_government_story = {
				add_to_variable_list = {
					name = merchant_families
					target = prev
				}
			}
			var:republican_government_story = { end_story = yes }
			set_variable = {
				name = republican_government_story
				value = scope:liege.var:republican_government_story
			}
		}
		else = { #We are not Merchant Republic, End Them: They must be destroyed!!!
			var:republican_government_story = { save_scope_as = republican_government_story }
			remove_merchant_family_effect = yes
			scope:republican_government_story = { end_story = yes }
		}
	}
	every_vassal_or_below = { #We handle the non-merchant republics, oh how things are simple. :_)
		limit = {
			is_nested_active_republic_character = yes
			is_republican_government_story_owner = yes
			NOT = { government_has_flag = government_is_merchant_republic }
		}
		var:republican_government_story = { end_story = yes }
		change_government = republic_government
	}
}
