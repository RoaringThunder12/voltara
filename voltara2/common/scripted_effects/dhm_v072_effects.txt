# DHM v0.7.2: event-local outcomes; compatibility never overwrites a new result.
# Old queued events do not contain a unique result. Specific remaining effects
# provide a fallback; a generic reconciliation bonus is deliberately insufficient.
dhm_v072_restore_union_outcome_effect = {
	if = {
		limit = { NOT = { exists = scope:$KEY$ } }
		save_scope_value_as = { name = $KEY$ value = flag:unknown }
		if = {
			limit = { OR = { $SUBJECT$ = { has_character_modifier = $NEGATIVE$ } $SPOUSE$ = { has_character_modifier = $NEGATIVE$ } } }
			save_scope_value_as = { name = $KEY$ value = flag:unsettled }
		}
		else_if = {
			limit = { $SUBJECT$ = { has_character_modifier = dhm_established_household_modifier } $SPOUSE$ = { has_character_modifier = dhm_established_household_modifier } }
			save_scope_value_as = { name = $KEY$ value = flag:established }
		}
		else_if = {
			limit = { $SUBJECT$ = { has_character_modifier = dhm_marital_self_reliance_modifier } $SPOUSE$ = { has_character_modifier = dhm_marital_self_reliance_modifier } }
			save_scope_value_as = { name = $KEY$ value = flag:independent }
		}
	}
}

# The second decision can repair harsh beginnings or undo early openness.
# Missing old-save evidence receives a neutral ending, not invented trust.
dhm_v072_prepare_parent_ending_effect = {
	save_scope_value_as = { name = dhm072_parent_ending value = flag:unknown }
	if = {
		limit = { exists = scope:dhm072_parent_reply scope:dhm072_parent_reply = flag:speak }
		if = {
			limit = { scope:dhm_arc_first = { dhm_v071_choice_is_trigger = { KEY = dhm071_parent_choice VALUE = flag:duty LEGACY = dhm_parent_arc_duty } } }
			save_scope_value_as = { name = dhm072_parent_ending value = flag:repaired }
		}
		else_if = {
			limit = { exists = scope:dhm071_parent_choice OR = { scope:dhm071_parent_choice = flag:listen scope:dhm071_parent_choice = flag:test } }
			save_scope_value_as = { name = dhm072_parent_ending value = flag:trust }
		}
		else_if = {
			limit = { NOT = { exists = scope:dhm071_parent_choice } scope:dhm_arc_first = { has_character_flag = dhm_parent_arc_listened } }
			save_scope_value_as = { name = dhm072_parent_ending value = flag:trust }
		}
	}
	else_if = {
		limit = { exists = scope:dhm072_parent_reply scope:dhm072_parent_reply = flag:silence }
		if = {
			limit = { scope:dhm_arc_first = { dhm_v071_choice_is_trigger = { KEY = dhm071_parent_choice VALUE = flag:duty LEGACY = dhm_parent_arc_duty } } }
			save_scope_value_as = { name = dhm072_parent_ending value = flag:resentment }
		}
		else = { save_scope_value_as = { name = dhm072_parent_ending value = flag:distance } }
	}
}
