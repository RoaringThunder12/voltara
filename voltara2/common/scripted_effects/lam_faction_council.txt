lam_fac_council_split_effect = {
	scope:lam_vt_head.liege = {
		save_scope_as = lam_vs_lord
	}
	clear_saved_scope = lam_vs_out
	scope:lam_vs_lord = {
		lam_fac_seat_open_effect = yes
	}
	if = {
		limit = {
			scope:lam_vs_lord = {
				NOT = {
					has_variable = lam_fac_seat_open
				}
			}
			scope:lam_vt_head = {
				has_variable = lam_fac_share
				var:lam_fac_share >= 25
			}
		}
		scope:lam_vs_lord = {
			ordered_in_list = {
				variable = lam_bloc_leads
				limit = {
					is_alive = yes
					NOT = {
						this = scope:lam_vt_head
					}
					NOT = {
						has_variable = lam_fac_courtparty
					}
					NOT = {
						has_variable = lam_fac_disband
					}
				}
				order_by = {
					value = 0
					subtract = var:lam_fac_share
				}
				save_scope_as = lam_vs_out
			}
		}
	}
	if = {
		limit = {
			OR = {
				scope:lam_vs_lord = {
					has_variable = lam_fac_seat_open
				}
				exists = scope:lam_vs_out
			}
		}
		scope:lam_vs_out ?= {
			scope:lam_vs_lord = {
				remove_list_variable = {
					name = lam_bloc_leads
					target = scope:lam_vs_out
				}
			}
			set_variable = {
				name = lam_fac_disband
				value = yes
			}
			set_variable = {
				name = lam_fac_end_why
				value = flag:displaced
			}
			lam_fac_drop_identity_effect = yes
		}
		scope:lam_vt_head = {
			set_variable = {
				name = lam_fac_split_cd
				value = yes
				years = 10
			}
			lam_fac_timer_set_effect = {
				NAME = lam_fac_split_cd
				YEARS = 10
			}
			lam_gn_event_effect = {
				KEY = split
			}
			if = {
				limit = {
					NOT = {
						has_variable = lam_fac_branched
					}
				}
				set_variable = {
					name = lam_fac_branched
					value = 0
				}
			}
			change_variable = {
				name = lam_fac_branched
				add = 1
			}
			lam_fac_cohesion_add_effect = {
				ADD = -10
			}
		}
		scope:lam_vs_lord = {
			set_variable = {
				name = lam_fac_split_court_cd
				value = yes
				years = 3
			}
		}
		lam_fac_stat_add_effect = {
			KEY = split
		}
		lam_fac_stat_add_effect = {
			KEY = split_$ISSUE$
		}
		scope:lam_vt_lead = {
			set_variable = {
				name = lam_gn_why
				value = flag:split
			}
			set_variable = {
				name = lam_gn_sr
				value = flag:$ISSUE$
			}
			if = {
				limit = {
					scope:lam_vt_head = {
						has_variable = lam_fac_rec
						has_variable = lam_fac_rec_title
					}
				}
				set_variable = {
					name = lam_gn_from
					value = scope:lam_vt_head.var:lam_fac_rec
				}
				set_variable = {
					name = lam_gn_from_book

				}
			}
			if = {

				}

			}
			if = {
				limit = {
					has_variable = lam_fac_pinned
				}

			}

			lam_fac_new_identity_effect = yes

				PARENT = scope:lam_vt_head
			}

					has_variable = lam_fac_cp_broken_m
