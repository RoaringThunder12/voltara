# Scope : Character
# PROVINCE : province
E_kCAFG_set_rite_from_province_random = {
    $PROVINCE$ = {
        random_in_list = {
            variable = kCAFG_province_rite_structs

            weight = {
                base = 0
                modifier = {
                    add = {
                        value = this.var:factor
                        multiply = 10000
                    }
                }
            }

            var:rite = { save_temporary_scope_as = E_kCAFG_set_rite_from_province_random_selection }
        }
    }
    if = {
        limit = { exists = scope:E_kCAFG_set_rite_from_province_random_selection }

        set_character_rite = scope:E_kCAFG_set_rite_from_province_random_selection
    }
}
