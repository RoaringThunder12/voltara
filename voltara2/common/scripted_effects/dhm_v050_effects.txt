# -----------------------------------------------------------------------------
# HOUSEHOLD COHESION
# The value is save-wide because this is a deliberately single-player mod and
# the household's recent temper should survive succession.
# -----------------------------------------------------------------------------
dhm_v050_refresh_cohesion_modifier_effect = {
	hidden_effect = {
		remove_character_modifier = dhm_cohesion_fractured_modifier
		remove_character_modifier = dhm_cohesion_uneasy_modifier
		remove_character_modifier = dhm_cohesion_stable_modifier
		remove_character_modifier = dhm_cohesion_united_modifier
		remove_character_modifier = dhm_cohesion_devoted_modifier
	}
	if = {
		limit = { NOT = { has_global_variable = dhm_household_cohesion } }
		# A newly migrated global variable is not readable until the next tick.
		# Show the neutral state during that first tick instead of evaluating an unset scope.
		add_character_modifier = { modifier = dhm_cohesion_stable_modifier }
	}
	else_if = {
		limit = { global_var:dhm_household_cohesion < 20 }
		add_character_modifier = { modifier = dhm_cohesion_fractured_modifier }
	}
	else_if = {
		limit = { global_var:dhm_household_cohesion < 40 }
		add_character_modifier = { modifier = dhm_cohesion_uneasy_modifier }
	}
	else_if = {
		limit = { global_var:dhm_household_cohesion < 60 }
		add_character_modifier = { modifier = dhm_cohesion_stable_modifier }
	}
	else_if = {
		limit = { global_var:dhm_household_cohesion < 80 }
		add_character_modifier = { modifier = dhm_cohesion_united_modifier }
	}
	else = {
		add_character_modifier = { modifier = dhm_cohesion_devoted_modifier }
	}
}

dhm_v050_clamp_cohesion_effect = {
	if = {
		limit = { has_global_variable = dhm_household_cohesion }
		if = {
			limit = { global_var:dhm_household_cohesion > 100 }
			set_global_variable = { name = dhm_household_cohesion value = 100 }
		}
		else_if = {
			limit = { global_var:dhm_household_cohesion < 0 }
			set_global_variable = { name = dhm_household_cohesion value = 0 }
		}
	}
	else = {
		# A console event or incomplete older save may precede normal setup.
		set_global_variable = { name = dhm_household_cohesion value = 50 }
	}
	dhm_v050_refresh_cohesion_modifier_effect = yes
}

dhm_v050_clear_ambition_effect = {
	remove_character_flag = dhm_ambition_love
	remove_character_flag = dhm_ambition_status
	remove_character_flag = dhm_ambition_learning
	remove_character_flag = dhm_ambition_faith
	remove_character_flag = dhm_ambition_security
	remove_character_flag = dhm_ambition_freedom
	remove_character_flag = dhm_ambition_children
	remove_character_flag = dhm_ambition_reconciliation
}

dhm_v050_assign_ambition_effect = {
	dhm_v050_clear_ambition_effect = yes
	if = {
		limit = { has_trait = ambitious }
		add_character_flag = dhm_ambition_status
	}
	else_if = {
		limit = { has_trait = zealous }
		add_character_flag = dhm_ambition_faith
	}
	else_if = {
		limit = { has_trait = diligent learning >= 10 }
		add_character_flag = dhm_ambition_learning
	}
	else_if = {
		limit = { has_trait = content }
		add_character_flag = dhm_ambition_security
	}
	else_if = {
		limit = { has_trait = forgiving }
		add_character_flag = dhm_ambition_reconciliation
	}
	else_if = {
		limit = { is_married = yes NOT = { any_child = { always = yes } } }
		add_character_flag = dhm_ambition_children
	}
	else_if = {
		limit = { has_trait = compassionate }
		add_character_flag = dhm_ambition_love
	}
	else = {
		random_list = {
			20 = { add_character_flag = dhm_ambition_love }
			18 = { add_character_flag = dhm_ambition_status }
			18 = { add_character_flag = dhm_ambition_learning }
			14 = { add_character_flag = dhm_ambition_faith }
			16 = { add_character_flag = dhm_ambition_security }
			12 = { add_character_flag = dhm_ambition_freedom }
			12 = { add_character_flag = dhm_ambition_children }
			10 = { add_character_flag = dhm_ambition_reconciliation }
		}
	}
}

dhm_v050_cleanup_officers_effect = {
	save_scope_as = dhm_manager
	# All work stays within the current court. Loyalty and past memories survive.
	every_courtier = { remove_character_flag = dhm073_office_claimed remove_character_flag = dhm074_office_reserved }
	# Reserve every valid appointment before recovering orphaned role flags.
	# A vacant early-listed office must never take the holder of a later office.
	if = { limit = { var:dhm_officer_steward ?= { dhm_v073_officer_available_trigger = yes } } var:dhm_officer_steward = { add_character_flag = dhm074_office_reserved } }
	if = { limit = { var:dhm_officer_matchmaker ?= { dhm_v073_officer_available_trigger = yes } } var:dhm_officer_matchmaker = { add_character_flag = dhm074_office_reserved } }
	if = { limit = { var:dhm_officer_tutor ?= { dhm_v073_officer_available_trigger = yes } } var:dhm_officer_tutor = { add_character_flag = dhm074_office_reserved } }
	if = { limit = { var:dhm_officer_chronicler ?= { dhm_v073_officer_available_trigger = yes } } var:dhm_officer_chronicler = { add_character_flag = dhm074_office_reserved } }
	dhm_v073_reconcile_office_effect = { REFERENCE = dhm_officer_steward ROLE = dhm_office_steward SKILL = stewardship }
	dhm_v073_reconcile_office_effect = { REFERENCE = dhm_officer_matchmaker ROLE = dhm_office_matchmaker SKILL = diplomacy }
	dhm_v073_reconcile_office_effect = { REFERENCE = dhm_officer_tutor ROLE = dhm_office_tutor SKILL = learning }
	dhm_v073_reconcile_office_effect = { REFERENCE = dhm_officer_chronicler ROLE = dhm_office_chronicler SKILL = learning }
	every_courtier = {
		if = {
			limit = { has_character_flag = dhm_office_steward scope:dhm_manager = { NOT = { var:dhm_officer_steward ?= prev } } }
			remove_character_flag = dhm_office_steward
		}
		if = {
			limit = { has_character_flag = dhm_office_matchmaker scope:dhm_manager = { NOT = { var:dhm_officer_matchmaker ?= prev } } }
			remove_character_flag = dhm_office_matchmaker
		}
		if = {
			limit = { has_character_flag = dhm_office_tutor scope:dhm_manager = { NOT = { var:dhm_officer_tutor ?= prev } } }
			remove_character_flag = dhm_office_tutor
		}
		if = {
			limit = { has_character_flag = dhm_office_chronicler scope:dhm_manager = { NOT = { var:dhm_officer_chronicler ?= prev } } }
			remove_character_flag = dhm_office_chronicler
		}
		# Old duplicate/orphan flags may be cleaned without passing through an
		# appointment. Only a character holding no real DHM office loses tenure.
		if = {
			limit = {
				NOR = {
					has_character_flag = dhm_office_steward
					has_character_flag = dhm_office_matchmaker
					has_character_flag = dhm_office_tutor
					has_character_flag = dhm_office_chronicler
				}
			}
			dhm_v080_close_officer_tenure_effect = yes
		}
		remove_character_flag = dhm073_office_claimed
		remove_character_flag = dhm074_office_reserved
	}
}

dhm_v050_maintain_household_effect = {
	save_scope_as = dhm_manager
	dhm_v050_cleanup_officers_effect = yes

	# At most two new close relatives receive a persistent ambition per year.
	ordered_close_family_member = {
		limit = {
			is_alive = yes
			is_ai = yes
			age >= 12
			is_in_the_same_court_as = scope:dhm_manager
			NOT = { dhm_v050_has_ambition_trigger = yes }
		}
		order_by = age
		max = 2
		check_range_bounds = no
		dhm_v050_assign_ambition_effect = yes
	}

	dhm_v050_refresh_cohesion_modifier_effect = yes
	if = {
		limit = { has_global_variable = dhm_setting_legacy_mechanics_disabled }
		dhm_v050_clear_legacy_modifier_effect = yes
	}
	else = { dhm_v050_apply_legacy_modifier_effect = yes }
}

dhm_v050_story_pulse_effect = {
	save_scope_as = dhm_manager
	if = {
		limit = {
			is_alive = yes
			is_ai = no
			NOT = { has_character_flag = dhm_living_story_cooldown }
			NOT = { has_character_flag = dhm081_aftermath_pause }
			NOT = { has_global_variable = dhm_setting_stories_off }
		}
		random_list = {
			16 = {
				trigger = { any_close_family_member = { dhm_v050_ambition_target_trigger = yes } }
				random_close_family_member = {
					limit = { dhm_v050_ambition_target_trigger = yes }
					save_scope_as = dhm_ambition_target
					add_character_flag = { flag = dhm_ambition_story_recent years = 6 }
				}
				dhm_apply_living_story_cooldown_effect = yes
				trigger_event = { id = dhm_legacy.0200 days = { 4 12 } }
			}
			14 = {
				trigger = {
					any_close_family_member = {
						dhm_v050_arc_relative_trigger = yes
						any_sibling = { dhm_v050_arc_relative_trigger = yes }
					}
				}
				random_close_family_member = {
					limit = {
						dhm_v050_arc_relative_trigger = yes
						any_sibling = { dhm_v050_arc_relative_trigger = yes }
					}
					save_scope_as = dhm_arc_first
					random_sibling = {
						limit = { dhm_v050_arc_relative_trigger = yes }
						save_scope_as = dhm_arc_second
					}
