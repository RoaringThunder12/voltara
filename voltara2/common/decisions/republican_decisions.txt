convert_to_merchant_republic_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/republican_decision_become_merchant_republic.dds"
	}
	decision_group_type = major
	desc = convert_to_merchant_republic_decision_desc

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 8
		kingdom = 4
		empire = 0
		hegemony = 0
	}

	is_shown = {
		has_ep3_dlc_trigger = yes
		OR = {
			AND = {
				is_independent_ruler = yes
				NOT = {	government_has_flag = government_is_landless_adventurer }
				NOT = {	government_has_flag = government_is_active_republic }
				NOT = { government_has_flag = government_is_tribal }
			}
			AND = {
				government_has_flag = government_is_active_republic
				NOT = {	government_has_flag = government_is_merchant_republic }
			}
		}
		NAND = { #We'd rather show Hanseatic League decision if this is the case
			OR = {
				is_independent_ruler = yes
				liege ?= { highest_held_title_tier = tier_empire }
			}
			has_sea_access = yes
			capital_county = title:c_voltara_11371
		}
	}

	is_valid = {
		has_ep3_dlc_trigger = yes
		OR = {
			AND = {
				can_convert_to_merchant_republic_trigger = yes
				is_independent_ruler = yes
			}
			AND = {
				has_sea_access = yes
				custom_description = {
					text = has_trade_legacy_4
					AND = {
						government_has_flag = government_is_active_republic
						NOT = {	government_has_flag = government_is_merchant_republic }
						is_republican_government_story_owner = yes
						var:republican_government_story.var:legacy_level_trade >= 4
					}
				}
			}
		}
	}
	
	is_valid_showing_failures_only = {
		is_at_war = no
	}

	cost = {
		prestige = {
			value = 500
		}
		gold = {
			value = 200
			#Income
			if = {
				limit = { root.monthly_character_income >= 100 }
				multiply = 3
			}
			else_if = {
				limit = { root.monthly_character_income >= 80 }
				multiply = 2.5
			}
			else_if = {
				limit = { root.monthly_character_income >= 60 }
				multiply = 2
			}
			else_if = {
				limit = { root.monthly_character_income >= 40 }
				multiply = 1.5
			}
			else_if = {
				limit = { root.monthly_character_income >= 20 }
				multiply = 1.1
			}
		}
	}

	effect = {
		convert_to_merchant_republic_government_effect = yes
	}

	ai_potential = {
		OR = {
			is_independent_ruler = yes
			AND = {
				government_has_flag = government_is_active_republic
				NOT = {	government_has_flag = government_is_merchant_republic }
				is_republican_government_story_owner = yes
				var:republican_government_story.var:legacy_level_trade >= 4
			}
		}
	}

	ai_will_do = {
		base = -50
		ai_value_modifier = {
			ai_greed = 1
		}
		modifier = {
			add = 100
			has_trait = greedy
		}
		modifier = {
			culture = {
				OR = {
					has_cultural_pillar = ethos_bureaucratic 
					has_cultural_pillar = ethos_courtly 
					has_cultural_pillar = ethos_egalitarian 
				}
			}
			add = 50
		}
		modifier = {
			culture = {
				OR = {
					has_cultural_tradition = tradition_legalistic
					has_cultural_tradition = tradition_hereditary_hierarchy 
				}
			}
			add = 50
		}
		modifier = {
			culture = {
				OR = {
					has_cultural_tradition = tradition_fp3_enlightened_magnates 
					has_cultural_tradition = tradition_loyal_soldiers 
				}
			}
			add = 20
		}
		modifier = {
			culture = {
				OR = {
					has_cultural_tradition = tradition_staunch_traditionalists
					has_cultural_tradition = tradition_ep3_audacious_cadets
				}
			}
			add = -20
		}
		modifier = {
			culture = {
				OR = {
					has_cultural_tradition = tradition_ep3_indomitable_azatani
					has_cultural_tradition = tradition_quarrelsome
					has_cultural_tradition = tradition_warrior_culture
					has_cultural_tradition = tradition_fp3_fierce_independence
				}
			}
			add = -50
		}
		modifier = {
			culture = {
				OR = {
					has_cultural_pillar = ethos_stoic
					has_cultural_pillar = ethos_bellicose 
					has_cultural_pillar = ethos_communal 
				}
			}
			add = -50
		}
	}
}
