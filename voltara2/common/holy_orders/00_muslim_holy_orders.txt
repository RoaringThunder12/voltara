ho_muslim_safavid_order = {

	type = military
	government = holy_order_government
	title = d_safavid_order
	name = "holy_order_safavid_order"

	trigger = {
		trigger_if = {
			limit = {
				scope:founder = { has_variable = military_order_game_start_founder }
			}
			always = no
		}
		trigger_else = {
			scope:founder.faith = { religion = religion:islam_religion }
			scope:founder.rite = { rite_has_doctrine = muhammad_succession_sunni_doctrine }
		}
	}

	weight = {
		value = 0
		if = {
			limit = {
				scope:founder = {
					culture = {
						has_cultural_pillar = heritage_iranian
					}
				}
			}
			add = 25
		}
		if = { limit = {
				# Optional native title is absent on the Voltara map.
# scope:province.county ?= title:c_gilan
always = no
			} # Redundant unreachable effects archived in error-batch/audit.json.
}
		if = {
			limit = {
				scope:province = {
					always = no
				}
			}
			add = 100
		}
		if = {
			limit = {
				scope:province = {
					geographical_region = world_middle_east_persia
				}
			}
			add = 25
		}
		if = {
			limit = {
				current_year < 1200
			}
			multiply = 0.5
		}
		if = {
			limit = {
				NOT = { scope:founder.rite = rite:sufri }
			}
			multiply = 0.05
		}
	}
}

#Culture Ghazis
ho_muslim_culture_ghazis = {
	type = military
	government = holy_order_government

	trigger = {
		#Stop game start creation
		trigger_if = {
			limit = {
				scope:founder = { has_variable = military_order_game_start_founder }
			}
			always = no
		}
		trigger_else = {
			scope:founder.faith.religion = religion:islam_religion
			#Need to access barony being leased again
			exists = scope:founder.culture
			#Check that this dynamic template isn't used with the same title
			NOT = {
				scope:leader.faith = {
					any_faith_holy_order = {
						title = {
							has_variable = culture_ghazis
							var:culture_ghazis = scope:founder.culture
						}
					}
				}
			}
		}
	}

	effect = {
		#Need to be able to run effects here so I can save the kingdom the first leased barony is in
		scope:founder.culture = {
			save_scope_as = culture_ghazis
		}
		#Setting the culture as a variable so that we don't get duplicate names
		set_variable = {
			name = culture_ghazis
			value = scope:culture_ghazis
		}
		scope:title = {
			set_variable = {
				name = order_heritage
				value = scope:founder.culture
			}
		}
		scope:leader = {
			if = {
				limit = {
					NOT = {
						has_realm_law = cultural_holy_order_succession_law
					}
				}
				add_realm_law = cultural_holy_order_succession_law
			}
		}
	}

	# this will be used for both the HO name and the title name, since we don't define a title
	name = "holy_order_culture_ghazis"
	weight = 10
}
