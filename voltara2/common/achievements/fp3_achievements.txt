fp3_rich_in_diversity_achievement = {
	possible = {  }
	happened = {
		custom_description = {
			text = fp3_rich_in_diversity_achievement_trigger
			is_ai = no
			OR = {
				rite = { rite_has_tenet = tenet_tax_nonbelievers }
				faith = { has_doctrine = special_doctrine_jizya }
			}
			any_vassal = {
				faith != root.faith
				count >= 10
			}
		}
	}
}
