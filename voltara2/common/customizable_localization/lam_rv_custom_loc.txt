# This = a faction head: why the rise their council voted for raised no demand.
LamRvStall = {
	type = character
	text = {
		trigger = {
			exists = joined_faction
		}
		localization_key = lam_rv_stall_joined
	}
	text = {
		trigger = {
			has_variable = lam_pb_retry
		}
		localization_key = lam_rv_stall_retry
	}
	text = {
		trigger = {
			NOT = {
				lam_pb_legal_liberty_faction_trigger = yes
			}
			NOT = {
				lam_pb_legal_independence_faction_trigger = yes
			}
		}
		localization_key = lam_rv_stall_nolaw
	}
	text = {
		localization_key = lam_rv_stall_other
		fallback = yes
	}
}
