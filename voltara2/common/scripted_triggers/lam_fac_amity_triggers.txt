# Living Factions - sworn amity between factions (tools/claude_fixes/amity.py)
# Two faction heads of one court who may pledge amity: not already sworn, not rivals.
lam_fac_amity_pair_trigger = {
	$A$ = {
		is_alive = yes
		has_variable = lam_bloc_lead
		exists = liege
		NOT = {
			has_variable = lam_fac_disband
		}
	}
	$B$ = {
		is_alive = yes
		has_variable = lam_bloc_lead
		liege ?= $A$.liege
		NOT = {
			this = $A$
		}
		NOT = {
			has_variable = lam_fac_disband
		}
		NOT = {
			has_relation_rival = $A$
		}
		NOT = {
			has_relation_nemesis = $A$
		}
	}
	NOT = {
		$A$ = {
			is_target_in_variable_list = {
				name = lam_fac_allies
				target = $B$
			}
		}
	}
}
