# This = the lord: a faction with a stand on issue T has gone over to a demand or a war against the crown - the issue
# waits for that outcome (no drift, no clock, no settling meanwhile).
lam_is_held_trigger = {
	any_in_list = {
		variable = lam_bloc_leads
		is_alive = yes
		lam_wc_committed_trigger = yes
		OR = {
			lam_is_side_$T$_x_trigger = yes
			lam_is_side_$T$_y_trigger = yes
		}
	}
}

# This = a faction member with an issue dispute (lam_gg_source = flag:issue): it stays a live cause while the issue it
# was about is still open in the court of HEAD's liege (used by lam_gg_current_cause_trigger and the split council).
lam_is_dissent_live_trigger = {
	exists = var:lam_is_dis_lord
	var:lam_is_dis_lord ?= $HEAD$.liege
	OR = {
		AND = {
			var:lam_is_dis ?= flag:crown
			var:lam_is_dis_lord = {
				has_variable = lam_is_crown_open
			}
		}
		AND = {
			var:lam_is_dis ?= flag:council
			var:lam_is_dis_lord = {
				has_variable = lam_is_council_open
			}
		}
		AND = {
			var:lam_is_dis ?= flag:faith
			var:lam_is_dis_lord = {
				has_variable = lam_is_faith_open
			}
		}
		AND = {
			var:lam_is_dis ?= flag:tax
			var:lam_is_dis_lord = {
				has_variable = lam_is_tax_open
			}
		}
		AND = {
			var:lam_is_dis ?= flag:war
			var:lam_is_dis_lord = {
				has_variable = lam_is_war_open
			}
		}
	}
}

			}

		}
		AND = {
			scope:lam_cr_head = {
				var:lam_cr_st_t ?= flag:war

			OR = { AND = { scope:lam_cr_lord = { var:lam_is_war_mode ?= flag:campaign } NOT = { OR = { has_trait = brave has_trait = ambitious has_trait = wrathful } } OR = { has_trait = craven has_trait = content has_trait = lazy } } AND = { scope:lam_cr_lord = { var:lam_is_war_mode ?= flag:peace } OR = { has_trait = brave has_trait = ambitious has_trait = wrathful } } }
		}
	}
