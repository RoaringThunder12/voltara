lam_er_heir_conflict_trigger = {
	trigger_if = {
		limit = {
			lam_ev_member_trigger = yes
			is_ai = yes
		}
		trigger_if = {
			limit = {
				exists = scope:lam_ev_head.var:lam_fac_heir
				exists = scope:lam_ev_head.liege.primary_heir
				scope:lam_ev_head.liege = {
					is_alive = yes
				}
			}
			trigger_if = {
				limit = {
					scope:lam_ev_head.var:lam_fac_heir = {
						is_alive = yes
					}
					scope:lam_ev_head.liege.primary_heir = {
						is_alive = yes
					}
				}
				NOT = {
					scope:lam_ev_head.var:lam_fac_heir = scope:lam_ev_head.liege.primary_heir
				}
				NOT = {
					scope:lam_ev_head.liege.primary_heir = scope:lam_ev_head.liege
				}
				NOT = {
					var:lam_er_accepted_heir ?= scope:lam_ev_head.var:lam_fac_heir
				}
				OR = {
					AND = {
						opinion = {
							target = scope:lam_ev_head.liege.primary_heir
							value >= 60
						}
						"opinion(scope:lam_ev_head.liege.primary_heir)" >= {
							value = "opinion(scope:lam_ev_head.var:lam_fac_heir)"
							add = 40
						}
						OR = {
							is_close_family_of = scope:lam_ev_head.liege.primary_heir
							has_relation_friend = scope:lam_ev_head.liege.primary_heir
							has_relation_best_friend = scope:lam_ev_head.liege.primary_heir
							has_relation_lover = scope:lam_ev_head.liege.primary_heir
						}
					}
					AND = {
						OR = {
							has_trait = just
							has_trait = honest
						}
						opinion = {
							target = scope:lam_ev_head.liege.primary_heir
							value >= 20
						}
						"opinion(scope:lam_ev_head.liege.primary_heir)" >= {
							value = "opinion(scope:lam_ev_head.var:lam_fac_heir)"
							add = 20
						}
					}
				}
			}
			trigger_else = {
				always = no
			}
		}
		trigger_else = {
			always = no
		}
	}
	trigger_else = {
		always = no
	}
}
lam_er_keep_primary_dispute_trigger = {
	trigger_if = {
		limit = {
			exists = scope:lam_cr_head
		}
		scope:lam_cr_head = {
			is_alive = yes
		}
		AND = {
			trigger_if = {
				limit = {
					scope:lam_cr_head = {
						is_alive = yes
					}
				}
				scope:lam_cr_head = {}
				AND = {
					trigger_if = {
						limit = {
							OR = {
								lam_ss_council_position_trigger = yes
								lam_hi_council_dormant_trigger = yes
							}
						}
						OR = {
							var:lam_bp_position_topic ?= flag:heir
							var:lam_bp_position_side ?= flag:fair
							has_variable = lam_gg_source
						}
						NOT = {
							var:lam_bp_proposed_topic ?= flag:heir
						}
						NOT = {
							var:lam_bp_proposed_side ?= flag:fair
						}
						NOT = {
							var:lam_bp_proposed_topic ?= var:lam_bp_position_topic
						}
					}
					trigger_else = {
						always = no
					}
				}
			}
			trigger_else = {
				always = no
			}
		}
	}
	trigger_else = {
		always = no
	}
}
lam_er_current_backend_consent_trigger = {
	trigger_if = {
		limit = {
			is_alive = yes
			exists = scope:faction
		}
		trigger_if = {
			limit = {
				scope:faction = {
					has_variable = lam_pb_managed
					has_variable = lam_pb_approved
				}
			}
			joined_faction ?= scope:faction
			scope:faction = {
				is_target_in_variable_list = {
					name = lam_pb_consented
					target = prev
				}
			}
		}
		trigger_else = {
			always = no
		}
	}
	trigger_else = {
		always = no
	}
}
