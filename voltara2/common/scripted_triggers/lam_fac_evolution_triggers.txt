lam_ev_member_trigger = {
	trigger_if = {
		limit = {
			is_alive = yes
			exists = scope:lam_ev_head
		}
		trigger_if = {
			limit = {
				scope:lam_ev_head = {
					is_alive = yes
					exists = liege
				}
			}
			var:lam_bloc ?= scope:lam_ev_head
			OR = {
				liege ?= scope:lam_ev_head.liege
				is_courtier_of = scope:lam_ev_head.liege
			}
			is_adult = yes
			is_imprisoned = no
			NOT = {
				this = scope:lam_ev_head
			}
			NOT = {
				has_variable = lam_bloc_lead
			}
		}
		trigger_else = {
			always = no
		}
	}
	trigger_else = {
		always = no
	}
}

lam_ev_unresolved_trigger = {
	trigger_if = {
		limit = {
			has_variable = lam_gg_source
		}
		lam_gg_current_cause_trigger = yes
	}
	trigger_else = {
		trigger_if = {
			limit = {
				var:lam_bp_position_topic ?= flag:heir
			}
			trigger_if = {
				limit = {
					exists = var:lam_bp_position_target
				}
				var:lam_bp_position_target = {
					is_alive = yes
				}
				scope:lam_ev_head = {
					var:lam_fac_heir ?= prev.var:lam_bp_position_target
				}
				trigger_if = {
					limit = {
						var:lam_bp_position_side ?= flag:retain
					}
					trigger_if = {
						limit = {
							exists = var:lam_bp_position_alternative
						}
						var:lam_bp_position_alternative = {
							is_alive = yes
						}
						NOT = {
							var:lam_bp_position_alternative = scope:lam_ev_head.liege
						}
					}
					trigger_else = {
						always = no
					}
				}
			}
			trigger_else = {
				always = no
			}
		}
		trigger_else_if = {
			limit = {
				var:lam_bp_position_side ?= flag:fair
			}
			trigger_if = {
				limit = {
					has_variable = lam_st_overlooked
				}
				var:lam_st_overlooked >= 1
				var:lam_st_reward_head ?= scope:lam_ev_head
				NOT = {
					is_councillor_of = scope:lam_ev_head.liege
				}
			}
			trigger_else = {
				always = no
			}
		}
		trigger_else = {
			always = yes
		}
	}
}
lam_ev_ready_trigger = {
	trigger_if = {
		limit = {
			is_alive = yes
		}
		trigger_if = {
			limit = {
				lam_ev_position_trigger = yes
				has_variable = lam_ev_since
			}
			var:lam_ev_since <= {
				value = lam_now_m
				subtract = lam_ev_repeat_months
			}
			OR = {
				var:lam_cr_unrest >= 2
				var:lam_ev_since <= {
					value = lam_now_m
					subtract = lam_ev_single_months
				}
			}
			OR = {
				opinion = {
					target = scope:lam_ev_head
					value <= 40
				}
				var:lam_cr_unrest >= 3
				var:lam_ev_since <= {
					value = lam_now_m
					subtract = 24
				}
			}
			OR = {
				var:lam_bp_position_topic ?= flag:heir
				var:lam_bp_position_side ?= flag:fair
				scope:lam_ev_head = {
					trigger_if = {
						limit = {
							has_variable = lam_fac_cohesion
						}
						var:lam_fac_cohesion < 55
					}
					trigger_else = {
						always = no
					}
				}
				var:lam_cr_unrest >= 3
				has_variable = lam_gg_source
			}
		}
		trigger_else = {
			always = no
		}
	}
	trigger_else = {
		always = no
	}
}
