dhm_v074_appoint_steward_trigger = {
 OR = {
  AND = { exists = scope:dhm074_appointment_role scope:dhm074_appointment_role = flag:steward }
  AND = {
   NOT = { exists = scope:dhm074_appointment_role }
   has_character_flag = dhm_appoint_steward_context
   NOR = { has_character_flag = dhm_appoint_matchmaker_context has_character_flag = dhm_appoint_tutor_context has_character_flag = dhm_appoint_chronicler_context }
  }
 }
}

# Ownership is per proposal, not just per character or per proposed spouse.
# Legacy queued windows are accepted only when no new proposal owns the subject.
dhm_v074_proposal_owned_trigger = {
 exists = scope:dhm_marriage_subject
 scope:dhm_marriage_subject = {
  is_alive = yes
  OR = {
   AND = {
    exists = scope:dhm074_proposal_id
    var:dhm074_proposal_id ?= scope:dhm074_proposal_id
    var:dhm074_proposal_candidate ?= scope:dhm_marriage_candidate
    var:dhm074_proposal_manager ?= scope:dhm_manager
   }
   AND = {
    NOT = { exists = scope:dhm074_proposal_id }
    NOT = { has_variable = dhm074_proposal_id }
    NOT = { has_variable = dhm074_proposal_candidate }
    NOT = { has_variable = dhm074_proposal_manager }
   }
  }
 }
}

# Cheap window checks do not run acceptance scoring on every option hover.
dhm_v074_proposal_context_trigger = {
 is_alive = yes
 is_ai = no
 exists = scope:dhm_manager
 this = scope:dhm_manager
 exists = scope:dhm_marriage_candidate
 dhm_v074_proposal_owned_trigger = yes
 scope:dhm_marriage_subject = {
  is_ai = yes
  age >= 12
  is_in_the_same_court_as = root
  is_imprisoned = no
  is_hostage = no
  is_betrothed = no
  NOT = { has_trait = incapable }
  NOT = { has_character_flag = dhm_matchmaking_cooldown }
  NOT = { has_character_flag = dhm_matchmaking_retry_cooldown }
  NOT = { has_variable = dhm_courtship_reserved_for }
 }
 scope:dhm_marriage_candidate = {
  is_alive = yes
  is_ai = yes
  this != scope:dhm_marriage_subject
  is_imprisoned = no
  is_hostage = no
  is_betrothed = no
  NOT = { has_trait = incapable }
  exists = matchmaker
  matchmaker = { is_alive = yes is_imprisoned = no is_hostage = no }
 }
}
