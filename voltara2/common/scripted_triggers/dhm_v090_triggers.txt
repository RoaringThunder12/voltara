# v0.9.0: bounded, evidence-based household checks. No world searches.
dhm_v090_school_has_reason_trigger = {
 OR = {
  NOT = { has_variable = dhm090_review_age }
  NOT = { exists = var:dhm090_review_tutor }
  AND = { age >= 12 var:dhm090_review_age < 12 }
  AND = { age >= 14 var:dhm090_review_age < 14 }
  any_relation = { type = dhm_specialist_tutor exists = prev.var:dhm090_review_tutor NOT = { this = prev.var:dhm090_review_tutor } }
  AND = { has_focus = education_diplomacy NOT = { var:dhm090_review_focus ?= flag:diplomacy } }
  AND = { has_focus = education_martial NOT = { var:dhm090_review_focus ?= flag:martial } }
  AND = { has_focus = education_stewardship NOT = { var:dhm090_review_focus ?= flag:stewardship } }
  AND = { has_focus = education_intrigue NOT = { var:dhm090_review_focus ?= flag:intrigue } }
  AND = { has_focus = education_learning NOT = { var:dhm090_review_focus ?= flag:learning } }
 }
}
# Subject = child. Retention never protects an absent, hostile or incapable tutor.
dhm_v090_keep_capable_tutor_trigger = {
 has_global_variable = dhm090_keep_capable_tutors
 any_relation = {
  type = dhm_specialist_tutor
  is_alive = yes is_imprisoned = no is_hostage = no
  NOT = { has_trait = incapable }
  is_in_the_same_court_as = scope:dhm_manager
  NOT = { has_relation_rival = prev }
  NOT = { has_relation_nemesis = prev }
  learning >= 8
  OR = {
   AND = { diplomacy >= 12 prev = { has_focus = education_diplomacy } }
   AND = { martial >= 12 prev = { has_focus = education_martial } }
   AND = { stewardship >= 12 prev = { has_focus = education_stewardship } }
   AND = { intrigue >= 12 prev = { has_focus = education_intrigue } }
   AND = { learning >= 12 prev = { has_focus = education_learning } }
  }
 }
}

# Current scope = letter writer. Personality never overrides actual hostility.
dhm_v090_rc2_letter_marriage_strained_trigger = {
 any_spouse = {
  is_alive = yes
  OR = {
   opinion = { target = prev value < 0 }
   has_relation_rival = prev
   has_relation_nemesis = prev
   prev = { opinion = { target = prev value < 0 } }
   prev = { has_relation_rival = prev }
   prev = { has_relation_nemesis = prev }
  }
 }
}

   is_alive = yes

    stress_level >= 2
    is_imprisoned = yes
    is_hostage = yes
    has_trait = incapable
    opinion = { target = prev value < 20 }
    prev = { opinion = { target = prev value < 20 } }
   }
