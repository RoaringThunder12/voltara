}
# Current character is the initiator, saved scope is the chosen court.

 lam_ff_not_head_trigger = yes
 scope:lam_ff_lord ?= { is_alive = yes is_ruler = yes }
 OR = { this = scope:lam_ff_lord liege ?= scope:lam_ff_lord }
 lam_ff_can_found_in_trigger = { LORD = scope:lam_ff_lord }
}

 lam_ff_founder_current_trigger = yes

}

  liege ?= scope:lam_ff_lord
