mde_can_be_kingsguard_trigger = {
    is_ruler = no
    is_married = no
	is_capable_adult = yes
	NOT = { has_trait = dragon }
	NOR = {
        has_trait = kingsguard
		has_trait = nightswatch
		has_trait = order_member #Go away Areo Hotah
		has_trait = devoted
		has_trait = septon
		has_trait = maester
	}
    OR = {
        is_within_diplo_range = { CHARACTER = title:h_voltara_severos.holder }
        is_vassal_or_below_of = title:h_voltara_severos.holder
    }
    valid_kingsguard_gender_trigger = yes
}
mde_player_holds_iron_throne_trigger = {
    exists = title:h_voltara_severos.holder
    title:h_voltara_severos.holder = {
        is_ai = no
    }
}
mde_kingsguard_candidate_faith_trigger = {
