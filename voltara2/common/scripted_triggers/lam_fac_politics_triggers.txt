		}

lam_pol_no_rival_trigger = {
	trigger_if = {
		limit = {
			is_alive = yes
			exists = liege
		}
		save_temporary_scope_as = lam_pol_rival_head
		liege = {
			NOT = {
				any_in_list = {
					variable = lam_bloc_leads
					trigger_if = {
						limit = {
							is_alive = yes
						}
						has_variable = lam_bloc_lead
						liege ?= scope:lam_pol_rival_head.liege
						NOT = {
							this = scope:lam_pol_rival_head
						}
						NOT = {
							var:lam_fac_stage ?= 9
						}
						NOT = {
							has_variable = lam_fac_disband
						}
						save_temporary_scope_as = lam_pol_rival_other
						NOT = {
							is_target_in_variable_list = {
								name = lam_fac_allies
								target = scope:lam_pol_rival_head
							}
							scope:lam_pol_rival_head = {
								is_target_in_variable_list = {
									name = lam_fac_allies
									target = scope:lam_pol_rival_other
								}
							}
						}
						trigger_if = {
							limit = {
								has_variable = lam_fac_share
								has_variable = lam_cr_share
							}
							OR = {
								var:lam_fac_share > lam_tune_seize_rival_share
								var:lam_cr_share > lam_tune_pol_rival_share
							}
						}
						trigger_else = {
							always = yes
						}
					}
					trigger_else = {
						always = no
					}
				}
			}
		}
	}
	trigger_else = {
		always = no
	}
}
