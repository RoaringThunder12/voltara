# v0.8.0 continuity checks. Every memory lookup is bounded to event-carried
# participants and first revalidates the living people and current relationship.

# Timed globals lose two ticks per natural day in the tested 1.19.0.6 runtime.
# An absolute day-count deadline stays correct across pulses, saves and rulers.
# Retain expired deadlines: their presence prevents old flags restarting the wait.
dhm_v080_triumph_cooldown_active_trigger = {
	OR = {
		AND = {
			has_global_variable = dhm_v080_triumph_until
			global_var:dhm_v080_triumph_until > { value = current_date }
		}
		AND = {
			NOT = { has_global_variable = dhm_v080_triumph_until }
			OR = {
				has_global_variable = dhm_v080_triumph_recent
				has_character_flag = dhm_v080_triumph_recent
			}
		}
	}
}

# Current scope = the human manager at the two-year relationship midpoint.
# Continue only while the relationship named at the opening still makes sense.
dhm_v080_relationship_arc_context_valid_trigger = {
	exists = scope:dhm_arc_first
	exists = scope:dhm_arc_second
	scope:dhm_arc_first = { is_alive = yes }
	scope:dhm_arc_second = { is_alive = yes }
	OR = {
		AND = {
			dhm_v071_choice_is_trigger = { KEY = dhm071_arc_kind VALUE = flag:siblings LEGACY = dhm_arc_siblings_context }
			scope:dhm_arc_first = { is_in_the_same_court_as = root is_sibling_of = scope:dhm_arc_second }
			scope:dhm_arc_second = { is_in_the_same_court_as = root }
		}
		AND = {
			dhm_v071_choice_is_trigger = { KEY = dhm071_arc_kind VALUE = flag:tutor LEGACY = dhm_arc_tutor_context }
			scope:dhm_arc_first = { is_in_the_same_court_as = root }
			scope:dhm_arc_second = { is_in_the_same_court_as = root this != root }
		}
		AND = {
			dhm_v071_choice_is_trigger = { KEY = dhm071_arc_kind VALUE = flag:inlaw LEGACY = dhm_arc_inlaw_context }
			scope:dhm_arc_first = { is_in_the_same_court_as = root is_spouse_of = scope:dhm_arc_second }
			scope:dhm_arc_second = { is_in_the_same_court_as = root }
		}
		AND = {
			dhm_v071_choice_is_trigger = { KEY = dhm071_arc_kind VALUE = flag:estranged LEGACY = dhm_arc_estranged_context }
			scope:dhm_arc_first = { is_close_or_extended_family_of = root }
			scope:dhm_arc_second = { is_close_or_extended_family_of = root }
		}
	}
}

# Current scope = the officer; root = the human manager receiving the story.
# The post and the manager's authoritative appointment must still agree.
dhm_v080_story_officer_current_trigger = {
	is_alive = yes
	is_in_the_same_court_as = root
	OR = {
		AND = { has_character_flag = dhm_office_steward root = { has_variable = dhm_officer_steward var:dhm_officer_steward ?= prev } }
		AND = { has_character_flag = dhm_office_matchmaker root = { has_variable = dhm_officer_matchmaker var:dhm_officer_matchmaker ?= prev } }
		AND = { has_character_flag = dhm_office_tutor root = { has_variable = dhm_officer_tutor var:dhm_officer_tutor ?= prev } }
		AND = { has_character_flag = dhm_office_chronicler root = { has_variable = dhm_officer_chronicler var:dhm_officer_chronicler ?= prev } }
	}
}
