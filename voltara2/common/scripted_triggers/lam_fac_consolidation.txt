lam_cs_join_compatible_trigger = {
	trigger_if = {
		limit = {
			exists = scope:lam_cs_join_head
			is_alive = yes
		}
		trigger_if = {
			limit = {
				scope:lam_cs_join_head = {
					is_alive = yes
				}
			}
			trigger_if = {
				limit = {
					has_variable = lam_gui_att
					scope:lam_cs_join_head = {
						has_variable = lam_gui_att
					}
				}
				scope:lam_cs_join_head = {
					lam_fac_cause_of_trigger = {
						WHO = prev
					}
				}
				OR = {
					AND = {
						var:lam_gui_att >= 25
						scope:lam_cs_join_head = {
							var:lam_gui_att >= 25
						}
					}
					AND = {
						var:lam_gui_att <= -15
						scope:lam_cs_join_head = {
							var:lam_gui_att <= -15
						}
					}
					AND = {
						var:lam_gui_att > -15
						var:lam_gui_att < 25
						scope:lam_cs_join_head = {
							var:lam_gui_att > -15
							var:lam_gui_att < 25
						}
					}
				}
				NOT = {
					has_relation_rival = scope:lam_cs_join_head
				}
			}
			trigger_else = {
				always = no
			}
		}
		trigger_else = {
			always = no
		}
	}
	trigger_else = {
		always = no
	}
}

		always = no
	}
}
lam_cs_pair_trigger = {
	trigger_if = {
		limit = {
			exists = $OTHER$
		}
		save_temporary_scope_as = lam_cs_self
		NOT = {
			lam_fac_issue_opposite_side_trigger = {
				A = scope:lam_cs_self
				B = $OTHER$
			}
		}
	}
	trigger_if = {
		limit = {
			exists = $OTHER$
		}
		trigger_if = {
			limit = {
				lam_cs_head_trigger = yes
				$OTHER$ = {
					lam_cs_head_trigger = yes
				}
			}
			trigger_if = {
				limit = {
					has_variable = lam_fac_att
					$OTHER$ = {
						has_variable = lam_fac_att
					}
				}
				NOT = {
					this = $OTHER$
				}
				liege = $OTHER$.liege
				NOT = {
					has_relation_rival = $OTHER$
				}
				NOT = {
					has_relation_nemesis = $OTHER$
				}
				$OTHER$ = {
					NOT = {
						has_relation_rival = prev
					}
					NOT = {
						has_relation_nemesis = prev
					}
				}
				NOT = {
					is_target_in_variable_list = {
						name = lam_fac_foes
						target = $OTHER$
					}
				}
				$OTHER$ = {
					NOT = {
						is_target_in_variable_list = {
							name = lam_fac_foes
							target = prev
						}
					}
				}
				opinion = {
					target = $OTHER$
					value >= lam_cs_opinion_value
				}
				$OTHER$ = {
					opinion = {
						target = prev
						value >= lam_cs_opinion_value
					}
				}
				lam_cs_agenda_compatible_trigger = {
					OTHER = $OTHER$
				}
				lam_cs_alignment_compatible_trigger = {
					OTHER = $OTHER$
				}
				trigger_if = {
					limit = {
						exists = var:lam_fac_heir
					}
					$OTHER$ = {
						var:lam_fac_heir ?= prev.var:lam_fac_heir
					}
				}
				trigger_else = {
					$OTHER$ = {
						NOT = {
							exists = var:lam_fac_heir
						}
					}
				}
				NOT = {
					lam_cs_parent_grievance_trigger = {
						OTHER = $OTHER$
					}
				}
				$OTHER$ = {
					NOT = {
						lam_cs_parent_grievance_trigger = {
							OTHER = prev
						}
					}
				}
				trigger_if = {
					limit = {
						OR = {
							var:lam_fac_agenda ?= flag:zealot
							$OTHER$ = {
								var:lam_fac_agenda ?= flag:zealot
							}
						}
					}
					faith = $OTHER$.faith
				}
				trigger_if = {
					limit = {
						OR = {
							var:lam_fac_agenda ?= flag:minority
							$OTHER$ = {
								var:lam_fac_agenda ?= flag:minority
							}
						}
					}
					faith = $OTHER$.faith
					culture = $OTHER$.culture
				}
				trigger_if = {
					limit = {
						OR = {
							var:lam_fac_agenda ?= flag:parochial
							$OTHER$ = {
								var:lam_fac_agenda ?= flag:parochial
							}
						}
					}
					culture = $OTHER$.culture
				}
			}
			trigger_else = {
				always = no
			}
		}
		trigger_else = {
			always = no
		}
	}
	trigger_else = {
		always = no
	}
}

lam_cs_agenda_compatible_trigger = {
	trigger_if = {
		limit = {
			has_variable = lam_fac_agenda
			$OTHER$ = {
				has_variable = lam_fac_agenda
			}
		}
		OR = {
			var:lam_fac_agenda = $OTHER$.var:lam_fac_agenda
			AND = {
				OR = {
					var:lam_fac_agenda ?= flag:belligerent
					var:lam_fac_agenda ?= flag:glory_hound
				}
				$OTHER$ = {
					OR = {
						var:lam_fac_agenda ?= flag:belligerent
						var:lam_fac_agenda ?= flag:glory_hound
					}
				}
			}
			AND = {
				OR = {
					var:lam_fac_agenda ?= flag:parochial
					var:lam_fac_agenda ?= flag:barons_and_minor_landholders
				}
				$OTHER$ = {
					OR = {
						var:lam_fac_agenda ?= flag:parochial
						var:lam_fac_agenda ?= flag:barons_and_minor_landholders
					}
				}
			}
			AND = {
				OR = {
					var:lam_fac_agenda ?= flag:courtly
					var:lam_fac_agenda ?= flag:none
				}
				$OTHER$ = {
					OR = {
						var:lam_fac_agenda ?= flag:courtly
						var:lam_fac_agenda ?= flag:none
					}
				}
			}
		}
	}
	trigger_else = {
		always = no
	}
}
lam_cs_alignment_compatible_trigger = {
	NOT = {
		AND = {
			var:lam_fac_att >= 25
			$OTHER$ = {
				var:lam_fac_att <= -15
			}
		}
	}
	NOT = {
		AND = {
			var:lam_fac_att <= -15
			$OTHER$ = {
				var:lam_fac_att >= 25
			}
		}
	}
	var:lam_fac_att >= {
		value = $OTHER$.var:lam_fac_att
		subtract = lam_cs_alignment_gap_value
	}
	var:lam_fac_att <= {
		value = $OTHER$.var:lam_fac_att
		add = lam_cs_alignment_gap_value
	}
}
