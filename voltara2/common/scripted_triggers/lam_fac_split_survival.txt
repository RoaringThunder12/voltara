lam_ss_commitment_trigger = {
	trigger_if = {
		limit = {
			is_alive = yes
		}
		trigger_if = {
			limit = {
				is_ai = yes
				exists = var:lam_ss_home
				exists = var:lam_bloc
				has_variable = lam_ss_until
			}
			trigger_if = {
				limit = {
					var:lam_ss_home = {
						is_alive = yes
						exists = liege
					}
				}
				var:lam_bloc = var:lam_ss_home
				var:lam_ss_until > lam_now_m
				var:lam_ss_lord ?= var:lam_ss_home.liege
				OR = {
					liege ?= var:lam_ss_home.liege
					is_courtier_of = var:lam_ss_home.liege
				}
				var:lam_ss_home = {
					has_variable = lam_bloc_lead
					NOT = {
						has_variable = lam_fac_disband
					}
				}
				NOT = {
					has_variable = lam_fac_unbound
				}
				NOT = {
					has_variable = lam_fac_let_go
				}
				NOT = {
					has_relation_rival = var:lam_ss_home
				}
				opinion = {
					target = var:lam_ss_home
					value > -40
				}
				AND = {
					trigger_if = {
						limit = {
							var:lam_ss_home = {
								has_variable = lam_fac_rec
							}
						}
						var:lam_ss_record ?= var:lam_ss_home.var:lam_fac_rec
					}
					trigger_else = {
						var:lam_ss_home = {
							has_variable = lam_hi_record_evicted
						}
						NOT = {
							has_variable = lam_ss_record
						}
					}
				}
				trigger_if = {
					limit = {
						var:lam_ss_topic ?= flag:heir
					}
					trigger_if = {
						limit = {
							exists = var:lam_ss_alternative
						}
						var:lam_ss_alternative = {
							is_alive = yes
						}
						var:lam_ss_home = {
							var:lam_fac_heir ?= prev.var:lam_ss_alternative
						}
					}
					trigger_else = {
						always = no
					}
				}
				trigger_else_if = {
					limit = {
						var:lam_ss_side ?= flag:fair
					}
					var:lam_ss_home = {
						NOT = {
							has_variable = lam_fac_hoarded
						}
					}
					NOT = {
						AND = {
							trigger_if = {
								limit = {
									has_variable = lam_st_overlooked
								}
								var:lam_st_overlooked >= 2
								var:lam_st_reward_head ?= var:lam_ss_home
							}
							trigger_else = {
								always = no
							}
						}
					}
				}
				trigger_else_if = {
					limit = {
						var:lam_ss_side ?= flag:pressure
					}
					var:lam_ss_home = {
						NOR = {
							var:lam_fac_stage ?= 5
							var:lam_fac_stage ?= 6
						}
					}
				}
				trigger_else_if = {
					limit = {
						var:lam_ss_side ?= flag:restraint
					}
					var:lam_ss_home = {
						NOR = {
							var:lam_fac_stage ?= 3
							var:lam_fac_stage ?= 4
						}
					}
				}
				trigger_else = {
					always = yes
				}
			}
			trigger_else = {
				always = no
			}
		}
		trigger_else = {
			always = no
		}
	}
	trigger_else = {
		always = no
	}
}
lam_ss_establishing_trigger = {
	trigger_if = {
		limit = {
			is_alive = yes
		}
		trigger_if = {
			limit = {
				has_variable = lam_ss_birth_m
				has_variable = lam_fac_count
				exists = liege
			}
			var:lam_ss_birth_m > {
				value = lam_now_m
				subtract = 24
			}
			var:lam_fac_count >= 2
			any_in_list = {
				variable = lam_fac_members
				trigger_if = {
					limit = {
						is_alive = yes
					}
					var:lam_bloc ?= prev
					NOT = {
						this = prev
					}
					OR = {
						liege ?= prev.liege
						is_courtier_of = prev.liege
					}
				}
				trigger_else = {
					always = no
				}
			}
		}
		trigger_else = {
			always = no
		}
	}
	trigger_else = {
		always = no
	}
}
