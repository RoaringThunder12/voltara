# Scope : character
T_kCAFG_should_start_with_intolerant_rite_law = {
	NOR = {
		culture = { has_cultural_tradition = tradition_religion_blending }
		culture = { has_cultural_tradition = tradition_steppe_tolerance }
		rite = { rite_has_tenet = tenet_pacifism }
		rite = { rite_has_tenet = tenet_dharmic_pacifism }
		rite = { rite_has_doctrine = doctrine_pluralism_pluralistic }
	}
	OR = {
		culture = { has_cultural_parameter = same_faith_inheritance_only }
		culture = { has_cultural_tradition = tradition_zealous_people }
		rite = { rite_has_tenet = tenet_pursuit_of_power }
		rite = { rite_has_tenet = tenet_unrelenting_faith }
		rite = { rite_has_tenet = tenet_warmonger }
		ai_zeal >= 90
		AND = {
			ai_zeal > 25
			rite = { rite_has_doctrine = doctrine_pluralism_fundamentalist }
		}
	}
}

# Scope : character
T_kCAFG_should_start_with_tolerant_rite_law = {
	NOR = {
		culture = { has_cultural_parameter = same_faith_inheritance_only }
		culture = { has_cultural_tradition = tradition_zealous_people }
		culture = { has_cultural_tradition = tradition_by_the_sword }
		rite = { rite_has_tenet = tenet_pursuit_of_power }
		rite = { rite_has_tenet = tenet_unrelenting_faith }
		rite = { rite_has_tenet = tenet_warmonger }
		rite = { rite_has_doctrine = doctrine_pluralism_fundamentalist }
	}
	OR = {
		culture = { has_cultural_tradition = tradition_religion_blending }
		culture = { has_cultural_tradition = tradition_steppe_tolerance }
		rite = { rite_has_tenet = tenet_pacifism }
		rite = { rite_has_tenet = tenet_dharmic_pacifism }
		ai_zeal <= -90
		AND = {
			ai_zeal > 25
			rite = { rite_has_doctrine = doctrine_pluralism_pluralistic }
		}
	}
}

T_kCAFG_can_title_have_rite_tolerance_law = {
    NOT = { has_game_rule = kCAFG_tolerance_laws_off }
    tier >= tier_county
	NOT = { is_temporal_head_of_faith_trigger = yes }
	exists = holder
	holder = {
		government_has_flag = government_is_settled
		is_landed = yes
	}
}

T_kCAFG_can_have_rite_tolerance_law = {
    NOT = { has_game_rule = kCAFG_tolerance_laws_off }
	is_landed = yes
    primary_title.tier >= tier_county
	government_has_flag = government_is_settled
    NOT = { primary_title = { is_temporal_head_of_faith_trigger = yes } }
}

T_kCAFG_overrides_rite_tolerance_law = {
	trigger_if = {
		limit = { has_realm_law_flag = max_authority_level }
		always = yes
	}
	trigger_else_if = {
		limit = { realm_law_use_tribal_authority = yes }
		custom_description = {
			text = kCAFG_overrides_culture_tolerance_law
			OR = {
				has_realm_law = tribal_authority_3
			}
		}
	}
	trigger_else_if = {
		limit = { realm_law_use_imperial_bureaucracy = yes }
		custom_description = {
			text = kCAFG_overrides_culture_tolerance_law
			OR = {
				has_realm_law = imperial_bureaucracy_2
				has_realm_law = imperial_bureaucracy_3
			}
		}
	}
	trigger_else_if = {
		limit = { realm_law_use_meritocratic_bureaucracy = yes }
		custom_description = {
			text = kCAFG_overrides_culture_tolerance_law
			OR = {
				has_realm_law = meritocratic_bureaucracy_2
				has_realm_law = meritocratic_bureaucracy_3
			}
		}
	}
	trigger_else_if = {
		limit = { realm_law_use_japanese_bureaucracy = yes }
		custom_description = {
			text = kCAFG_overrides_culture_tolerance_law
			OR = {
				has_realm_law = japanese_bureaucracy_2
				has_realm_law = japanese_bureaucracy_3
			}
		}
	}
	trigger_else_if = {
		limit = { realm_law_use_celestial_bureaucracy = yes }
		custom_description = {
			text = kCAFG_overrides_culture_tolerance_law
			OR = {
				has_realm_law = celestial_bureaucracy_2
				has_realm_law = celestial_bureaucracy_3
			}
		}
	}
	trigger_else_if = {
		limit = { realm_law_use_crown_authority = yes }
		custom_description = {
			text = kCAFG_overrides_culture_tolerance_law
			OR = {
				has_realm_law = crown_authority_2
				has_realm_law = crown_authority_3
			}
		}
	}
	trigger_else = {
		always = no
	}
}

T_kCAFG_has_rite_tolerance_realm_law = {
	NOT = { has_global_variable = kCAFG_ignore_tolerance_laws }
	NOT = { has_character_flag = kCAFG_TMP_applying_rite_indifference_law }
	OR = {
		has_realm_law_flag = F_kCAFG_tolerant_to_other_rites
		has_character_flag = kCAFG_TMP_applying_rite_tolerance_law
	}
}

T_kCAFG_has_rite_intolerance_realm_law = {
	NOT = { has_global_variable = kCAFG_ignore_tolerance_laws }
	NOT = { has_character_flag = kCAFG_TMP_applying_rite_indifference_law }
	OR = {
		has_realm_law_flag = F_kCAFG_intolerant_to_other_rites
		has_character_flag = kCAFG_TMP_applying_rite_intolerance_law
	}
}

T_kCAFG_can_decrease_rite_tolerance_law_trigger = {
	T_kCAFG_can_change_rite_tolerance_law_trigger = yes
	custom_description = {
		text = "no_powerful_vassal_with_negative_opinion"
		NOT = {
			any_vassal = {
				is_powerful_vassal = yes
				opposes_succession_law_change_trigger = yes
			}
		}
	}
}

T_kCAFG_can_increase_rite_tolerance_law_trigger = {
	T_kCAFG_can_change_rite_tolerance_law_trigger = yes
	custom_description = {
		text = "no_powerful_vassal_with_negative_opinion"
		NOT = {
			any_vassal = {
				is_powerful_vassal = yes
				opposes_succession_law_change_trigger = yes
				NAND = {
					NOT = { has_same_culture_as = liege }
					ai_zeal < V_kCAFG_ai_law_change_negative_threshold
				}
			}
		}
	}
}
