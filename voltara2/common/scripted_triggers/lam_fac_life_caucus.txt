lam_lf_caucus_current_trigger = {
	is_alive = yes
	has_variable = lam_bloc_lead
	var:lam_bloc ?= this
	NOT = {
		has_variable = lam_fac_disband
	}
	var:lam_lf_caucus_lord ?= liege
	has_variable = lam_lf_caucus_until
	var:lam_lf_caucus_until > lam_now_m
	OR = {
		AND = {
			NOT = {
				has_variable = lam_fac_rec
			}
			NOT = {
				has_variable = lam_lf_caucus_record
			}
		}
		var:lam_lf_caucus_record ?= var:lam_fac_rec
	}
	var:lam_lf_caucus_rep ?= {
		is_alive = yes
		is_adult = yes
		is_imprisoned = no
		var:lam_bloc ?= prev
		liege ?= prev.liege
		NOT = {
			has_variable = lam_bloc_lead
		}
	}
}
lam_lf_pledge_current_trigger = {
	lam_lf_caucus_current_trigger = yes
	has_variable = lam_lf_pledge_until
	var:lam_lf_pledge_until > lam_now_m
	var:lam_lf_pledged_member ?= var:lam_lf_caucus_rep
}
lam_lf_caucus_window_trigger = {
	lam_lf_caucus_current_trigger = yes
	is_imprisoned = no
	has_variable = lam_lf_caucus_open
	var:lam_lf_caucus_token ?= scope:lam_lf_caucus_token
}
