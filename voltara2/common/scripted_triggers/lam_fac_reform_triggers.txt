lam_cr_member_trigger = {
    is_alive = yes
    is_adult = yes
    is_imprisoned = no
    liege ?= scope:lam_cr_lord
    OR = {
        this = scope:lam_cr_head
        AND = { var:lam_bloc ?= scope:lam_cr_head OR = { var:lam_fac_tier ?= 1 var:lam_fac_tier ?= 2 } }
    }
    NOT = { AND = { this = scope:lam_cr_head scope:lam_cr_head = { var:lam_cr_issue ?= flag:confidence } } }
}
lam_cr_owner_valid_trigger = {
    is_alive = yes
    is_adult = yes
    is_imprisoned = no
    has_variable = lam_bloc_lead
    has_variable = lam_cr_active
    NOT = { has_variable = lam_fac_disband }
    var:lam_bloc ?= this
    var:lam_cr_lord ?= { is_alive = yes }
    liege ?= var:lam_cr_lord
    OR = {
        AND = { NOT = { has_variable = lam_cr_record } NOT = { has_variable = lam_fac_rec } }
        AND = { has_variable = lam_cr_record has_variable = lam_fac_rec var:lam_cr_record = var:lam_fac_rec }
    }
}
# Event-scoped token prevents an old window from voting in a later council.
lam_cr_ballot_valid_trigger = {
    exists = scope:lam_cr_head
    exists = scope:lam_cr_token
    scope:lam_cr_head = { lam_cr_owner_valid_trigger = yes var:lam_cr_id = scope:lam_cr_token }
    var:lam_cr_ballot_head ?= scope:lam_cr_head
    var:lam_cr_ballot_id ?= scope:lam_cr_token
    var:lam_cr_choice ?= flag:pending
    lam_cr_member_trigger = yes
}

		has_variable = lam_fac_disband

		is_alive = yes

	}
	NOT = {

			name = lam_fac_allies
			target = scope:lam_pr_reactor
		}
	}
	save_temporary_scope_as = lam_pd_pact_test_a
