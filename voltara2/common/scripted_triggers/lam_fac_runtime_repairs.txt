lam_pb_head_belongs_trigger = {
	trigger_if = {
		limit = {
			is_alive = yes
			exists = $COURT$
		}
		has_variable = lam_bloc_lead
		NOT = {
			has_variable = lam_fac_disband
		}
		NOT = {
			this = $COURT$
		}
		trigger_if = {
			limit = {
				exists = var:lam_wc_war
			}
			var:lam_wc_home ?= $COURT$
			var:lam_wc_war = {
				is_participant = prev
			}
		}
		trigger_else = {
			is_ruler = yes
			OR = { liege ?= $COURT$ AND = { has_variable = vip_imperial_bloc_head voltara_in_hegemony = yes $COURT$ = { vip_court = yes } } }
		}
	}
	trigger_else = {
		always = no
	}
}
