lam_gg_pressure_trigger = {
	trigger_if = {
		limit = {
			lam_ev_member_trigger = yes
			is_ai = yes
		}
		trigger_if = {
			limit = {
				scope:lam_ev_head.liege = {
					is_alive = yes
				}
			}
			OR = {
				scope:lam_ev_head = {
					OR = {
						var:lam_fac_stage ?= 5
						var:lam_fac_stage ?= 6
						var:lam_fac_stage ?= 8
					}
				}
				AND = {
					scope:lam_ev_head = {
						var:lam_fac_stage ?= 7
					}
					lam_dd_alignment_gap_value <= -30
				}
			}
			lam_gg_crown_alignment_value <= -10
			NOT = {
				is_councillor_of = scope:lam_ev_head.liege
			}
		}
		trigger_else = {
			always = no
		}
	}
	trigger_else = {
		always = no
	}
}
lam_gg_restraint_trigger = {
	trigger_if = {
		limit = {
			lam_ev_member_trigger = yes
			is_ai = yes
		}
		trigger_if = {
			limit = {
				scope:lam_ev_head.liege = {
					is_alive = yes
				}
			}
			OR = {
				scope:lam_ev_head = {
					OR = {
						var:lam_fac_stage ?= 2
						var:lam_fac_stage ?= 3
					}
				}
				AND = {
					scope:lam_ev_head = {
						var:lam_fac_stage ?= 7
					}
					lam_dd_alignment_gap_value >= 30
				}
			}
			lam_gg_crown_alignment_value >= 20
		}
		trigger_else = {
			always = no
		}
	}
	trigger_else = {
		always = no
	}
}
lam_gg_rewarded_trigger = {
	trigger_if = {
		limit = {
			has_variable = lam_lf_reward_until
		}
		var:lam_lf_reward_until > lam_now_m
		var:lam_lf_reward_head ?= scope:lam_ev_head
	}
	trigger_else = {
		always = no
	}
}
lam_gg_service_trigger = {
	trigger_if = {
		limit = {
			lam_ev_member_trigger = yes
			is_ai = yes
		}
		trigger_if = {
			limit = {
				scope:lam_ev_head.liege = {
					is_alive = yes
				}
			}
			trigger_if = {
				limit = {
					has_variable = lam_cr_service
				}
				var:lam_cr_service_head ?= scope:lam_ev_head
				var:lam_cr_service >= 6
				OR = {
					var:lam_fac_tier ?= 1
					var:lam_fac_tier ?= 2
				}
				OR = {
					has_trait = ambitious
					has_trait = greedy
					has_trait = just
				}
				NOT = {
					is_councillor_of = scope:lam_ev_head.liege
				}
				NOT = {
					lam_gg_rewarded_trigger = yes
				}
				scope:lam_ev_head = {
					is_councillor_of = liege
				}
			}
			trigger_else = {
				always = no
			}
		}
		trigger_else = {
			always = no
		}
	}
	trigger_else = {
		always = no
	}
}
lam_gg_exclusion_trigger = {
	trigger_if = {
		limit = {
			lam_ev_member_trigger = yes
			is_ai = yes
		}
		trigger_if = {
			limit = {
				has_variable = lam_st_overlooked
			}

			NOT = {
				is_councillor_of = scope:lam_ev_head.liege
			}
			NOT = {
				lam_gg_rewarded_trigger = yes
			}
		}
		trigger_else = {
			always = no
		}
	}
	trigger_else = {
		always = no
	}
}
lam_gg_assent_trigger = {
	trigger_if = {
		limit = {

		}
		var:lam_gg_assent_until > lam_now_m
		var:lam_gg_assent_head ?= scope:lam_ev_head
		var:lam_gg_assent_lord ?= scope:lam_ev_head.liege

		always = no
	}
}
lam_gg_saved_yes_trigger = {
	trigger_if = {
		limit = {
			is_alive = yes
			exists = scope:lam_cr_head
			exists = scope:lam_cr_lord
		}
		trigger_if = {
			limit = {
				scope:lam_cr_head = {
					is_alive = yes
				}
			}
			trigger_if = {
				limit = {
					scope:lam_cr_head = {
						has_variable = lam_cr_id
					}
					has_variable = lam_cr_ballot_id
					has_variable = lam_cr_weight
				}
				lam_cr_saved_voter_trigger = yes
				var:lam_cr_choice ?= flag:yes
			}
			trigger_else = {
				always = no
			}
		}
		trigger_else = {
			always = no
		}
	}
	trigger_else = {
		always = no
	}
}
lam_gg_current_cause_trigger = {
	trigger_if = {
		limit = {
			var:lam_gg_source ?= flag:pressure
		}
		lam_gg_pressure_trigger = yes
	}
	trigger_else_if = {
		limit = {
			var:lam_gg_source ?= flag:restraint
		}
		lam_gg_restraint_trigger = yes
	}
	trigger_else_if = {
		limit = {
			var:lam_gg_source ?= flag:service
		}
		lam_gg_service_trigger = yes
	}
	trigger_else_if = {
		limit = {
			var:lam_gg_source ?= flag:exclusion
		}
		lam_gg_exclusion_trigger = yes
	}
	trigger_else_if = {
		limit = {
			var:lam_gg_source ?= flag:issue
		}
		lam_is_dissent_live_trigger = {
			HEAD = scope:lam_ev_head
		}
	}
	trigger_else = {
		always = no
	}
}
