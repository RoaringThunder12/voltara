lam_lf_member_trigger = {
	is_alive = yes
	is_adult = yes
	is_imprisoned = no
	var:lam_bloc ?= scope:lam_lf_head
	liege ?= scope:lam_lf_head.liege
	OR = {
		this = scope:lam_lf_head
		var:lam_fac_tier ?= 1
		var:lam_fac_tier ?= 2
	}
}
lam_lf_office_person_trigger = {
	lam_lf_member_trigger = yes
	NOT = {
		is_councillor_of = liege
	}
	lam_st_office_available_trigger = yes
}
lam_lf_beneficiary_trigger = {
	trigger_if = {
		limit = {
			is_alive = yes
			exists = scope:lam_lf_head
		}
		trigger_if = {
			limit = {
				scope:lam_lf_head = {
					is_alive = yes
				}
			}
			trigger_if = {
				limit = {
					scope:lam_lf_head = {
						exists = var:lam_pd_lord
					}
				}
				is_adult = yes
				is_imprisoned = no
				liege ?= scope:lam_lf_head.var:lam_pd_lord
				OR = {
					var:lam_bloc ?= scope:lam_lf_head
					AND = {
						trigger_if = {
							limit = {
								scope:lam_lf_head = {
									lam_lf_exchange_current_trigger = yes
								}
							}
							this = scope:lam_lf_head.var:lam_lf_exchange_person
						}
						trigger_else = {
							always = no
						}
					}
				}
			}
			trigger_else = {
				always = no
			}
		}
		trigger_else = {
			always = no
		}
	}
	trigger_else = {
		always = no
	}
}
lam_lf_office_valid_trigger = {
	trigger_if = {
		limit = {
			is_alive = yes
			exists = var:lam_pd_lord
			exists = var:lam_lf_beneficiary
		}
		trigger_if = {
			limit = {
				var:lam_pd_lord = {
					is_alive = yes
				}
				var:lam_lf_beneficiary = {
					is_alive = yes
				}
			}
			save_temporary_scope_as = lam_lf_head
			var:lam_lf_beneficiary ?= {
				lam_lf_beneficiary_trigger = yes
				NOT = {
					is_councillor_of = liege
				}
			}
			OR = {
				AND = {
					var:lam_pd_seat ?= flag:marshal
					var:lam_pd_lord = {
						lam_nc_appoint_trigger = {
							SEAT = marshal
							CANDIDATE = scope:lam_lf_head.var:lam_lf_beneficiary
						}
					}
					AND = {
						trigger_if = {
							limit = {
								exists = var:lam_pd_incumbent
							}
							var:lam_pd_incumbent = {
								is_alive = yes
							}
							exists = var:lam_pd_lord.cp:councillor_marshal
							var:lam_pd_lord.cp:councillor_marshal ?= var:lam_pd_incumbent
						}
						trigger_else = {
							NOT = {
								has_variable = lam_pd_incumbent
							}
							NOT = {
								exists = var:lam_pd_lord.cp:councillor_marshal
							}
						}
					}
				}
				AND = {
					var:lam_pd_seat ?= flag:steward
					var:lam_pd_lord = {
						lam_nc_appoint_trigger = {
							SEAT = steward
							CANDIDATE = scope:lam_lf_head.var:lam_lf_beneficiary

					}

							NOT = {
								has_variable = lam_pd_incumbent
							}
							NOT = {
								exists = var:lam_pd_lord.cp:councillor_steward
							}
						}
					}
				}
				AND = {
					var:lam_pd_seat ?= flag:spymaster

						lam_nc_appoint_trigger = {
							SEAT = spymaster
							CANDIDATE = scope:lam_lf_head.var:lam_lf_beneficiary
						}
					}

							limit = {

							}

								is_alive = yes
							}
							exists = var:lam_pd_lord.cp:councillor_spymaster
							var:lam_pd_lord.cp:councillor_spymaster ?= var:lam_pd_incumbent
						}
						trigger_else = {
							NOT = {
								has_variable = lam_pd_incumbent
							}
							NOT = {
								exists = var:lam_pd_lord.cp:councillor_spymaster
							}
						}
					}
				}
				AND = {
					var:lam_pd_seat ?= flag:chancellor
					var:lam_pd_lord = {
						lam_nc_appoint_trigger = {
							SEAT = chancellor
							CANDIDATE = scope:lam_lf_head.var:lam_lf_beneficiary
