                        target = scope:kCAFG_target_rite

        limit = {
            is_at_war = yes
            any_character_war = {
                primary_defender.rite != prev.rite
                using_holy_war_cb_trigger = yes
                is_attacker = prev
                primary_defender.rite = {
                    save_temporary_scope_as = kCAFG_rite
                    scope:kCAFG_province = {
                        T_kCAFG_does_province_contain_rite = { RITE = scope:kCAFG_rite }
                    }
                }
            }
        }
        ordered_character_war = {
            limit = {
                primary_defender.rite != prev.rite
                using_holy_war_cb_trigger = yes
                is_attacker = prev
                primary_defender.rite = {
                    save_temporary_scope_as = kCAFG_rite
                    scope:kCAFG_province = {
                        T_kCAFG_does_province_contain_rite = { RITE = scope:kCAFG_rite }
                    }
                }
            }
            order_by = {
                value = 0
                casus_belli = {
                    every_target_title = {
                        if = { limit = { tier = tier_county } add = 1 }
                        else_if = { limit = { tier = tier_duchy } add = 5 }
                        else_if = { limit = { tier = tier_kingdom } add = 15 }
                        else_if = { limit = { tier = tier_empire } add = 50 }
                        else_if = { limit = { tier = tier_hegemony } add = 100 }
                    }
                }
            }
            primary_defender.rite = { save_temporary_scope_as = kCAFG_rite }
            casus_belli = {
                ordered_target_title = {
                    order_by = tier
                    max = 1
                    save_temporary_scope_as = highest_tiered_title # used by V_kCAFG_rite_tension_on_religious_warring_quarterly
                }
            }
            scope:kCAFG_province = {
                add = {
                    value = V_kCAFG_rite_tension_on_religious_warring_quarterly
                    multiply = {
                        value = 0.1
                        add = {
                            value = V_kCAFG_province_rite_factor
                            multiply = {
                                value = 1
                                subtract = 0.1
                            }
                        }
                    }
                    divide = 3 # Quarterly to monthly
                }
            }
        }
    }
}

# Scope : province
V_kCAFG_rite_tension_monthly_drift = {
    value = 0
    if = {
        limit = {
            OR = {
                V_kCAFG_rite_tension_difference_from_target > V_kCAFG_rite_tension_drift_minimum_value
                V_kCAFG_rite_tension_difference_from_target < V_kCAFG_rite_tension_drift_minimum_value_negative
            }
        }
        value = V_kCAFG_rite_tension_difference_from_target
        multiply = V_kCAFG_rite_tension_monthly_drift_factor
        if = {
            limit = { var:kCAFG_target_rite_tension > var:kCAFG_rite_tension }
            min = V_kCAFG_rite_tension_drift_minimum_value
        }
        else_if = {
            limit = { var:kCAFG_target_rite_tension < var:kCAFG_rite_tension }
            max = V_kCAFG_rite_tension_drift_minimum_value_negative
        }
    }
    else_if = {
        limit = { var:kCAFG_rite_tension != var:kCAFG_target_rite_tension }
        value = V_kCAFG_rite_tension_difference_from_target
    }
}
