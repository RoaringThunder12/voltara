lam_pd_own_force_value = {
	value = current_military_strength
	save_temporary_scope_as = lam_pd_force_head
	every_in_list = {
		variable = lam_fac_members
		limit = {
			is_alive = yes
			is_imprisoned = no
			AND = {
				trigger_if = {
					limit = {
						exists = scope:lam_pd_force_head
					}
					var:lam_bloc ?= scope:lam_pd_force_head
				}
				trigger_else = {
					always = no
				}
			}
			AND = {
				trigger_if = {
					limit = {
						exists = scope:lam_pd_force_head.liege
					}
					liege ?= scope:lam_pd_force_head.liege
				}
				trigger_else = {
					always = no
				}
			}
			NOT = {
				AND = {
					trigger_if = {
						limit = {
							exists = scope:lam_pd_force_head
						}
						this = scope:lam_pd_force_head
					}
					trigger_else = {
						always = no
					}
				}
			}
			NOT = {
				is_at_war = yes
			}
		}
		add = current_military_strength
	}
}
