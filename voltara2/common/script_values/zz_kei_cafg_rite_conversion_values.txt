# Conversion multiplier for a rite converting in a province with no specific character associated with that conversion
# Scope : province
# kCAFG_converting_rite
V_kCAFG_province_rite_conversion_multiplier = {
	if = {
		limit = {
			NOT = { T_kCAFG_province_only_rite = { RITE = scope:kCAFG_converting_rite } }
		}
		save_temporary_scope_as = kCAFG_converted_province
		value = 1
		save_temporary_value_as = kCAFG_conversion_base_total
		multiply = {
			value = {
				value = 1
				desc = BASE_VALUE
			}

			## Converting Rites Modifiers
			add = scope:kCAFG_converting_rite.V_kCAFG_rite_conversion_converting_rite_modifiers

			## Converted Rites Modifiers
			add = {
				every_in_list = {
					variable = kCAFG_province_rite_structs
					limit = { scope:kCAFG_converting_rite != var:rite }
					add = {
						value = var:rite.V_kCAFG_rite_conversion_converted_rite_modifiers
						multiply = var:factor
					}
				}
				# rescale by excluding converting rite
				divide = {
					value = 1
					scope:kCAFG_converting_rite = { scope:kCAFG_converted_province = { subtract = V_kCAFG_province_prev_rite_factor } }
            		min = V_kCAFG_smallest_minority_factor
				}
				desc = kCAFG_task_conversion_various_rites
			}

			## Converted Cultures Modifiers
			add = {
				every_in_list = {
					variable = kCAFG_province_culture_structs
					add = {
						value = var:culture.V_kCAFG_rite_conversion_converted_culture_modifiers
						multiply = var:factor
					}
				}
				desc = kCAFG_task_conversion_various_cultures
			}

			min = 0
		}

		county = { add = V_kCAFG_rite_conversion_county_modifiers_add }

		## MULTIPLICATIVE MODIFIERS ##

		multiply = V_kCAFG_rite_conversion_fervor_multiplier

		multiply = V_kCAFG_rite_conversion_province_modifiers_mult

		county = { multiply = V_kCAFG_rite_conversion_county_modifiers_mult }

		scope:kCAFG_converting_rite = { multiply = V_kCAFG_rite_conversion_converting_rite_modifiers_mult }
	}
}

# Scope : province
# kCAFG_converting_rite
V_kCAFG_rite_conversion_fervor_multiplier = {
	value = 1
	add = {
		add = scope:kCAFG_converting_rite.V_kCAFG_fervor_factor
		subtract = {
			every_in_list = {
				variable = kCAFG_province_rite_structs
				limit = { scope:kCAFG_converting_rite != var:rite }
				add = {
					value = var:rite.V_kCAFG_fervor_factor
					multiply = var:factor
				}
			}
			# rescale by excluding converting rite
			divide = {
				value = 1
				scope:kCAFG_converting_rite = { scope:kCAFG_converted_province = { subtract = V_kCAFG_province_prev_rite_factor } }
            	min = V_kCAFG_smallest_minority_factor
			}
		}
		multiply = convert_faith_fervor_modifier_scale
		desc = kCAFG_rite_conversion_fervor_against_converting
	}
}
