republican_oligarchs_support_target = {
	value = {
		value = 50
		desc = "BASE"
	} 
	if = {
		limit = { is_lowborn = yes }
		subtract = 5
		desc = "republican_ruler_is_lowborn"
	}
	add = {
		value = prestige_level
		add = 1
		multiply = 5
		subtract = 10
		desc = "republican_ruler_prestige_level"
	}
	add = {
		value = dynasty.dynasty_prestige_level
		multiply = 2
		subtract = 4
		desc = "republican_ruler_dynasty_prestige_level"
	}
	add = {
		value = 0
		if = {	limit = { has_realm_law = republican_2_year_term_law } subtract = 10 }
		else_if = {	limit = { has_realm_law = republican_4_year_term_law } subtract = 5 }
		else_if = {	limit = { has_realm_law = republican_lifetime_term_law } add = 5 }
		desc = "republican_term_length"
	}
	add = {
		if = {	limit = { has_realm_law = republican_authority_0 } subtract = 5 }
		else_if = {	limit = { has_realm_law = republican_authority_2 } add = 5 }
		desc = "republican_authority"
	}

	subtract = {
		value = tyranny
		divide = 2
		desc = "republican_tyranny"
	}
	divide = 100
}
