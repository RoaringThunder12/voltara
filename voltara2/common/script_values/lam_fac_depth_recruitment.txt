lam_pd_recruit_interest_value = {
	value = 0
	if = {
		limit = {
			is_alive = yes
		}
		value = lam_tune_invite_base
		add = {
			value = "opinion(scope:lam_rc_head)"
			multiply = 0.4
		}
		if = {
			limit = {
				AND = {
					trigger_if = {
						limit = {
							exists = scope:lam_rc_head.house
						}
						house ?= scope:lam_rc_head.house
					}
					trigger_else = {
						always = no
					}
				}
			}
			add = 25
		}
		if = {
			limit = {
				has_relation_friend = scope:lam_rc_head
			}
			add = 20
		}
		if = {
			limit = {
				has_relation_rival = scope:lam_rc_head
			}
			add = -80
		}
		if = {
			limit = {
				gold < 0
			}
			add = 25
		}
		if = {
			limit = {
				gold < 100
			}
			add = 10
		}
		if = {
			limit = {
				has_trait = ambitious
			}
			add = 8
		}
		if = {
			limit = {
				has_trait = content
			}
			add = -12
		}
		if = {
			limit = {
				has_variable = lam_bloc
			}
			add = {
				value = "opinion(var:lam_bloc)"
				multiply = -0.35
			}
			if = {
				limit = {
					AND = {
						trigger_if = {
							limit = {
								exists = var:lam_bloc.house
							}
							house ?= var:lam_bloc.house
						}
						trigger_else = {
							always = no
						}
					}
				}
				add = -25
			}
			if = {
				limit = {
					var:lam_fac_tier ?= 1
				}
				add = -30
			}
			if = {
				limit = {
					has_variable = lam_fac_merit
				}
				add = -15
			}
			if = {
				limit = {
					is_councillor_of = liege
				}
				add = -20
			}
			if = {
				limit = {
					var:lam_bloc = {
						trigger_if = {
							limit = {
								is_alive = yes
							}
							has_variable = lam_fac_hoarded
						}
						trigger_else = {
							always = no
						}
					}
				}
				add = 25
			}
			if = {
				limit = {
					var:lam_bloc = {
						trigger_if = {
							limit = {
								is_alive = yes
							}
							has_variable = lam_fac_betrayed
						}
						trigger_else = {
							always = no
						}
					}
				}
				add = 20
			}
			if = {
				limit = {
					var:lam_bloc = {
						trigger_if = {
							limit = {
								is_alive = yes
							}
							has_variable = lam_fac_promise
						}
						trigger_else = {
							always = no
						}
					}
				}
				add = -15
			}
		}
		if = {
			limit = {
				scope:lam_rc_head = {
					trigger_if = {
						limit = {
							is_alive = yes
						}
						has_variable = lam_fac_betrayed
					}
					trigger_else = {
						always = no
					}
				}
			}
			add = -25
		}
		if = {
			limit = {
				scope:lam_rc_head = {
					trigger_if = {
						limit = {
							is_alive = yes
						}
						has_variable = lam_cr_share
					}
					trigger_else = {
						always = no
					}
				}
			}
			add = {
				value = scope:lam_rc_head.var:lam_cr_share
				multiply = 0.2
			}
		}
		subtract = lam_pd_independent_cost_value
		if = {
			limit = {
				has_variable = lam_fac_unbound
			}
			add = -100
		}
		if = {
			limit = {
				AND = {
					trigger_if = {
						limit = {
							exists = var:lam_bloc
						}
						var:lam_st_reward_head ?= var:lam_bloc
					}
					trigger_else = {
						always = no
					}
				}
				has_variable = lam_st_overlooked
			}
			add = {
				value = var:lam_st_overlooked
				multiply = 4
			}
		}
		if = {
			limit = {
				scope:lam_rc_head ?= {
					trigger_if = {
						limit = {
							is_alive = yes
						}
						has_variable = lam_lf_gov_rating
						var:lam_lf_gov_lord ?= liege
					}
					trigger_else = {
						always = no
					}
				}
			}
			add = {
				value = scope:lam_rc_head.var:lam_lf_gov_rating
				multiply = 2
			}
		}
	}
}
lam_pd_recruit_price_value = {
	value = minor_gold_value
	if = {
		limit = {
			var:lam_fac_tier ?= 1
		}
		multiply = 2
	}
	if = {
		limit = {
			has_trait = greedy
		}
		multiply = 1.5
	}
	if = {
		limit = {
			scope:lam_rc_head.diplomacy >= 16
		}
		multiply = 0.8
	}
	min = 25
	max = 150
}
