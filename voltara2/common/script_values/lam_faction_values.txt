# Love & Marriage - living factions: who a faction quietly wants on the throne next.
# This = a candidate (one of the lord's children, or the primary heir).
# scope:lam_fh_c = this, scope:lam_fh_lead = the faction's leader, scope:lam_fh_lord = the lord.
lam_fac_heir_score_value = {
	value = 0
	# What the leader thinks of them.
	add = {
		value = { scope:lam_fh_lead = { value = "opinion(scope:lam_fh_c)" } }
		multiply = 0.5
	}
	# The lawful heir carries the weight of the law - less so for those who want the crown humbled.
	if = {
		limit = { exists = scope:lam_fh_lord.primary_heir this = scope:lam_fh_lord.primary_heir }
		if = {
			limit = { scope:lam_fh_lead = { has_variable = lam_gui_att var:lam_gui_att <= -15 } }
			add = 5
		}
		else = { add = 25 }
		if = {
			limit = {
				OR = {
					scope:lam_fh_lead.var:lam_fac_agenda ?= flag:loyalist
					scope:lam_fh_lead.var:lam_fac_agenda ?= flag:dynastic
				}
			}
			add = 15
		}
	}
	# Blood: a child of their own house's daughter (or son) is their child too.
	if = {
		limit = {
			any_parent = {
				NOT = { this = scope:lam_fh_lord }
				OR = {
					this = scope:lam_fh_lead
					is_close_family_of = scope:lam_fh_lead
					AND = {
						exists = house
						exists = scope:lam_fh_lead.house
						house = scope:lam_fh_lead.house
					}
				}
			}
		}
		add = 35
	}
	# Married or betrothed into their house.
	if = {
		limit = {
			exists = scope:lam_fh_lead.house
			OR = {
				any_spouse = { exists = house house = scope:lam_fh_lead.house }
				betrothed ?= { exists = house house = scope:lam_fh_lead.house }
			}
		}
		add = 20
	}
	if = {
		limit = {
			OR = {
				has_relation_friend = scope:lam_fh_lead
				has_relation_best_friend = scope:lam_fh_lead
			}
		}
		add = 25
	}
	if = {
		limit = {
			OR = {
				has_relation_rival = scope:lam_fh_lead
				has_relation_nemesis = scope:lam_fh_lead
			}
		}
		add = -50
	}
	if = {
		limit = { faith = scope:lam_fh_lead.faith }
		add = 10
	}
	else = { add = -10 }
	if = {
		limit = { culture = scope:lam_fh_lead.culture }
		add = 5
		if = {
			limit = { scope:lam_fh_lead.var:lam_fac_agenda ?= flag:minority }
			add = 15
		}
	}
	if = {
		limit = { age >= lam_tune_heir_adult_age }
		add = 10
	}
	else_if = {
		limit = { age < lam_tune_heir_child_age }
		add = -10
	}
	if = {
		limit = { has_trait = incapable }
		add = -40
	}
	if = {
		limit = { has_trait = ill }
		add = -10
	}
	# What kind of ruler they would make, by the faction's own lights.
	if = {
		limit = {
			OR = {
				scope:lam_fh_lead.var:lam_fac_agenda ?= flag:glory_hound
				scope:lam_fh_lead.var:lam_fac_agenda ?= flag:belligerent
			}
		}
		add = {
			value = martial
			multiply = 1.5
		}
		if = { limit = { has_trait = brave } add = 10 }
		if = { limit = { has_trait = craven } add = -15 }
	}
	else_if = {
		limit = { scope:lam_fh_lead.var:lam_fac_agenda ?= flag:zealot }
		add = learning
		if = { limit = { has_trait = zealous } add = 15 }
		if = { limit = { has_trait = cynical } add = -15 }
	}
	else_if = {
		limit = { scope:lam_fh_lead.var:lam_fac_agenda ?= flag:courtly }
		add = {
			value = diplomacy
			multiply = 1.5
		}
	}
	else_if = {
		limit = {
			OR = {
				scope:lam_fh_lead.var:lam_fac_agenda ?= flag:parochial
				scope:lam_fh_lead.var:lam_fac_agenda ?= flag:barons_and_minor_landholders
			}
		}
		if = { limit = { has_trait = just } add = 10 }
		if = { limit = { has_trait = content } add = 5 }
		if = { limit = { has_trait = arbitrary } add = -10 }
	}
}

	if = {

	}

		add = 35
		if = { limit = { OR = { var:lam_fac_agenda ?= flag:parochial var:lam_fac_agenda ?= flag:barons_and_minor_landholders } } add = 15 }
	}
	else_if = {
		limit = { scope:lam_deal_lord = { var:lam_deal_give ?= flag:gold } }
		add = 20
		if = { limit = { has_trait = greedy } add = 15 }
	}
	else_if = {
		limit = { scope:lam_deal_lord = { var:lam_deal_give ?= flag:marriage } }
		add = 20
		if = { limit = { var:lam_fac_agenda ?= flag:dynastic } add = 10 }
	}
	else_if = {
		limit = { scope:lam_deal_lord = { var:lam_deal_give ?= flag:law } }
		add = 30
		if = { limit = { OR = { var:lam_fac_agenda ?= flag:parochial var:lam_fac_agenda ?= flag:barons_and_minor_landholders } } add = 20 }
		if = { limit = { var:lam_fac_agenda ?= flag:loyalist } add = -20 }
	}
	else_if = {
		limit = { scope:lam_deal_lord = { var:lam_deal_give ?= flag:heir } }
		add = 40
	}
	else_if = {
		limit = { scope:lam_deal_lord = { var:lam_deal_give ?= flag:amnesty } }
		add = 50
	}
	else_if = {
		limit = { scope:lam_deal_lord = { var:lam_deal_give ?= flag:bribe } }
		add = 20
		if = { limit = { has_trait = greedy } add = 25 }
		if = { limit = { OR = { has_trait = honest has_trait = just } } add = -30 }
	}
	# What I am asked.
	if = {
		limit = { scope:lam_deal_lord = { var:lam_deal_ask ?= flag:oath } }
		add = -25
		if = { limit = { has_variable = lam_fac_att var:lam_fac_att <= -15 } add = -25 }

	}
	else_if = {
		limit = { scope:lam_deal_lord = { var:lam_deal_ask ?= flag:heir } }
		add = -30
		if = { limit = { lam_fac_heir_mismatch_trigger = { LORD = scope:lam_deal_lord } } add = -15 }
	}
	else_if = {
		limit = { scope:lam_deal_lord = { var:lam_deal_ask ?= flag:disband } }
		add = -60
		if = { limit = { has_variable = lam_fac_age_tier } add = { value = var:lam_fac_age_tier multiply = -20 } }
	}
	else_if = {
		limit = { scope:lam_deal_lord = { var:lam_deal_ask ?= flag:counter } }
		add = -20
	}
	else_if = {
		limit = { scope:lam_deal_lord = { var:lam_deal_ask ?= flag:arms } }
		add = -25
		if = { limit = { OR = { var:lam_fac_agenda ?= flag:glory_hound var:lam_fac_agenda ?= flag:belligerent } } add = 15 }
	}
	else_if = {
		limit = { scope:lam_deal_lord = { var:lam_deal_ask ?= flag:law_support } }
		add = -30
		if = { limit = { OR = { var:lam_fac_agenda ?= flag:parochial var:lam_fac_agenda ?= flag:barons_and_minor_landholders } } add = -20 }
		if = { limit = { var:lam_fac_agenda ?= flag:loyalist } add = 20 }
	}
	# Standing between us.
	if = {
		limit = { has_variable = lam_fac_favour }
		add = { value = var:lam_fac_favour multiply = 0.3 }
	}
	add = {
		value = "opinion(scope:lam_deal_lord)"
		multiply = 0.2
	}
	if = { limit = { has_variable = lam_fac_betrayed } add = -20 }
	min = 5
	max = 95
}
