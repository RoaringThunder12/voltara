# Vanilla multiplicative (i.e. Scaled by councillor skill) modifiers applied to a conversion from converted rite to converter rite
# Scope : converting rite
# kCAFG_converted_rite : rite
# Optional Scopes :
# kCAFG_converting_culture
# kCAFG_converted_culture
# kCAFG_converted_province
V_kCAFG_county_setup_rite_conversion_power_mult = {
    value = 1
    if = {
		limit = {
			rite_has_parameter = pluralism_fundamentalist_proselytization_bonus
		}
        add = V_kCAFG_rite_conversion_pluralism_fundamentalist_proselytization_bonus
    }

    if = {
		limit = {
		    rite_has_parameter = pluralism_pluralistic_proselytization_penalty
		}
        add = V_kCAFG_rite_conversion_pluralism_pluralistic_proselytization_malus
    }

    if = {
		limit = {
			scope:kCAFG_converted_rite = { rite_has_parameter = pluralism_fundamentalist_proselytization_bonus }
        }
        subtract = V_kCAFG_rite_conversion_pluralism_fundamentalist_proselytization_bonus
    }

    if = {
		limit = {
			scope:kCAFG_converted_rite = { rite_has_parameter = pluralism_pluralistic_proselytization_penalty }
        }
        subtract = V_kCAFG_rite_conversion_pluralism_pluralistic_proselytization_malus
    }

    if = {
        limit = {
			exists = scope:kCAFG_converted_culture
            exists = scope:kCAFG_converting_culture
        }

        if = {
            limit = {
                rite_has_parameter = same_culture_conversion_bonus_active
                scope:kCAFG_converted_culture = scope:kCAFG_converting_culture
            }
            add = {
                value = V_kCAFG_rite_conversion_tenet_communal_identity_bonus
                multiply = 1.5 # Boost a bit
            }
        }

        if = {
            limit = {
				rite_has_parameter = other_culture_conversion_penalty_active
                NOT = { scope:kCAFG_converted_culture = scope:kCAFG_converting_culture }
            }
            add = {
                value = V_kCAFG_rite_conversion_tenet_communal_identity_malus
                multiply = 1.5 # Boost a bit
            }
        }
    }
}

# Multiplier applied to a conversion from converted rite to converter rite
# Scope : converting rite
# kCAFG_converted_rite : rite
V_kCAFG_county_setup_rite_conversion_mult = {
    value = 1
    
    if = {
        limit = { this.religion = scope:kCAFG_converted_rite.religion }
        add = 0.45
    }

    if = {
        limit = {
            scope:kCAFG_converted_rite = {
                OR = {
                    rite_has_doctrine = doctrine_pluralism_pluralistic
                    rite_hostility_level = {
                        target = prev
                        value = faith_fully_accepted_level
                    }
                }
            }
        }
        add = 0.25
    }

    if = {
        limit = {
            faith = { has_doctrine_parameter = unreformed }
            scope:kCAFG_converted_rite.faith = { NOT = { has_doctrine_parameter = unreformed } }
        }
        multiply = 0.7
    }

    if = {
        limit = {
            scope:kCAFG_converted_rite = {
                rite_hostility_level = {
                    target = prev
                    value >= faith_astray_level
                }
            }
        }
        multiply = 0.9
    }
    else_if = {
        limit = {
            scope:kCAFG_converted_rite = {
                rite_hostility_level = {
                    target = prev
                    value >= faith_hostile_level
                }
            }
        }
        multiply = 0.55
    }
    else_if = {
        limit = {
            scope:kCAFG_converted_rite = {
                rite_hostility_level = {
                    target = prev
                    value >= faith_evil_level
                }
            }
        }
        multiply = 0.01
    }
}

# Scope : rite
# Required scopes :
# kCAFG_converted_province : the province we want to convert the rite in
# kCAFG_converting_rite : the rite we want to convert to
V_kCAFG_province_rite_conversion_weight = {
    if = {
        limit = { scope:kCAFG_converting_rite = this }
        value = 0 # Cannot convert to itself
    }
    else = {
        # Scale based on level
        save_temporary_scope_as = kCAFG_rite # used by V_kCAFG_province_rite_factor
        value = scope:kCAFG_converted_province.V_kCAFG_province_rite_factor
		save_temporary_value_as = converted_rite_factor
        multiply = 10000 # Maybe I'm imagining things but weight might work better with higher numbers
        # Rescale by excluding converting rite
        divide = {
            value = 1
            scope:kCAFG_converting_rite = {
                scope:kCAFG_converted_province = {
                    subtract = V_kCAFG_province_prev_rite_factor
                }
            }
            min = V_kCAFG_smallest_minority_factor
        }
        # Kei : maybe have a "resistance" or "health" factor for certain rites in certain provinces or rite types/tenets/doctrines
        # The smaller, the harder to convert if there are other rites with higher levels
        if = {
            limit = {
                scope:kCAFG_converted_province = {
                    scope:converted_rite_factor <= 0.5
                    any_in_list = {
                        variable = kCAFG_province_rite_structs
                        scope:kCAFG_converting_rite != var:rite
                        var:factor > scope:converted_rite_factor
                    }
                }
            }
            multiply = {
                value = 0.1
                add = {
                    value = scope:converted_rite_factor
                    divide = 0.5
                    multiply = 0.9
                }
            }
        }
        # If very small minority, improbable to convert if there are other rites with higher levels
        if = {
            limit = {
                scope:kCAFG_converted_province = {
                    scope:converted_rite_factor <= V_kCAFG_very_small_minority_factor
                    any_in_list = {
                        variable = kCAFG_province_rite_structs
                        scope:kCAFG_converting_rite != var:rite
                        var:factor > V_kCAFG_very_small_minority_factor
                    }
                }
            }
            multiply = 0.01 # Already reduced by the "small minority" multiplier so we don't ovedo it, the conversion might still happen 
        }
    }
}

# The less a rite is present in the world, the harder it is to convert (preserve small minority rites)
# kCAFG_converting_rite : rite
V_kCAFG_rite_world_presence_conversion_factor = {
    value = num_rite_county_followers
    subtract = V_kCAFG_world_counties_very_small_amount
    divide = {
        value = V_kCAFG_world_counties_medium_amount
        subtract = V_kCAFG_world_counties_very_small_amount
    }
    max = 1

    # heresy mitigation
    if = {
        limit = {
            religion = scope:kCAFG_converting_rite.religion
            "divergence(scope:kCAFG_converting_rite)" >= scope:kCAFG_converting_rite.faith.heresy_threshold
        }
        multiply = 2
        add = 1
        divide = 3
    }
    # unreformed mitigation
    if = {
        limit = {
            faith = { has_doctrine_parameter = unreformed }
        }
        add = 1
        divide = 2
    }

    multiply = 0.9
    add = 0.1
}

# Can be negative
# Scope : province
# kCAFG_converting_rite : rite
V_kCAFG_province_natural_rite_conversion_weight = {
    ## Save factor
    value = V_kCAFG_province_rite_factor
    save_temporary_value_as = rite_factor
    ## ----

    value = V_kCAFG_province_natural_rite_conversion_base
    multiply = 0.5

    # Same rite neighbours
    if = {
        limit = { any_neighboring_province = { always = yes } }
        add = {
            value = 0
            every_neighboring_province = {
                every_in_list = {
                    variable = kCAFG_province_rite_structs
                    limit = { scope:kCAFG_converting_rite != var:rite }
                    add = var:factor
                }
            }
            divide = {
                value = 0
                every_neighboring_province = { add = 1 }
            }
            multiply = V_kCAFG_province_natural_rite_conversion_base
            multiply = 0.5
        }
    }

    # Tension reduces conversion weight
    subtract = {
        value = var:kCAFG_rite_tension
        subtract = 0.1
        min = 0
        divide = 0.9
        divide = V_kCAFG_province_natural_rite_conversion_max_tension
        multiply = V_kCAFG_province_natural_rite_conversion_tension_impact
    }
    
    # # Small Minorities are harder to convert
    if = {
        limit = {
            scope:rite_factor >= V_kCAFG_inverted_small_minority_factor
        }
        add = {
            value = 1
            # Especially if they have low world presence
            subtract = {
                value = 0
                every_in_list = {
                    variable = kCAFG_province_rite_structs
                    limit = { scope:kCAFG_converting_rite != var:rite }

                    add = {
                        value = var:rite.V_kCAFG_rite_world_presence_conversion_factor
                        multiply = var:factor # Scale by converted rite factor
                    }
                }
                # rescale by excluding converting rite
                divide = {
                    value = 1
                    subtract = scope:rite_factor
                }
            }

            multiply = {
                value = scope:rite_factor
                subtract = V_kCAFG_inverted_small_minority_factor
                min = 0
                divide = {
                    value = V_kCAFG_small_minority_factor
                    subtract = V_kCAFG_very_small_minority_factor
                }

                # Reduce impact
                multiply = 0.5
                add = 0.5
            }

            multiply = V_kCAFG_province_natural_rite_conversion_low_adherents_impact
            multiply = -1
        }
    }
    
    multiply = V_kCAFG_province_rite_conversion_multiplier

    # The closer to zero we are, the more likely the province will be converted
    # 100% multiplier at 0% rite presence
    # 50% multiplier at 50% rite presence
    # 15% multiplier at 100% rite presence
    if = {
        limit = { scope:rite_factor <= 0.5 }
        multiply = {
            value = 1
            subtract = scope:rite_factor
        }
    }
    if = {
        limit = { scope:rite_factor > 0.5 }
        multiply = {
            value = 0.5
            subtract = {
                value = scope:rite_factor # 0.5-1
                subtract = 0.5 # 0-0.5
                multiply = 2 # 0-1
                multiply = 0.35 # 0-0.35
            }
        }
    }
    # unwelcoming holder
    if = {
        limit = { county.holder.rite != scope:kCAFG_converting_rite }
        divide = 2
    }
    # holder is hof
    if = {
        limit = {
            exists = scope:kCAFG_converting_rite.religious_head
            county.holder = scope:kCAFG_converting_rite.religious_head
        }
        multiply = 1.5
    }

    # tolerance laws
    multiply = V_kCAFG_rite_tension_tolerance_law_factor
}
