# Living Factions - secrets.
# Under an edict against factions, the leaders and core of a banned faction are outlaws. Their liege's
# spies can find out; once the liege knows, it is a crime - grounds to imprison, or a hook to hold.
lam_secret_outlaw_faction = {
	category = civil

	is_valid = {
		scope:secret_owner = {
			is_alive = yes
			has_variable = lam_fac_outlaw
			liege ?= { has_variable = lam_fac_ban }
		}
	}

	is_shunned = { always = no }

	is_criminal = {
		scope:secret_owner.liege ?= { has_variable = lam_fac_ban }
	}

	on_discover = {
		save_scope_as = secret
		if = {
			limit = { scope:discoverer = { this = scope:secret_owner.liege } }
			scope:discoverer = {
				send_interface_message = {
					type = event_generic_neutral
					title = lam_secret_outlaw_found_title
					right_icon = scope:secret_owner
				}
			}
		}
	}

	on_expose = {
		save_scope_as = secret
		scope:secret_owner = {
			set_variable = { name = lam_fac_exposed value = yes years = 2 }
			add_prestige = medium_prestige_loss
		}
		secret_exposed_notification_effect = yes
	}
}
# The head of a faction took the lord's money for themselves. If it comes out, the faction turns on them.
lam_secret_bribed = {
	category = civil

	is_valid = {
		scope:secret_owner = {
			is_alive = yes
			has_variable = lam_bloc_lead
		}
	}

	is_shunned = { always = yes }
	is_criminal = { always = no }

	on_discover = {
		save_scope_as = secret
	}

	on_expose = {
		save_scope_as = secret
		scope:secret_owner = {
			lam_fac_cohesion_add_effect = { ADD = -40 }
			set_variable = { name = lam_fac_disgraced value = yes years = 1 }
			every_in_list = {
				variable = lam_fac_members
				limit = { is_alive = yes }
				add_opinion = { target = scope:secret_owner modifier = lam_fac_hoarded_opinion }
			}
		}
		secret_exposed_notification_effect = yes
	}
}
# Informed on their own faction to the lord. If it comes out, the faction throws them out.
lam_secret_informer = {
	category = civil

	is_valid = {
		scope:secret_owner = { is_alive = yes }
	}

	is_shunned = { always = yes }
	is_criminal = { always = no }

	on_discover = {
		save_scope_as = secret
	}

	on_expose = {
		save_scope_as = secret
		scope:secret_owner = {
			var:lam_bloc ?= {
				add_opinion = { target = scope:secret_owner modifier = lam_fac_betrayed_opinion }
				every_in_list = {
					variable = lam_fac_members
					limit = { is_alive = yes var:lam_fac_tier ?= 1 }
					add_opinion = { target = scope:secret_owner modifier = lam_fac_betrayed_opinion }
				}
			}
			if = { limit = { has_variable = lam_bloc } remove_variable = lam_bloc }
			if = { limit = { has_variable = lam_fac_tier } remove_variable = lam_fac_tier }
			if = { limit = { has_variable = lam_fac_pinned } remove_variable = lam_fac_pinned }
			set_variable = { name = lam_fac_unbound value = yes years = 3 }
		}
		secret_exposed_notification_effect = yes
	}
}
