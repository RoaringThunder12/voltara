 }

                primary_title = { exists = var:kingsguard_2 }

                primary_title = { exists = var:kingsguard_4 }
                primary_title = { exists = var:kingsguard_5 }

       }

}

mde_open_white_book = {
    scope = character
    effect = {
        if = {
            limit = {
                NOT = { has_variable = selected_white_book_page }
                OR = {
                    AND = {
                        has_title = title:h_voltara_severos
                        title:h_voltara_severos = { title_has_kingsguard_trigger = yes }
                    }
                    NOT = { has_multiple_players = yes }
                }
            }
            trigger_event = agot_kingsguard_white_book.0001
        }
    }
}
