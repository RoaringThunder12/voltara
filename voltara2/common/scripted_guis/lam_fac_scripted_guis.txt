lam_sgui_fac_favour = {
	scope = character
	saved_scopes = {
		target
	}
	is_valid = {
		scope:target = {
			liege ?= root
			has_variable = lam_fac_favour
			NOT = {
				has_variable = lam_fac_favour_cd
			}
			is_alive = yes
			has_variable = lam_bloc_lead
			NOT = {
				has_variable = lam_fac_disband
			}
		}
		prestige >= minor_prestige_value
	}
	effect = {
		if = {
			limit = {
				scope:target = {
					liege ?= root
					has_variable = lam_fac_favour
					NOT = {
						has_variable = lam_fac_favour_cd
					}
					is_alive = yes
					has_variable = lam_bloc_lead
					NOT = {
						has_variable = lam_fac_disband
					}
				}
				prestige >= minor_prestige_value
			}
			add_prestige = {
				value = minor_prestige_value
				multiply = -1
			}
			lam_fac_favour_effect = {
				ADD = 15
			}
			lam_is_favour_effect = {
				DIR = 1
				KIND = favour
			}
			scope:target = {
				add_opinion = {
					target = root
					modifier = lam_fac_granted_opinion
				}
			}
		}
	}
}

lam_sgui_fac_join = {
	scope = character
	saved_scopes = {
		target
	}
	is_shown = {
		is_independent_ruler = no
		NOT = {
			has_variable = lam_bloc_lead
		}
		scope:target = {
			is_alive = yes
			has_variable = lam_bloc_lead
			liege ?= root.liege
			NOT = {
				this = root
			}
		}
		NOT = {
			var:lam_bloc ?= scope:target
		}
	}
	is_valid = {
		NOT = {
			has_variable = lam_fac_left_cd
		}
		scope:target = {
			opinion = {
				target = root
				value >= 0
			}
		}
	}
	effect = {
		set_variable = {
			name = lam_bloc
			value = scope:target
		}
		set_variable = {
			name = lam_fac_pinned
			value = yes
		}
		scope:target = {
			add_opinion = {
				target = root
				modifier = lam_fac_granted_opinion
			}
		}
	}
}
lam_sgui_fac_leave = {
	scope = character
	saved_scopes = {
		target
	}
	is_shown = {
		var:lam_bloc ?= scope:target
		NOT = {
			has_variable = lam_bloc_lead
		}
	}
	effect = {
		scope:target = {
			add_opinion = {
				target = root
				modifier = lam_fac_left_opinion
			}
		}
		remove_variable = lam_bloc
		if = {
			limit = {
				has_variable = lam_fac_pinned
			}
			remove_variable = lam_fac_pinned
		}
		set_variable = {
			name = lam_fac_left_cd
			value = yes
			years = 1
		}
	}
}
