		has_variable = lam_bloc_lead
		var:lam_bloc ?= this

			is_target_in_variable_list = {
				name = lam_bloc_leads
				target = root
			}
		}
		NOT = {
			has_variable = lam_cr_active
		}

	}
	is_valid = {
		custom_description = {
			text = lam_pt_head_available_tt
			lam_pt_head_ready_trigger = yes
		}
		custom_description = {
			text = lam_cv_committed_tt
			NOT = {
				lam_wc_committed_trigger = yes
			}
		}
		custom_description = {
			text = lam_cv_call_retry_tt
			NOT = {
				has_variable = lam_cr_retry_goal
			}
		}
		custom_description = {
			text = lam_cv_no_seat_tt
			save_temporary_scope_as = vip_seat_head
			any_in_list = {
				variable = lam_fac_members
				is_alive = yes
				is_adult = yes
				is_imprisoned = no
				var:lam_bloc ?= scope:vip_seat_head
				NOT = {
					this = scope:vip_seat_head
				}
				NOT = {
					scope:vip_seat_head = {
						var:lam_fac_seat_pick ?= prev
					}
				}
				liege ?= scope:vip_seat_head.liege
				NOT = {
					is_councillor_of = scope:vip_seat_head.liege
				}
				OR = {
					can_be_marshal_trigger = {
						COURT_OWNER = scope:vip_seat_head.liege
					}
					can_be_steward_trigger = {
						COURT_OWNER = scope:vip_seat_head.liege
					}
					can_be_spymaster_trigger = {
						COURT_OWNER = scope:vip_seat_head.liege
					}
					can_be_chancellor_trigger = {
						COURT_OWNER = scope:vip_seat_head.liege
					}
				}
			}
		}
	}
	effect = {
		remove_variable = lam_cr_view

	}
}
