republican_support_candidate = {
	scope = character
	is_valid = {
		republican_election_ongoing = yes
		OR = {
			NOT = { scope:candidate = { has_variable = election_boost } }
			scope:candidate.var:election_boost < 5
		}
		gold > support_candidate_gold_cost
	}
	effect = {
		remove_short_term_gold = support_candidate_gold_cost
		scope:candidate = { republican_government_boost_election_weight_effect = yes }
	}
}

republican_play_as_same_family = {
	scope = character
	is_valid = {
		is_republican_government_story_owner = yes
	}
	effect = {
		var:republican_government_story = {
			if = {
				limit = { has_variable = play_as_same_family }
				remove_variable = play_as_same_family	
			}
			else = {
				set_variable = {
					name = play_as_same_family
					value = yes
				}
			}
		}
	}
}
