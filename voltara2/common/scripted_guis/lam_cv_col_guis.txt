# Living Factions - the faction council chamber columns (tools/claude_fixes/council_ui.py)
lam_cv_col_yes = {
	scope = character
	saved_scopes = {
		target
	}
	is_shown = {
		scope:target = {
			var:lam_cr_choice ?= flag:yes
		}
	}
}
lam_cv_lcol_yes = {
	scope = character
	saved_scopes = {
		target
	}
	is_shown = {
		scope:target = {
			var:lam_cv_ch ?= flag:yes
		}
	}
}
lam_cv_fold_yes = {
	scope = character
	effect = {
		if = {
			limit = {
				has_variable = lam_cv_fold_yes
			}
			remove_variable = lam_cv_fold_yes
		}
		else = {
			set_variable = {
				name = lam_cv_fold_yes
				value = yes
			}
		}
	}
}
