# Living Factions - court issues (tools/claude_fixes/issue.py). Push buttons on the issue cards.
lam_is_push_heir_x = {
	scope = character
	saved_scopes = {
		target
	}
	is_shown = {
		lam_is_can_push_trigger = yes
		scope:target = {
			has_variable = lam_is_heir_open
		}
	}
	is_valid = {
		lam_is_can_push_trigger = yes
		lam_is_push_ready_trigger = yes
		scope:target = {
			has_variable = lam_is_heir_open
		}

		custom_description = {
			text = lam_rv_is_held_tt
			scope:target = {
				save_temporary_scope_as = lam_is_lord
				trigger_if = {
					limit = {
						exists = var:lam_is_heir_a
					}
					var:lam_is_heir_a = {
						save_temporary_scope_as = lam_is_ca
					}
				}
				trigger_if = {
					limit = {
						exists = var:lam_is_heir_b
					}
					var:lam_is_heir_b = {
						save_temporary_scope_as = lam_is_cb
					}
				}
				NOT = {
					lam_is_held_trigger = {
						T = heir
					}
				}
			}
		}
	}
	effect = {
		lam_is_player_push_effect = {
			T = heir
			DIR = -1
			FOR = x
			AGAINST = y
		}
	}
}
