necromancy_novice_perk = {
	lifestyle = dark_arts_lifestyle
	tree = necromancy
	position = { 0 0 }
	icon = node_necromancy

	can_be_picked = { 
		vt_ek_is_necromancer = yes
	}
	
	auto_selection_weight = { value = 10 if = { limit = { can_magic_trigger = no } multiply = 0 } }
	
	effect = {	
		if = {
			limit = { vt_ek_is_necromancer = yes }
			custom_description_no_bullet = {
				text = necromancy_novice_perk_effect
			}
		}
		else = {
			custom_description_no_bullet = {
				text = no_necromancy_trait_perk_effect
			}
		}
    }
}

necromancy_master_perk = {
	lifestyle = dark_arts_lifestyle
	tree = necromancy
	position = { 0 4 }
	icon = school_necromancy
	
	parent = necromancy_expert_perk
	parent = necromancy_major_rituals_perk
	
	effect = {
		if = {
			limit = { vt_ek_is_necromancer = yes }	
			custom_description_no_bullet = {
				text = necromancy_master_perk_effect
			}
			custom_description_no_bullet = {
				text = necromancy_master_perk_effect_lich
			}
		}
		else = {
			custom_description_no_bullet = {
				text = no_necromancy_trait_perk_effect
			}
		}
		custom_description_no_bullet = {
			text = restoration_master_perk_effect_lifespan
		}
		if = {
			limit = {
				NOT = { exists = var:mastered_schools }
			}
			set_variable = {
				name = mastered_schools
				value = 1
			}
		}
		else = {
			change_variable = {
				name = mastered_schools
				add = 1
			}
		}
		if = {
            limit = { culture = { has_cultural_tradition = tradition_mystical_ancestors } } ###EK EDIT: new parameter
            add_dark_arts_lifestyle_xp = 330
        }
		if = {
			limit = { can_increase_lifespan_trigger = yes }
			ek_can_extend_lifespan_notification_effect = yes
		}
	}
}
