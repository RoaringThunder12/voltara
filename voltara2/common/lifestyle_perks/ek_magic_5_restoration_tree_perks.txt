restoration_master_perk = {
	lifestyle = vt_threadbinding_magic_lifestyle
	tree = restoration
	position = { 0 4 }
	icon = school_restoration
	
	parent = restoration_expert_perk
	parent = restoration_major_rituals_perk
	
	effect = {	
		custom_description_no_bullet = {
			text = restoration_master_perk_effect
		}
		custom_description_no_bullet = {
			text = restoration_master_perk_effect_lifespan
		}
		if = {
			limit = {
				NOT = { exists = var:mastered_schools }
			}
			set_variable = {
				name = mastered_schools
				value = 1
			}
		}
		else = {
			change_variable = {
				name = mastered_schools
				add = 1
			}
		}
		custom_description_no_bullet = {
			text = restoration_master_perk_effect_cloning
		}
        if = {
            limit = { culture = { has_cultural_tradition = tradition_mystical_ancestors } } ###EK EDIT: new parameter
            add_vt_threadbinding_magic_lifestyle_xp = 500
        }
		if = {
			limit = { can_increase_lifespan_trigger = yes }
			ek_can_extend_lifespan_notification_effect = yes
		}
    }
}
