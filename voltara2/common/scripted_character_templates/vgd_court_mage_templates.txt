court_mage_template = {
	age = { 20 40 }
	faith = ROOT.faith
	culture = root.culture
	dynasty = none
	gender_female_chance = 50
	random_traits = yes

	after_creation = {
		random_list = {
			50 = {}
			20 = { add_trait = vt_ek_arcana_good_1 }
			10 = { add_trait = vt_ek_arcana_good_2 }
			10 = { add_trait = vt_ek_arcana_good_3 }
			10 = { add_trait = experimentalist }
			5 = { vgd_recruited_mage_necromancy_effect = yes }
		}		
		add_arcana_skill = { VALUE = "{ 8 10 }" }
		# Culture and faith are inherited from the recruiting ruler.
		if = { # In case they arent magic yet, make them magic
			limit = { can_magic_trigger = no }
			ek_magic_perks_effect = yes
		}
	}
}
