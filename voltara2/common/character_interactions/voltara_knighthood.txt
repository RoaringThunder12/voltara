 }

  scope:recipient = { add_trait = squire add_trait_xp = { trait = squire track = knight value = 100 } remove_variable = voltara_knight_mentor }
  agot_add_to_knight_tree = { KNIGHT = scope:actor SQUIRE = scope:recipient }
 }
 ai_targets = { ai_recipients = courtiers } ai_frequency = 12
 ai_will_do = { base = 10 }
}

voltara_assign_squire = {
 category = interaction_category_friendly common_interaction = yes icon = knight
 desc = voltara_assign_squire_desc
 is_shown = { scope:actor = scope:recipient scope:actor = { is_landed = yes NOT = { voltara_is_ruins_placeholder = yes } } }
 populate_actor_list = { scope:actor = {
 if = { limit = { voltara_realm_knight_mentor = yes } add_to_list = characters }
 every_courtier = { limit = { voltara_realm_knight_mentor = yes } add_to_list = characters }
 every_vassal_or_below = {
  if = { limit = { voltara_realm_knight_mentor = yes } add_to_list = characters }
  every_courtier = { limit = { voltara_realm_knight_mentor = yes } add_to_list = characters }
 }
} }
 populate_recipient_list = { scope:actor = {
 if = { limit = { voltara_realm_squire_candidate = yes } add_to_list = characters }
 every_courtier = { limit = { voltara_realm_squire_candidate = yes } add_to_list = characters }
 every_vassal_or_below = {
  if = { limit = { voltara_realm_squire_candidate = yes } add_to_list = characters }
  every_courtier = { limit = { voltara_realm_squire_candidate = yes } add_to_list = characters }
 }
} }
 is_valid_showing_failures_only = { scope:actor = { is_alive = yes is_imprisoned = no } }
 can_send = {
  scope:actor = scope:recipient
  exists = scope:secondary_actor exists = scope:secondary_recipient
  scope:secondary_actor = { voltara_realm_knight_mentor = yes }
  scope:secondary_recipient = { voltara_realm_squire_candidate = yes }
 }
 auto_accept = yes
 on_accept = {
  if = { limit = { scope:actor = scope:recipient scope:secondary_actor = { voltara_realm_knight_mentor = yes } scope:secondary_recipient = { voltara_realm_squire_candidate = yes } }
   scope:secondary_recipient = { add_trait = squire set_variable = { name = voltara_knight_mentor value = scope:secondary_actor } }
  }
 }
 ai_will_do = { base = 0 }
}
