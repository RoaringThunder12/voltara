#Establish trade deal
trade_deal_interaction = {
	icon = icon_scheme_befriend
	category = interaction_category_friendly
	common_interaction = yes

	send_name = START_SCHEME

	interface_priority = 120
	scheme = trade_deal_scheme
	ignores_pending_interaction_block = yes

	cooldown = { years = 2 } 
	is_shown = {
		scope:actor = { can_use_trade = yes }
		NOT = { scope:actor = scope:recipient }
		scope:recipient = {
			NOR = {
				is_independent_ruler = no
				can_use_trade = yes
			}
		}
	}

	is_valid = {
		scope:actor = {
			can_start_scheme = {
				type = trade_deal_scheme
				target_character = scope:recipient
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			can_start_scheme = {
				type = trade_deal_scheme
				target_character = scope:recipient
			}
		}
	}

	desc = scheme_interaction_tt_trade_deal_scheme_approved

	on_accept = {
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = trade_deal_interaction_notification

				left_icon = scope:actor					
				right_icon = scope:recipient

				begin_scheme_basic_effect = {
					SCHEME_TYPE = trade_deal_scheme
					TARGET_TYPE = target_character
					TARGET_SCOPE = scope:recipient
				}
			}
		}
	}
	
	auto_accept = yes
}
