# Living Factions (split from Love & Marriage).

# Look into a realm's living factions (independent rulers; one's own court and liege's are in the window itself).
lam_view_factions_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	icon = lam_ic_terms
	desc = lam_view_factions_interaction_desc
	auto_accept = yes
	ignores_pending_interaction_block = yes

	is_shown = {
		scope:actor = { is_ai = no }
		scope:recipient = {
			is_ruler = yes
			any_vassal = { lam_fac_peer_trigger = yes }
		}
	}

	on_accept = {
		scope:actor = {
			set_variable = { name = lam_fac_view value = scope:recipient }
			scope:recipient = { lam_gn_book_ensure_effect = yes }
			lam_gn_view_main_effect = yes
			lam_factions_refresh_effect = yes
			set_variable = { name = lam_factions_open value = yes }
		}
	}

	ai_will_do = {
		base = 0
	}
}

# The window of the faction this character leads or belongs to.
lam_view_their_faction_interaction = {
	category = interaction_category_diplomacy
	common_interaction = yes
	icon = lam_ic_issue
	desc = lam_view_their_faction_interaction_desc
	auto_accept = yes
	ignores_pending_interaction_block = yes

	is_shown = {
		scope:actor = { is_ai = no }
		scope:recipient = {
			# (Only a faction seated in a court.)
			OR = {
				AND = { has_variable = lam_bloc_lead liege ?= { is_target_in_variable_list = { name = lam_bloc_leads target = prev } } }
				var:lam_bloc ?= { is_alive = yes has_variable = lam_bloc_lead liege ?= { is_target_in_variable_list = { name = lam_bloc_leads target = prev } } }
			}
			# (What the crown has not uncovered stays hidden.)
			NOT = { var:lam_bloc ?= { has_variable = lam_fac_vl has_variable = lam_fac_hide } }
		}
	}

	on_accept = {
		scope:actor = {
			if = {
				limit = { scope:recipient = { has_variable = lam_bloc_lead } }
				set_variable = { name = lam_fac_detail value = scope:recipient }
			}
			else = { set_variable = { name = lam_fac_detail value = scope:recipient.var:lam_bloc } }
		}
	}

	ai_will_do = {
		base = 0
	}
}

# Ask to join someone's faction (both vassals of the same lord).
lam_join_faction_interaction = {
	category = interaction_category_diplomacy
	icon = lam_ic_terms
	desc = lam_join_faction_interaction_desc

	is_shown = {
		scope:actor = {
			is_ai = no
			is_independent_ruler = no
			NOT = { has_variable = lam_bloc_lead }
		}
		scope:recipient = {
			has_variable = lam_bloc_lead
			liege ?= scope:actor.liege
			NOT = { this = scope:actor }
		}
		NOT = { scope:actor.var:lam_bloc ?= scope:recipient }
	}
	is_valid_showing_failures_only = {
		scope:actor = { NOT = { has_variable = lam_fac_left_cd } }
	}

	# The head weighs what the newcomer brings: strength above all, then a shared cause, ties and trust.
	ai_accept = {
		base = -10
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.6
		}
		modifier = {
			add = 40
			desc = lam_ai_fac_strong_reason
			scope:actor.max_military_strength >= scope:recipient.max_military_strength
		}
		modifier = {
			add = 20
			desc = lam_ai_fac_useful_reason
			scope:actor.max_military_strength < scope:recipient.max_military_strength
			scope:actor.max_military_strength >= { value = scope:recipient.max_military_strength multiply = 0.4 }
		}
		modifier = {
			add = -25
			desc = lam_ai_fac_weak_reason
			scope:actor.max_military_strength < { value = scope:recipient.max_military_strength multiply = 0.15 }
		}
		modifier = {
			add = 15
			desc = lam_ai_fac_cause_reason
			scope:recipient = { lam_fac_cause_of_trigger = { WHO = scope:actor } }
		}
		modifier = {
			add = 25
			desc = lam_ai_fac_kin_reason
			scope:actor.house ?= scope:recipient.house
		}
		modifier = {
			add = -50
			desc = lam_ai_fac_foe_reason
			scope:actor.var:lam_bloc ?= { is_target_in_variable_list = { name = lam_fac_foes target = scope:recipient } }
		}
		modifier = {
			add = -100
			desc = lam_ai_fac_rival_reason
			scope:recipient = { OR = { has_relation_rival = scope:actor has_relation_nemesis = scope:actor } }
		}
		# A faction set against the crown does not take in one of the crown's own council.
		modifier = {
			add = -30
			desc = lam_ai_fac_spy_reason
			scope:recipient = { OR = { var:lam_fac_stage ?= 3 var:lam_fac_stage ?= 4 has_variable = lam_fac_hide } }
			scope:actor = { is_councillor_of = scope:actor.liege }
		}
	}

	on_accept = {
		scope:actor = {
			set_variable = { name = lam_bloc value = scope:recipient }
			set_variable = { name = lam_fac_pinned value = yes }
			send_interface_message = { type = event_generic_good title = lam_fac_joined_title right_icon = scope:recipient }
		}
		hidden_effect = { scope:actor = { lam_fac_quick_refresh_court_effect = yes } }
	}
	ai_will_do = { base = 0 }
}

# Invite a peer into my faction.
lam_invite_faction_interaction = {
	category = interaction_category_diplomacy
	icon = lam_ic_terms
	desc = lam_invite_faction_interaction_desc

	is_shown = {
		scope:actor = {
			is_ai = no
			has_variable = lam_bloc_lead
		}
		scope:recipient = {
			is_ai = yes
			is_ruler = yes
			liege ?= scope:actor.liege
			NOT = { this = scope:actor.liege }
			# (Not one who outranks the head: a king does not follow a count.)
			highest_held_title_tier <= scope:actor.highest_held_title_tier
			NOT = { has_variable = lam_bloc_lead }
			NOT = { var:lam_bloc ?= scope:actor }
		}
	}

	# The invited weigh it: what the head and the faction amount to, what it stands for, and where they stand now.
	ai_accept = {
		base = -20
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
		}
		modifier = {
			add = 20
			desc = lam_ai_inv_stronger_reason
			scope:actor.max_military_strength >= { value = scope:recipient.max_military_strength multiply = 2 }
		}
		modifier = {
			add = -30
			desc = lam_ai_inv_weaker_reason
			scope:actor.max_military_strength < scope:recipient.max_military_strength
		}
		modifier = {
			add = 15
			desc = lam_ai_inv_standing_reason
			scope:actor = { has_variable = lam_fac_share var:lam_fac_share >= 20 }
		}
		modifier = {
			add = -15
			desc = lam_ai_inv_small_reason
			scope:actor = { OR = { NOT = { has_variable = lam_fac_count } var:lam_fac_count < 3 } }
		}
		modifier = {
			add = 15
			desc = lam_ai_inv_parted_reason
			scope:recipient = { var:lam_fac_parted ?= scope:actor }
		}
		modifier = {
			add = 20
			desc = lam_ai_inv_cause_reason
			scope:actor = { lam_fac_cause_of_trigger = { WHO = scope:recipient } }
		}
		# Loyal to the crown and asked into a faction set against it - or the other way round.
		modifier = {
			add = -40
			desc = lam_ai_inv_against_crown_reason
			scope:actor = { OR = { var:lam_fac_stage ?= 2 var:lam_fac_stage ?= 3 var:lam_fac_stage ?= 4 } }
			scope:recipient = { has_variable = lam_gui_att var:lam_gui_att >= 25 }
		}
		modifier = {
			add = -25
			desc = lam_ai_inv_for_crown_reason
			scope:actor = { OR = { var:lam_fac_stage ?= 5 var:lam_fac_stage ?= 6 var:lam_fac_stage ?= 8 } }
			scope:recipient = { has_variable = lam_gui_att var:lam_gui_att <= -15 }
		}
		modifier = {
			add = 10
			desc = lam_ai_inv_renown_reason
			scope:actor = { has_variable = lam_fac_renown var:lam_fac_renown >= 20 }
		}
		modifier = {
			add = -100
			desc = lam_ai_fac_rival_reason
			scope:recipient = { OR = { has_relation_rival = scope:actor has_relation_nemesis = scope:actor } }
		}
		modifier = {
			add = -15
			desc = lam_ai_inv_content_reason
			scope:recipient = { has_trait = content }
		}
		modifier = {
			add = 10
			desc = lam_ai_inv_ambitious_reason
			scope:recipient = { has_trait = ambitious }
		}
		modifier = {
			add = 30
			desc = lam_ai_fac_kin_reason
			scope:recipient.house ?= scope:actor.house
		}
		modifier = {
			add = -40
			desc = lam_ai_fac_core_reason
			scope:recipient = { var:lam_fac_tier ?= 1 }
		}
		modifier = {
			add = -15
			desc = lam_ai_fac_client_reason
			scope:recipient = { var:lam_fac_tier ?= 2 }
		}
	}

	on_accept = {
		scope:recipient = {
			set_variable = { name = lam_bloc value = scope:actor }
			set_variable = { name = lam_fac_pinned value = yes years = 2 }
		}
		scope:actor = { send_interface_message = { type = event_generic_good title = lam_fac_invited_title right_icon = scope:recipient } }
		hidden_effect = { scope:actor = { lam_fac_quick_refresh_court_effect = yes } }
	}
	ai_will_do = { base = 0 }
}

# Leave that faction - or else (a hook, or dread).
lam_coerce_leave_interaction = {
	category = interaction_category_hostile
	icon = lam_ic_issue
	desc = lam_coerce_leave_interaction_desc
	auto_accept = yes

	is_shown = {
		scope:actor = { is_ai = no }
		scope:recipient = {
			is_ai = yes
			has_variable = lam_bloc
			NOT = { has_variable = lam_bloc_lead }
			NOT = { var:lam_bloc ?= scope:actor }
		}
	}
	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = lam_coerce_need_leverage_tt
			OR = {
				scope:actor = { has_usable_hook = scope:recipient }
				scope:recipient = { has_dread_level_towards = { target = scope:actor level >= 1 } }
			}
		}
	}

	on_accept = {
		if = {
			limit = { scope:actor = { has_usable_hook = scope:recipient } }
			scope:actor = { use_hook = scope:recipient }
		}
		scope:recipient = {
			var:lam_bloc ?= { if = { limit = { is_alive = yes } add_opinion = { target = scope:actor modifier = lam_fac_left_opinion } } }
			remove_variable = lam_bloc
			if = { limit = { has_variable = lam_fac_pinned } remove_variable = lam_fac_pinned }
			set_variable = { name = lam_fac_unbound value = yes years = 2 }
			add_opinion = { target = scope:actor modifier = lam_fac_coerced_opinion }
		}
		scope:actor = { send_interface_message = { type = event_generic_neutral title = lam_coerce_leave_done_title right_icon = scope:recipient } }
		hidden_effect = { scope:recipient = { lam_fac_quick_refresh_court_effect = yes } }
	}
	ai_will_do = { base = 0 }
}

# Join us - or else. Into my own faction, or (as liege) into my court party.
lam_coerce_join_interaction = {
	category = interaction_category_hostile
	icon = lam_ic_issue
	desc = lam_coerce_join_interaction_desc
	auto_accept = yes

	is_shown = {
		scope:actor = { is_ai = no }
		scope:recipient = {
			is_ai = yes
			NOT = { has_variable = lam_bloc_lead }
		}
		OR = {
			AND = {
				scope:actor = { has_variable = lam_bloc_lead is_independent_ruler = no }
				scope:recipient.liege ?= scope:actor.liege
				NOT = { scope:recipient.var:lam_bloc ?= scope:actor }
			}
			AND = {
				scope:recipient.liege ?= scope:actor
				scope:actor = { any_in_list = { variable = lam_bloc_leads has_variable = lam_fac_courtparty } }
			}
		}
	}
	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = lam_coerce_need_leverage_tt
			OR = {
				scope:actor = { has_usable_hook = scope:recipient }
				scope:recipient = { has_dread_level_towards = { target = scope:actor level >= 1 } }
			}
		}
	}

	on_accept = {
		if = {
			limit = { scope:actor = { has_variable = lam_bloc_lead } }
			scope:actor = { save_scope_as = lam_cj_target }
		}
		else = {
			scope:actor = {
				random_in_list = {
					variable = lam_bloc_leads
					limit = { is_alive = yes has_variable = lam_fac_courtparty }
					save_scope_as = lam_cj_target
				}
			}
		}
		if = {
			limit = { scope:actor = { has_usable_hook = scope:recipient } }
			scope:actor = { use_hook = scope:recipient }
		}
		scope:recipient = {
			var:lam_bloc ?= { if = { limit = { is_alive = yes } add_opinion = { target = scope:actor modifier = lam_fac_left_opinion } } }
			set_variable = { name = lam_bloc value = scope:lam_cj_target }
			set_variable = { name = lam_fac_pinned value = yes years = 2 }
			set_variable = { name = lam_fac_coerced value = yes years = 2 }
			if = { limit = { has_variable = lam_fac_unbound } remove_variable = lam_fac_unbound }
			add_opinion = { target = scope:actor modifier = lam_fac_coerced_opinion }
		}
		scope:actor = { send_interface_message = { type = event_generic_neutral title = lam_coerce_join_done_title right_icon = scope:recipient } }
		hidden_effect = { scope:recipient = { lam_fac_quick_refresh_court_effect = yes } }
	}
	ai_will_do = { base = 0 }
}
