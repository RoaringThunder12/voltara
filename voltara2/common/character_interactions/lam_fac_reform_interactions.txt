    category = interaction_category_diplomacy
    icon = lam_ic_terms
    desc = lam_cr_ballot_interaction_desc
    auto_accept = yes
    is_shown = {
        scope:actor = { is_ai = no has_variable = lam_cr_ballot_head var:lam_cr_choice ?= flag:pending }
        scope:actor = scope:recipient
    }
    on_accept = {
        hidden_effect = {
            scope:actor = {
                var:lam_cr_ballot_head ?= { save_scope_as = lam_cr_head var:lam_cr_lord ?= { save_scope_as = lam_cr_lord } }
                if = { limit = { has_variable = lam_cr_ballot_id }
                    save_scope_value_as = { name = lam_cr_token value = var:lam_cr_ballot_id }

                        scope:lam_cr_head = { lam_cr_restore_context_effect = yes }
                        trigger_event = lam_cr.0001
                    }
                }
            }
        }
    }
    ai_will_do = { base = 0 }
}
