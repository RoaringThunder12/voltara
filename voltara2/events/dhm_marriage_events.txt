# A newly adult AI child asks their human matchmaker's household office to run a
# prompt bounded pass immediately instead of waiting for the next quarter.
# Re-root through the existing hidden manager event; nested effect scope alone
# would leave ROOT on this birthday child. Existing scan cooldowns still apply.
dhm_marriage.0001 = {
    type = character_event
    hidden = yes

    trigger = {
        is_ai = yes
        age = 16
        exists = matchmaker
        matchmaker = { is_ai = no }
    }

    immediate = {
        if = {
            limit = { matchmaker = { has_character_flag = dhm_v080_birthday_story_pending } }
            # The graduation/recovery scene lands first; the hidden scan follows
            # after its three-month breathing room instead of being discarded.
            matchmaker = { trigger_event = { id = dhm_core.0001 months = 4 } }
        }
        else = { matchmaker = { trigger_event = dhm_core.0001 } }
    }
}

# Divorce cleanup is pair-specific, so polygamous characters keep state for any
# other DHM-arranged spouse they may still have.
dhm_marriage.0380 = {
    type = character_event
    hidden = yes
    trigger = {
        exists = scope:spouse
        OR = {
            any_relation = { type = dhm_arranged_spouse this = scope:spouse }
            AND = {
                has_variable = dhm_arranged_union_spouse
                var:dhm_arranged_union_spouse = scope:spouse
            }
            scope:spouse = {
                has_variable = dhm_arranged_union_spouse
                var:dhm_arranged_union_spouse = root
            }
        }
    }
    immediate = {
        if = {
            limit = { any_relation = { type = dhm_arranged_spouse this = scope:spouse } }
            remove_relation_dhm_arranged_spouse = scope:spouse
            dhm_reconcile_arranged_union_state_effect = yes
            scope:spouse = { dhm_reconcile_arranged_union_state_effect = yes }
        }
        if = {
            limit = {
                has_variable = dhm_arranged_union_spouse
                var:dhm_arranged_union_spouse = scope:spouse
            }
            dhm_clear_pending_arranged_union_state_effect = yes
        }
        scope:spouse = {
            if = {
                limit = {
                    has_variable = dhm_arranged_union_spouse
                    var:dhm_arranged_union_spouse = root
                }
                dhm_clear_pending_arranged_union_state_effect = yes
            }
        }
    }
}
