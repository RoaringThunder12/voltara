# Main household hub. Settings are save-wide global variables so they survive
# succession. This remains a lightweight event UI for performance.
dhm_manager.0001 = {
    type = character_event
    title = dhm_manager.0001.t
    desc = {
        desc = dhm_manager.0001.desc_intro
        first_valid = {
            triggered_desc = { trigger = { has_global_variable = dhm_setting_preset_quiet } desc = dhm_manager.dashboard.preset_quiet }
            triggered_desc = { trigger = { has_global_variable = dhm_setting_preset_chronicle } desc = dhm_manager.dashboard.preset_chronicle }
            triggered_desc = { trigger = { has_global_variable = dhm_setting_preset_custom } desc = dhm_manager.dashboard.preset_custom }
            desc = dhm_manager.dashboard.preset_living
        }

        first_valid = {
            triggered_desc = {
                trigger = { has_global_variable = dhm_setting_education_disabled }
                desc = dhm_manager.dashboard.education_disabled
            }
            desc = dhm_manager.dashboard.education_enabled
        }

        first_valid = {
            triggered_desc = {
                trigger = { has_global_variable = dhm_setting_marriage_manual }
                desc = dhm_manager.dashboard.marriage_manual
            }
            triggered_desc = {
                trigger = { has_global_variable = dhm_setting_marriage_full_auto }
                desc = dhm_manager.dashboard.marriage_full_auto
            }
            desc = dhm_manager.dashboard.marriage_assisted
        }

        first_valid = {
            triggered_desc = { trigger = { global_var:dhm_household_cohesion < 20 } desc = dhm_cohesion.fractured }
            triggered_desc = { trigger = { global_var:dhm_household_cohesion < 40 } desc = dhm_cohesion.uneasy }
            triggered_desc = { trigger = { global_var:dhm_household_cohesion < 60 } desc = dhm_cohesion.stable }
            triggered_desc = { trigger = { global_var:dhm_household_cohesion < 80 } desc = dhm_cohesion.united }
            desc = dhm_cohesion.devoted
        }
    }
    theme = stewardship
    override_background = { reference = dhm_living_council }
    immediate = {
      hidden_effect = {
        if = {
            limit = { NOT = { has_global_variable = dhm_migration_0700 } }
            dhm_v070_migration_effect = yes
        }
        if = {
            limit = { NOT = { has_global_variable = dhm_migration_0400 } }
            dhm_v040_migration_effect = yes
        }
        if = {
            limit = { NOT = { has_character_flag = dhm_v040_character_initialized } }
            dhm_v040_initialize_character_effect = yes
        }
        if = {
            limit = { NOT = { has_character_flag = dhm_v040_arranged_relations_repaired } }
            dhm_repair_arranged_union_relations_effect = yes
        }
        if = {
            limit = { NOT = { has_global_variable = dhm_migration_0500 } }
            dhm_v050_migration_effect = yes
        }
        if = {
            limit = { NOT = { has_character_flag = dhm_v050_character_initialized } }
            dhm_v050_initialize_character_effect = yes
        }
		if = {
			limit = { NOT = { has_global_variable = dhm_migration_0600 } }
			dhm_v060_migration_effect = yes
		}
		if = {
			limit = { NOT = { has_character_flag = dhm_v060_character_initialized } }
			dhm_v060_initialize_character_effect = yes
		}
		if = {
			limit = { NOT = { has_global_variable = dhm_migration_0710 } }
			dhm_v071_migration_effect = yes
		}
		else_if = {
			limit = {
				OR = {
					NOT = { has_global_variable = dhm_migration_0800 }
					AND = {
						OR = { has_character_flag = dhm_v080_triumph_recent has_global_variable = dhm_v080_triumph_recent }
						NOT = { has_global_variable = dhm_v080_triumph_until }
					}
				}
			}
			dhm_v080_triumph_migration_effect = yes
		}
      }
    }

    option = {
        name = dhm_manager.0001.play_style
        custom_tooltip = dhm_manager.0001.play_style_tt
        trigger_event = dhm_manager.0090
    }

    option = {
        name = dhm_manager.0001.marriage_desk
        custom_tooltip = dhm_manager.0001.marriage_desk_tt
        trigger_event = dhm_manager.0006
    }

    option = {
        name = dhm_manager.0001.education_desk
        custom_tooltip = dhm_manager.0001.education_desk_tt
        trigger_event = dhm_manager.0005
    }

    option = {
        name = dhm_manager.0001.chronicle_desk
        custom_tooltip = dhm_manager.0001.chronicle_desk_tt
        trigger_event = dhm_manager.0008
    }

    option = {
        name = dhm_manager.0001.advanced
        custom_tooltip = dhm_manager.0001.advanced_tt
        trigger_event = dhm_manager.0091
    }

    option = {
        name = dhm_manager.0001.close
        is_cancel_option = yes
    }
}
