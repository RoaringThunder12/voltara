namespace = dhm_education

# Single-player birth hook. If the newborn's court host is the human player and
# the child is close family, assign the upbringing Guardian immediately.
dhm_education.0001 = {
    type = character_event
    hidden = yes

    trigger = {
        age = 0
        NOT = { has_global_variable = dhm_setting_education_disabled }
        OR = {
            mother ?= {
                is_ai = no
                is_in_the_same_court_as = root
            }
            father ?= {
                is_ai = no
                is_in_the_same_court_as = root
            }
            host ?= {
                is_ai = no
                is_close_family_of = root
                is_in_the_same_court_as = root
            }
        }
    }

    immediate = {
        save_scope_as = dhm_child
        if = {
            limit = { mother ?= { is_ai = no is_in_the_same_court_as = root } }
            mother = { save_scope_as = dhm_manager }
        }
        else_if = {
            limit = { father ?= { is_ai = no is_in_the_same_court_as = root } }
            father = { save_scope_as = dhm_manager }
        }
        else = { host = { save_scope_as = dhm_manager } }
        scope:dhm_child = { dhm_try_assign_guardian_effect = yes }
    }
}

# Sixteenth birthday: Tutor role ends with childhood education.
dhm_education.0003 = {
    type = character_event
    hidden = yes

    trigger = {
        age = 16
        any_relation = {
            type = dhm_specialist_tutor
            always = yes
        }
    }

    immediate = {
        save_scope_as = dhm_former_student
        random_relation = { type = dhm_specialist_tutor save_scope_as = dhm_selected_tutor }
        if = { limit = { mother ?= { is_ai = no } } mother = { save_scope_as = dhm_manager } }
        else_if = { limit = { father ?= { is_ai = no } } father = { save_scope_as = dhm_manager } }
        else_if = { limit = { host ?= { is_ai = no } } host = { save_scope_as = dhm_manager } }

        if = {
            limit = { exists = scope:dhm_selected_tutor }
            create_character_memory = {
                type = dhm_guided_education_memory
                participants = { tutor = scope:dhm_selected_tutor }
            }
            add_opinion = { target = scope:dhm_selected_tutor modifier = dhm_guided_education_opinion }
            if = { limit = { learning >= 12 }
                dhm_v040_apply_or_renew_reward_effect = { MODIFIER = dhm_v072_tutor_excellence_modifier YEARS = 2 RENEWAL_TT = dhm_v072_tutor_excellence_modifier_renewed_2_tt }
            }
        }
        every_relation = {
            type = dhm_specialist_tutor
            save_scope_as = dhm_previous_tutor
            scope:dhm_former_student = {
                remove_relation_dhm_specialist_tutor = scope:dhm_previous_tutor
                remove_character_flag = dhm_tutor_focus_diplomacy
                remove_character_flag = dhm_tutor_focus_martial
                remove_character_flag = dhm_tutor_focus_stewardship
                remove_character_flag = dhm_tutor_focus_intrigue
                remove_character_flag = dhm_tutor_focus_learning
            }
        }
		# Recommended mode reports only exceptional results. Full Chronicle keeps
		# every graduation; a poor result uses the recovery story below instead.
        if = {
            limit = {
                exists = scope:dhm_manager
                NOT = { has_global_variable = dhm_setting_development_reports_disabled }
				NOT = { dhm_v060_failed_education_trigger = yes }
				OR = {
					has_global_variable = dhm_setting_extended_sagas
					dhm_v070_exceptional_education_trigger = yes
				}
            }
            if = {
                limit = { scope:dhm_manager = { NOT = { has_character_flag = dhm_v080_birthday_story_pending } } }
                scope:dhm_manager = {
                    add_character_flag = { flag = dhm_v080_birthday_story_pending months = 3 }
                    trigger_event = dhm_education.0102
                }
            }
            else_if = {
                limit = { scope:dhm_manager = { NOT = { has_character_flag = dhm_v080_birthday_story_deferred } } }
                # Twins still receive separate conclusions, but never on the
                # same day. One deferred slot also covers wards and modded births.
                scope:dhm_manager = {
                    add_character_flag = { flag = dhm_v080_birthday_story_pending months = 5 }
                    add_character_flag = { flag = dhm_v080_birthday_story_deferred months = 6 }
                    trigger_event = { id = dhm_education.0102 months = 4 }
                    trigger_event = { id = dhm_core.0001 months = 6 }
                }
            }
        }
		if = {
			limit = {
				exists = scope:dhm_manager
				NOT = { has_global_variable = dhm_setting_stories_off }
				dhm_v060_failed_education_trigger = yes
			}
			if = {
				limit = { scope:dhm_manager = { NOT = { has_character_flag = dhm_v080_birthday_story_pending } } }
				scope:dhm_manager = {
					add_character_flag = { flag = dhm_v080_birthday_story_pending months = 3 }
					trigger_event = { id = dhm_saga.6100 days = { 20 45 } }
				}
			}
			else_if = {
				limit = { scope:dhm_manager = { NOT = { has_character_flag = dhm_v080_birthday_story_deferred } } }
				scope:dhm_manager = {
					add_character_flag = { flag = dhm_v080_birthday_story_pending months = 5 }
					add_character_flag = { flag = dhm_v080_birthday_story_deferred months = 6 }
					trigger_event = { id = dhm_saga.6100 months = 4 }
					trigger_event = { id = dhm_core.0001 months = 6 }
				}
			}
		}
    }
}
