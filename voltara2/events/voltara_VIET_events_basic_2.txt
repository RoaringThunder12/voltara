		modifier = {

			has_trait = craven
		}
	}
	
	option = { 
		name = VIETmisc.0407.a
		VIET_huge_prestige_gain_effect = yes
	}	
	
	option = { 
		name = VIETmisc.0407.b
		VIET_medium_prestige_loss_effect = yes

		stress_impact = {

			shy = miniscule_stress_impact_gain

			arrogant = minor_stress_impact_gain

			lazy = minor_stress_impact_gain

		}
	}	


	after = {		
		if = {
			limit = {
				always = no
			}

		}
	}

VIETmisc.0424 = {
	type = character_event
	title = VIETmisc.0424.t
	desc = VIETmisc.0424.desc
	theme = friendly


	left_portrait = {
		character = root
		animation = personality_compassionate
	}

	right_portrait = {
		character = scope:friend
		animation = personality_forgiving
	}

	trigger = {
		OR = {
			any_relation = {
				type = friend
				is_available_ai_adult = yes
			}
			any_relation = {
				type = best_friend
				is_available_ai_adult = yes
			}
		}		
	}
	
	weight_multiplier = {
		base = 1
		modifier = {
			add = 0.2
			has_trait = compassionate
		}
		modifier = {
			add = -0.2
			has_trait = callous
		}
		modifier = {
			add = 0.2
			has_trait = generous
		}
		modifier = {
			add = -0.2
			has_trait = greedy
		}
		modifier = {
			add = 0.2
			has_trait = humble
		}
		modifier = {
			add = -0.2
			has_trait = arrogant
		}
	}

	immediate = {
		random_list = {
			50 = {
				trigger = {
					any_relation = {
						type = friend
						is_available_ai_adult = yes
					}
				}
				random_relation = {
					type = friend
					limit = {
						is_available_ai_adult = yes
					}
					save_scope_as = friend
				}
			}
			50 = {
				trigger = {
					any_relation = {
						type = best_friend
						is_available_ai_adult = yes
					}
				}
				random_relation = {
					type = best_friend
					limit = {
						is_available_ai_adult = yes
					}
					save_scope_as = friend
				}
			}
		}
	}

	option = { 
		name = VIETmisc.0424.a
		scope:friend = {
			add_stress = minor_stress_impact_loss
			add_opinion = {
				target = root
				modifier = friendliness_opinion
				opinion = 15
			}				
			reverse_add_opinion = {
				target = root
				modifier = friendliness_opinion
				opinion = 15
			}				
		}
		stress_impact = {
			base = minor_stress_impact_loss
			arrogant = medium_stress_impact_gain
			paranoid = medium_stress_impact_gain
			sadistic = medium_stress_impact_gain
			callous = medium_stress_impact_gain
		}
	}

	option = { 
		name = VIETmisc.0424.b
		VIET_huge_prestige_gain_effect = yes
		scope:friend = {
			add_opinion = {
				modifier = sadness_opinion
				opinion = -10
				target = root
			}
		}
		stress_impact = {
			humble = minor_stress_impact_gain
			trusting = minor_stress_impact_gain
			patient = minor_stress_impact_gain
			compassionate = medium_stress_impact_gain
			gregarious = medium_stress_impact_gain
		}
	}

}
