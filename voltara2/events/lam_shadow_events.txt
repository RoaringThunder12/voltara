lam_shadow.0003 = {
	type = character_event
	title = lam_shadow.0003.t
	desc = lam_shadow.0003.desc
	theme = intrigue
	override_background = {
		reference = corridor_day
	}
	left_portrait = {
		character = root
		animation = paranoia
	}
	right_portrait = {
		character = scope:lam_sh_rel
		animation = disapproval
	}
	trigger = {
		lam_nc_control_trigger = {
			LORD = scope:lam_sh_king
		}
	}
	immediate = {
		scope:lam_sh_king = {
			random_close_family_member = {
				limit = {
					is_adult = yes
					is_imprisoned = no
					NOT = {
						this = root
					}
					lam_nc_arrest_target_trigger = {
						LORD = scope:lam_sh_king
						MASTER = root
					}
				}
				save_scope_as = lam_sh_rel
			}
		}
	}
	option = {
		name = lam_shadow.0003.a
		trigger = {
			lam_nc_control_trigger = {
				LORD = scope:lam_sh_king
			}
			scope:lam_sh_rel ?= {
				lam_nc_arrest_target_trigger = {
					LORD = scope:lam_sh_king
					MASTER = root
				}
			}
		}
		hidden_effect = {
			if = {
				limit = {
					lam_nc_control_trigger = {
						LORD = scope:lam_sh_king
					}
					scope:lam_sh_rel ?= {
						lam_nc_arrest_target_trigger = {
							LORD = scope:lam_sh_king
							MASTER = root
						}
					}
				}
				scope:lam_sh_king = {
					lam_nc_arrest_effect = {
						MASTER = root
						TARGET = scope:lam_sh_rel
						TYPE = house_arrest
					}
				}
				scope:lam_sh_rel = {
					add_opinion = {
						target = root
						modifier = angry_opinion
						opinion = -30
					}
				}
			}
		}
	}
	option = {
		name = lam_shadow.0003.b
		trigger = {
			lam_nc_control_trigger = {
				LORD = scope:lam_sh_king
			}
			scope:lam_sh_rel ?= {
				lam_nc_arrest_target_trigger = {
					LORD = scope:lam_sh_king
					MASTER = root
				}
			}
		}
		hidden_effect = {
			if = {
				limit = {
					lam_nc_control_trigger = {
						LORD = scope:lam_sh_king
					}
					scope:lam_sh_rel ?= {
						lam_nc_arrest_target_trigger = {
							LORD = scope:lam_sh_king
							MASTER = root
						}
					}
				}
				pay_short_term_gold = {
					target = scope:lam_sh_rel
					gold = minor_gold_value
				}
				scope:lam_sh_rel = {
					add_opinion = {
						target = root
						modifier = bribed_opinion
						opinion = 25
					}
				}
			}
		}
	}
	option = {
		name = lam_shadow.0003.c
		add_stress = minor_stress_gain
	}
}

		name = lam_shadow.0016.a

			chance = { value = 50 add = { value = intrigue subtract = scope:lam_sh_king.intrigue multiply = 2 } if = { limit = { has_variable = lam_fac_share var:lam_fac_share >= 40 } add = 10 } min = 10 max = 90 }

			add_dread = medium_dread_gain
			scope:lam_sh_king = { add_stress = major_stress_gain add_opinion = { target = root modifier = scared_opinion opinion = -20 } }
		}
		if = { limit = { NOT = { exists = scope:lam_sh_cw } } custom_tooltip = lam_shadow.0016.lose_tt hidden_effect = { lam_sh_counter_lost_effect = yes } }
	}
	option = {
		name = lam_shadow.0016.b
		add_prestige = medium_prestige_loss
		set_variable = { name = lam_sd_law_cd value = yes years = 2 }

		scope:lam_sh_king = { add_stress = medium_stress_loss add_opinion = { target = root modifier = grateful_opinion opinion = 20 } }
	}
	option = {
		name = lam_shadow.0016.c
		trigger = { gold >= major_gold_value }
		remove_short_term_gold = major_gold_value
		random = {
			chance = 80
			save_scope_value_as = { name = lam_sh_cw value = yes }
			custom_tooltip = lam_shadow.0016.win_tt
			add_dread = medium_dread_gain
			scope:lam_sh_king = { add_stress = major_stress_gain add_opinion = { target = root modifier = scared_opinion opinion = -20 } }
		}
		if = { limit = { NOT = { exists = scope:lam_sh_cw } } custom_tooltip = lam_shadow.0016.lose_tt hidden_effect = { lam_sh_counter_lost_effect = yes } }
	}
	option = {
		name = lam_shadow.0016.d
		add_prestige = minor_prestige_gain
		custom_tooltip = lam_shadow.0016.go_tt
		scope:lam_sh_king = { add_opinion = { target = root modifier = grateful_opinion opinion = 20 } lam_fac_release_effect = yes }
