lam_cr.0004 = {
	hidden = yes
	immediate = {
		if = {
			limit = {
				is_alive = yes
				this = scope:lam_cr_lord
				scope:lam_cr_head = {
					var:lam_cr_contract_pending ?= scope:lam_cr_token
					is_alive = yes
					has_variable = lam_bloc_lead
					liege ?= root
					lam_cr_deal_valid_trigger = yes
				}
			}
			if = {
				limit = {
					scope:lam_cr_head = {
						var:lam_cr_give ?= flag:county
					}
				}
				scope:lam_cr_head = {
					save_scope_as = lam_deal_lead
				}
				trigger_event = lam_faction.1030
			}
			else = {
				scope:lam_cr_head = {
					remove_variable = lam_cr_contract_pending
					save_scope_as = lam_deal_lead
				}
				clear_saved_scope = lam_cr_old_give
				clear_saved_scope = lam_cr_old_ask
				if = {
					limit = {
						has_variable = lam_deal_give
					}
					clear_saved_scope = lam_cr_old_give
					if = {
						limit = {
							var:lam_deal_give ?= flag:amnesty
						}
						save_scope_value_as = {
							name = lam_cr_old_give
							value = flag:amnesty
						}
					}
					else_if = {
						limit = {
							var:lam_deal_give ?= flag:bribe
						}
						save_scope_value_as = {
							name = lam_cr_old_give
							value = flag:bribe
						}
					}
					else_if = {
						limit = {
							var:lam_deal_give ?= flag:county
						}

					remove_variable = lam_deal_ask
				}
			}
		}

				if = {
					limit = {
						var:lam_cr_contract_pending ?= scope:lam_cr_token
					}
