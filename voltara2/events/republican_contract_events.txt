namespace = republican_contract

scripted_effect set_contract_scopes_effect = {
	scope:task_contract.task_contract_taker ?= {
		save_scope_as = task_contract_taker
	}
	scope:task_contract.task_contract_employer ?= {
		save_scope_as = task_contract_employer
	}
	if = {
		limit = {
			exists = scope:task_contract.task_contract_destination
		}
		scope:task_contract.task_contract_destination = {
			save_scope_as = task_contract_destination
		}
	}
	else = {
		scope:task_contract.var:task_contract_destination ?= {
			save_scope_as = task_contract_destination
		}
	}
	if = {
		limit = {
			exists = scope:task_contract.task_contract_target
		}
		scope:task_contract.task_contract_target ?= {
			save_scope_as = task_contract_target
		}
	}
	else = {
		scope:task_contract.var:task_contract_target ?= {
			save_scope_as = task_contract_target
		}
	}
	scope:task_contract.var:task_contract_war ?= {
		save_scope_as = task_contract_war
	}
	if = {
		limit = {
			exists = scope:task_contract.var:task_contract_object
		}
		scope:task_contract.var:task_contract_object = {
			save_scope_as = task_contract_object
		}
	}
	if = {
		limit = {
			exists = scope:task_contract.var:escorted_artifact
		}
		scope:task_contract.var:escorted_artifact = {
			save_scope_as = escorted_artifact
		}
	}
	if = {
		limit = {
			exists = scope:task_contract.var:escorted_gold
		}
		scope:task_contract.var:escorted_gold = {
			save_scope_as = escorted_gold
		}
	}
	if = {
		limit = {
			exists = scope:task_contract.var:escorted_story
		}
		scope:task_contract.var:escorted_story = {
			save_scope_as = escorted_story
		}
	}
}

republican_contract.0002 = {

			}

# Trade Task - Intro
republican_contract.1000 = {
	type = character_event
	title = republican_contract.1000.t
	desc = republican_contract.1000.desc
	theme = diplomacy
	override_background = {
		reference = alley_day
	}
	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:merchant
		animation = debating
	}
	lower_right_portrait = {
		character = scope:task_contract_employer
	}
	
	immediate = {
		set_contract_scopes_effect = yes
		location.county = { save_scope_as = county }
		republican_contract_pick_trade_task = { TASK = trade_task }
		scope:task_contract = {
			set_variable = {
				name = trade_task
				value = scope:trade_task
			}
		}
		create_character = {
			template = merchant_template
			location = this.location
			faith = location.faith
			culture = location.culture
			random_traits = yes
			save_scope_as = merchant
		}
		location = {
			save_scope_as = location
		}
	}

	#Convince him with diplomacy and stewardship
	option = {
		name = republican_contract.1000.a

		duel = {
			skills = { diplomacy stewardship }
			target = scope:merchant
			
			50 = { #You win the debate
				desc = republican_contract.1000.a.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 2.5
					min = -49
				}
				min = 5
				scope:task_contract = {
					show_as_tooltip = { republican_contract_apply_trade_task_effect = yes }
					complete_task_contract = success_standard
				}
				add_character_flag = contract_success_flag
				trigger_event = republican_contract.1010
			}
			40 = { #You are unconvincing
				desc = republican_contract.1000.a.failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = -2.5
					min = -49
				}
				min = 5
				scope:task_contract = {
					complete_task_contract = failure_standard
				}
				add_character_flag = contract_failure_flag
				trigger_event = republican_contract.1010
			}
			10 = { #You actively damage your employer in the eyes of his subjects
				desc = republican_contract.1000.a.critfailure
				compare_modifier = {
					value = scope:duel_value
					multiplier = -2.5
					min = -49
				}
				min = 1
				scope:task_contract = {
					complete_task_contract = failure_critical
				}
				add_character_flag = contract_crit_failure_flag
				trigger_event = republican_contract.1010
			}
		}
		stress_impact = {
			honest = minor_stress_impact_loss
			arrogant = minor_stress_impact_loss
			compassionate = minor_stress_impact_loss
			sadistic = minor_stress_impact_gain
		}
		ai_chance = {
			base = 10
			modifier = {
				add = diplomacy
				add = stewardship
			}
		}
	}

	#Convince him with learning and stewardship
	option = {
		name = republican_contract.1000.b
		duel = {
			skills = { learning stewardship }
			target = scope:merchant
			
			30 = { #You win the 'debate'
				desc = republican_contract.1000.b.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 2.5
					min = -49
				}
				min = 3
				scope:task_contract = {
					show_as_tooltip = { republican_contract_apply_trade_task_effect = yes }
					complete_task_contract = success_standard
				}
				add_character_flag = contract_success_flag
				trigger_event = republican_contract.1010
			}
			50 = { #You are unconvincing
				desc = republican_contract.1000.b.failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = -2.5
					min = -49
				}
				min = 5
				scope:task_contract = {
					complete_task_contract = failure_standard
				}
				add_character_flag = contract_failure_flag
				trigger_event = republican_contract.1010
			}
			20 = { #You actively damage your employer in the eyes of his subjects
				desc = republican_contract.1000.b.critfailure
				compare_modifier = {
					value = scope:duel_value
					multiplier = -2.5
					min = -49
				}
				min = 1
				scope:merchant = {
					create_faction = {
						type = peasant_faction
						target = scope:task_contract_employer
					}
				}
				scope:task_contract = {
					complete_task_contract = failure_critical
				}
				add_character_flag = contract_crit_failure_flag
				trigger_event = republican_contract.1010
			}
		}
		stress_impact = {
			honest = minor_stress_impact_gain
			compassionate = minor_stress_impact_gain
			arrogant = minor_stress_impact_gain
		}
		ai_chance = {
			base = 10
			modifier = {
				add = learning
				add = stewardship
			}
		}
	}

	#Pay him off to sweeten the deal
	option = {
		name = republican_contract.1000.c

		pay_short_term_gold = {
			target = scope:merchant
			gold = minor_gold_value
		}

		random_list = {
			75 = {
				modifier = {
					add = 15
					scope:merchant = {
						OR = {
							has_trait = greedy
							has_trait = arbitrary
						}
					}
				}
				scope:task_contract = {
					show_as_tooltip = { republican_contract_apply_trade_task_effect = yes }
					complete_task_contract = success_standard
				}
				add_character_flag = contract_success_flag
				trigger_event = republican_contract.1010
			}
			25 = {
				modifier = {
					add = 25
					scope:merchant = {
						OR = {
							has_trait = stubborn
							has_trait = ambitious
						}
					}
				}
				scope:task_contract = {
					complete_task_contract = failure_standard
				}
				add_character_flag = contract_failure_flag
				trigger_event = republican_contract.1010
			}
		}


		stress_impact = {
			gregarious = minor_stress_impact_loss
			zealous = minor_stress_impact_loss
			compassionate = minor_stress_impact_loss
			cynical = minor_stress_impact_gain
			greedy = minor_stress_impact_gain
		}
		ai_chance = {
			base = 10
			modifier = {
				add = 10
				gold > 100
			}
		}
	}

	#invalidate the contract
	option = {
		name = ep3_contract_event.0001.invalidate
		scope:task_contract = {
			invalidate_contract = yes
		}
		#AI should always try to complete the contract
		ai_chance = {
			base = 0
		}
	}
}

# Trade Task - Resolution
republican_contract.1010 = {
	type = character_event
	title = {
		first_valid = {
			triggered_desc = {
				trigger = { has_character_flag = contract_success_flag }
				desc = republican_contract.1010.t.success
			}
			triggered_desc = {
				trigger = { has_character_flag = contract_failure_flag }
				desc = republican_contract.1010.t.failure
			}
			triggered_desc = {
				trigger = { has_character_flag = contract_crit_failure_flag }
				desc = republican_contract.1010.t.critfailure
			}
		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_character_flag = contract_success_flag }
				desc = republican_contract.1010.desc.success
			}
			triggered_desc = {
				trigger = { has_character_flag = contract_failure_flag }
				desc = republican_contract.1010.desc.failure
			}
			triggered_desc = {
				trigger = { has_character_flag = contract_crit_failure_flag }
				desc = republican_contract.1010.desc.critfailure
			}
		}
	}
	theme = diplomacy
	override_background = {
		reference = alley_day
	}
	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:merchant
		animation = debating
	}
	lower_right_portrait = {
		character = scope:task_contract_employer
	}
	
	immediate = {
		set_contract_scopes_effect = yes
		location = {
			save_scope_as = location
		}
		show_as_tooltip = {
			if = {
				limit = {
					has_character_flag = contract_success_flag
				}
				scope:task_contract = {
					republican_contract_apply_trade_task_effect = yes
					complete_task_contract = success_standard
				}
			}
			else_if = {
				limit = {
					has_character_flag = contract_failure_flag
				}
				scope:task_contract = {
					complete_task_contract = failure_standard
				}
			}
			else = {
				scope:task_contract = {
					complete_task_contract = failure_critical
				}
			}
		}
	}

	option = {
		name = republican_contract.1010.option.success
		add_trait_xp = {
			trait = republican_merchant
			value = 5
		}
		trigger = {
			has_character_flag = contract_success_flag
		}

		stress_impact = {
			base = miniscule_stress_impact_loss
		}
		ai_chance = {
			base = 10
		}
	}

	option = {
		name = republican_contract.1010.option.failure
		add_trait_xp = {
			trait = republican_merchant
			value = 1
		}
		trigger = {
			NOT = {
				has_character_flag = contract_success_flag
			}
		}

		stress_impact = {
			base = miniscule_stress_impact_gain
		}
		ai_chance = {
			base = 10
		}
	}

	after = {
		remove_character_flag = contract_crit_success_flag
		remove_character_flag = contract_success_flag
		remove_character_flag = contract_failure_flag
		remove_character_flag = contract_crit_failure_flag
	}
}
