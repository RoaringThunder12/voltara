lam_cv.0004 = {
	type = character_event
	title = lam_cv_menu_rise_t
	desc = lam_fac_head_rise_decision_desc
	theme = realm
	left_portrait = {
		character = root
		animation = thinking
	}
	option = {
		name = lam_cv_rise_council
		trigger = {
			lam_pt_head_ready_trigger = yes
			NOT = {
				lam_wc_committed_trigger = yes
			}
			NOT = {
				exists = joined_faction
			}
			NOT = {
				is_at_war_with = liege
			}
			custom_description = {
				text = lam_fac_rise_cd_tt
				NOT = {
					has_variable = lam_fac_rebel_cd
				}
			}
			custom_description = {
				text = lam_fac_rise_cause_tt
				lam_rv_rise_cause_trigger = yes
			}
			custom_description = {
				text = lam_fac_rise_blocked_tt
				OR = {
					NOT = {
						has_character_flag = joining_faction_block
					}
					has_variable = lam_fac_blocked
				}
			}
			custom_description = {
				text = lam_cv_call_retry_tt
				NOT = {
					has_variable = lam_cr_retry_rise
				}
			}
		}
		custom_tooltip = lam_cv_rise_preview_tt
		custom_tooltip = lam_fac_head_rise_decision_tt
		if = {
			limit = {
				has_variable = lam_fac_oath
			}
			custom_tooltip = lam_fac_rise_oath_tt
			add_prestige = major_prestige_loss
			hidden_effect = {
				remove_variable = lam_fac_oath
				lam_fac_timeline_add_effect = {
					KEY = oath_broken
				}
				liege = {
					add_opinion = {
						target = root
						modifier = lam_fac_betrayed_opinion
					}
				}
			}
		}
		hidden_effect = {
			set_variable = {
				name = lam_cr_view
				value = this
			}
			liege = {
				save_scope_as = lam_abs_lord
			}
			set_variable = {
				name = lam_fac_stage
				value = 4
			}
			lam_fac_rise_effect = yes
		}
	}
	option = {
		name = lam_cv_rise_puppet
		trigger = {
			lam_pt_head_ready_trigger = yes
			NOT = {
				lam_wc_committed_trigger = yes
			}
			NOT = {
				exists = joined_faction
			}
			NOT = {
				is_at_war_with = liege
			}
			custom_description = {
				text = lam_fac_rise_cd_tt
				NOT = {
					has_variable = lam_fac_rebel_cd
				}
			}
			custom_description = {
				text = lam_fac_rise_cause_tt
				lam_rv_rise_cause_trigger = yes
			}
			custom_description = {
				text = lam_fac_rise_blocked_tt
				OR = {
					NOT = {
						has_character_flag = joining_faction_block
					}
					has_variable = lam_fac_blocked
				}
			}
			custom_description = {
				text = lam_cv_call_retry_tt
				NOT = {
					has_variable = lam_cr_retry_rise
				}
			}
			NOT = {
				liege ?= {
					has_variable = lam_fac_puppet_master
				}
			}
			lam_pol_ready_trigger = yes
		}
		show_as_unavailable = {
			lam_pol_control_trigger = yes
		}
		custom_tooltip = lam_cv_rise_puppet_tt
		if = {
			limit = {
				has_variable = lam_fac_oath
			}
			custom_tooltip = lam_fac_rise_oath_tt
			add_prestige = major_prestige_loss
			hidden_effect = {
				remove_variable = lam_fac_oath
				lam_fac_timeline_add_effect = {
					KEY = oath_broken
				}
				liege = {
					add_opinion = {
						target = root
						modifier = lam_fac_betrayed_opinion
					}
				}
			}
		}
		hidden_effect = {
			liege = {
				save_scope_as = lam_abs_lord
			}
			lam_fac_ult_puppet_effect = yes
		}
	}
	option = {
		name = lam_cv_menu_back
		set_variable = {
			name = lam_cr_view
			value = this
		}
	}
}
