republican_venice.2010 = {
	type = character_event
	title = republican_venice.2010.t
	desc = republican_venice.2010.desc
	theme = war
	override_background = { reference = republican_narentine_shore }

	left_portrait = {
		character = root
		animation = aggressive_sword
	}
	immediate = { play_sound_effect = "event:/SFX/UI/Notifications/Popup/sfx_ui_popup_theme_neutral" }

	option = { #Go to war
		name = republican_venice.2010.a
		var:narentine_story = {
			set_variable = {
				name = narentine_war_triggered
				value = yes
			}
		}
		spawn_army = {
			name = event_troop_default_name
			men_at_arms = {
				type = pikemen_unit
				stacks = 4
			}

				type = armored_footmen
				stacks = 5
			}
			men_at_arms = {
				type = mangonel
				stacks = 3
			}
			location = root.location
			origin = root.location
			inheritable = no
		}
		#Declare war on Venice
		start_war = {
			cb = republican_sack_venice
			target = title:d_voltara_test_11336.holder
			target_title = title:d_voltara_test_11336
		}
		play_sound_effect = "event:/SFX/UI/War/sfx_ui_war_declare_war"
		play_music_cue = mx_cue_war_declared
		

		ai_chance = {
			base = 100
			ai_value_modifier = {
   				ai_boldness = 2
   				ai_compassion = -1
			}
			modifier = {
				add = -15
				has_trait = compassionate
			}
			modifier = {
				add = -15
				has_trait = craven
			}
		}
	}

	option = { #We must wait

				name = narentine_war_delays
				add = 1
			}
		}
		ai_chance = {
			ai_value_modifier = {
				ai_energy = -1
			}
			modifier = {
				add = 30
				has_trait = patient
			}
			modifier = {
				add = 50
				is_at_war = yes
			}
		}
	}

	option = { #We have no interest in fighting the Venetians
		name = republican_venice.2010.c
		var:narentine_story = {
			set_variable = {
				name = narentine_war_triggered

			title:d_voltara_test_11336.holder.var:republican_government_story = { save_scope_as = republican_government_story }

		}
		title:d_voltara_test_11336.holder = { #Let's pardon him
			if = {
				limit = { has_imprisonment_reason = root }
				consume_imprisonment_reasons = root
			}

				limit = { has_execute_reason = root }
				consume_execute_reasons = root
			}
			if = {
				limit = { has_revoke_title_reason = root }
				consume_revoke_title_reason = root
			}
			if = {
				limit = { has_banish_reason = root }
				consume_banish_reasons = root
			}
			if = {
				limit = { has_banish_reason = scope:actor }
					consume_banish_reasons = scope:actor
			}
		}
		set_culture = title:d_voltara_test_11336.holder.culture
		set_character_faith = title:d_voltara_test_11336.holder.faith

		ai_chance = {
			ai_value_modifier = {
				ai_greed = 2
				ai_vengefulness = -1
				ai_compassion = 1
			}
			modifier = {
				add = 15
				has_trait = compassionate
			}
			modifier = {

			}

			}

		}
		scope:attacker = {
			every_sub_realm_county = {
				limit = {
					culture = culture:voltara_western_corsairs
				}
				set_county_culture = scope:defender.culture
				set_county_faith = scope:defender.faith
				scope:defender = { add_short_term_gold = minor_gold_value }
				change_development_level = -10
			}
			every_held_title = { #Destroy all titles above duchy-level
				limit = { tier > tier_duchy}
				scope:attacker = { destroy_title = prev }
			}
			create_title_and_vassal_change = {
				type = conquest
				save_scope_as = change
				add_claim_on_loss = yes
			}
			every_held_title = { #Give all remaining titles

				}

			resolve_title_and_vassal_change = scope:change
			create_landless_adventurer_title_effect = {

			}

				ai_vengefulness = 2

				add = -15

			}
		}
	}

	option = { #Vassalize, remove piracy from their culture and convert them to the religion of Venice
		name = republican_venice.2030.c
		play_sound_effect = "event:/DLC/FP2/SFX/UI/fp2_struggle_ending_decision_confirm"
		scope:attacker = {
			play_sound_effect = "event:/DLC/FP2/SFX/UI/fp2_struggle_ending_decision_confirm"
			var:narentine_story = {
				end_story = yes
			}

				add_claim_on_loss = yes
			}

				limit = {
					NOT = { tier = tier_county }
					tier = scope:attacker.primary_title.tier
				}
				change_title_holder = {
					holder = scope:defender
					change = scope:change
				} 
			}
			change_liege = {
				liege = scope:defender
				change = scope:change
			}
			resolve_title_and_vassal_change = scope:change
			set_character_faith = scope:defender.faith
			culture:voltara_western_corsairs = {
				remove_culture_tradition = tradition_practiced_pirates
