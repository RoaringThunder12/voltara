lam_faction.0300 = {
	type = character_event
	title = lam_faction.0300.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					var:lam_ff_lord ?= root
				}
				desc = lam_faction.0300.own
			}
			desc = lam_faction.0300.peer
		}
	}
	theme = realm
	override_background = {
		reference = council_chamber
	}
	left_portrait = {
		character = root
		animation = personality_rational
	}
	immediate = {
		save_scope_as = lam_ff_founder
		var:lam_ff_lord = {
			save_scope_as = lam_ff_lord
		}
		clear_saved_scope = lam_ff_c1
		clear_saved_scope = lam_ff_c2
		clear_saved_scope = lam_ff_c3
		scope:lam_ff_lord = {
			ordered_vassal = {
				limit = {
					is_ai = yes
					is_ruler = yes
					is_adult = yes
					is_imprisoned = no
					NOT = {
						this = scope:lam_ff_founder
					}
					NOT = {
						has_variable = lam_bloc_lead
					}
					NOT = {
						has_variable = lam_ff_refused
					}
				}
				order_by = lam_ff_order_value
				max = 3
				check_range_bounds = no
				set_variable = {
					name = lam_ff_acc
					value = lam_ff_accept_value
				}
				if = {
					limit = {
						NOT = {
							exists = scope:lam_ff_c1
						}
					}
					save_scope_as = lam_ff_c1
				}
				else_if = {
					limit = {
						NOT = {
							exists = scope:lam_ff_c2
						}
					}
					save_scope_as = lam_ff_c2
				}
				else = {
					save_scope_as = lam_ff_c3
				}
			}
		}
	}
	option = {
		name = lam_faction.0300.me
		trigger = {
			lam_ff_self_current_trigger = yes
		}
		custom_tooltip = lam_faction.0300.me_tt
		custom_tooltip = lam_faction.0300.me_cofounders_tt
		hidden_effect = {
			if = {
				limit = {
					lam_ff_self_current_trigger = yes
				}
				scope:lam_ff_lord = {
					lam_ff_install_player_effect = {
						NEW = root
						FAVOUR = 0
						FOUNDER = root
					}
				}
				if = {
					limit = {
						has_variable = lam_fac_cof1
					}
					remove_variable = lam_fac_cof1
				}
				if = {
					limit = {
						has_variable = lam_fac_cof2
					}
					remove_variable = lam_fac_cof2
				}
				scope:lam_ff_lord = {
					ordered_vassal = {
						limit = {
							lam_ff_cofounder_trigger = yes
						}
						order_by = "opinion(root)"
						max = 2
						check_range_bounds = no
						set_variable = {
							name = lam_bloc
							value = root
						}
						set_variable = {
							name = lam_fac_tier
							value = 1
						}
						set_variable = {
							name = lam_fac_pinned
							value = yes
							years = 2
						}
						if = {
							limit = {
								has_variable = lam_fac_unbound
							}
							remove_variable = lam_fac_unbound
						}
						if = {
							limit = {
								house ?= root.house
							}
							set_variable = {
								name = lam_fac_cof_why
								value = flag:kin
							}
						}
						else_if = {
							limit = {
								OR = {
									has_relation_friend = root
									has_relation_best_friend = root
								}
							}
							set_variable = {
								name = lam_fac_cof_why
								value = flag:friend
							}
						}
						else = {
							set_variable = {
								name = lam_fac_cof_why
								value = flag:liking
							}
						}
						root = {
							if = {
								limit = {
									NOT = {
										has_variable = lam_fac_cof1
									}
								}
								set_variable = {
									name = lam_fac_cof1
									value = prev
								}
							}
							else = {
								set_variable = {
									name = lam_fac_cof2
									value = prev
								}
							}
						}
						root = {
							send_interface_message = {
								type = event_generic_good
								title = lam_ff_cofounder_title
								right_icon = prev
							}
						}
					}
				}
				set_variable = {
					name = lam_found_cd
					value = yes
					years = 5
				}
				if = {
					limit = {
						has_variable = lam_fac_pinned
					}
					remove_variable = lam_fac_pinned
				}
				save_scope_as = lam_fac_new
				scope:lam_ff_lord = {
					lam_fac_quick_refresh_effect = yes
				}
				trigger_event = lam_faction.0200
			}
		}
	}
	option = {
		name = lam_faction.0300.c1
		trigger = {
			lam_ff_candidate_current_trigger = {
				WHO = scope:lam_ff_c1
			}
		}
		custom_tooltip = lam_faction.0300.acc1
		hidden_effect = {
			if = {

				}
