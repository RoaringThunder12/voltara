republican_trade.1000 = { #Trade Deal ruler died, negotiate new trade deal
	type = character_event
	title = republican_trade.1000.t
	desc = republican_trade.1000.desc
	theme = realm
	override_background = { reference = republican_trade_deal }

	left_portrait = {
		character = root
		animation = personality_rational
	}
	lower_center_portrait = { character = scope:recipient }

	trigger = {
		OR = {
			can_start_scheme = {
				type = trade_deal_scheme
				target_character = scope:recipient
			}
			any_vassal = {
				can_start_scheme = {
					type = trade_deal_scheme
					target_character = scope:recipient
				}
			}
		}
	}

	option = { #Do it myself
		name = republican_trade.1000.a
		trigger = {
			can_start_scheme = {
				type = trade_deal_scheme
				target_character = scope:recipient
			}
		}
		send_interface_toast = {
			type = event_toast_effect_neutral
			title = trade_deal_interaction_notification

			left_icon = root				
			right_icon = scope:recipient

			begin_scheme_basic_effect = {
				SCHEME_TYPE = trade_deal_scheme
				TARGET_TYPE = target_character
				TARGET_SCOPE = scope:recipient
			}
		}

		ai_chance = {
			base = 0
			ai_value_modifier = { ai_energy = 0.5 }
			modifier = {
				add = -1000
				any_scheme = {
					scheme_type = trade_deal_scheme
				}
			}
			modifier = {
				add = -30
				any_scheme = {
					scheme_type = learn_language
				}
			}
			modifier = {
				add = -10
				any_scheme = {
					scheme_type = sway
				}
			}
		}
	}
	option = { #Ask the merchant families for help
		name = republican_trade.1000.b
		trigger = {
			any_vassal = {
				can_start_scheme = {
					type = trade_deal_scheme
					target_character = scope:recipient
				}
			}
		}
		custom_tooltip = {
			text = republican_trade.1000.b.tt
			every_vassal = {
				limit = { is_merchant_family = yes }
				trigger_event = republican_trade.1001
			}
		}
		ai_chance = {
			base = 0
			ai_value_modifier = {
				ai_greed = 0.5
				ai_energy = -0.5
			}
		}
	}

	option = { #It's not worth the effort
		name = republican_trade.1000.c
		custom_tooltip = republican_trade.1000.c.tt

		ai_chance = {
			base = 0
			ai_value_modifier = {
				ai_greed = -0.5
				ai_energy = -0.5
			}
		}
	}
}

		}

			modifier = pleased_opinion
			opinion = 10
		}

	}

	option = {
		name = republican_trade.1002.a
	}
}

republican_trade.1004 = { #AI start trade deal scheme
	hidden = yes
	immediate = {
		ordered_independent_ruler = {
			order_by = ai_trade_deal_target_weight
			position = { integer_range = { min = 0 max = 4 } }
			limit = {
				in_diplomatic_range = scope:actor
				scope:actor = {
					can_start_scheme = {
						type = trade_deal_scheme
						target_character = prev
					}
				}
			}
			scope:actor = {
				begin_scheme_basic_effect = {
					SCHEME_TYPE = trade_deal_scheme
					TARGET_TYPE = target_character
					TARGET_SCOPE = prev
				}
			}
		}
	}

# Rival merchants
republican_trade.1011 = {
	type = character_event
	title = republican_trade.1011.t
	desc = republican_trade.1011.desc
	theme = diplomacy
	override_background = { reference = republican_trade_deal }

	left_portrait = {
		character = scope:owner
		animation = eyeroll
		camera = camera_event_left
	}

	right_portrait = {
		character = scope:merchant_character
		animation = debating
		camera = camera_event_right
	}

	immediate = {
		create_character = {
			template = merchant_template
			faith = scope:target.faith
			culture = scope:target.culture
			gender = root
			location = location
			save_scope_as = merchant_character

			after_creation = {
				add_prestige = { integer_range = { min = 25 max = 500 } }
				add_piety = { integer_range = { min = 25 max = 250 } }
				add_gold = { integer_range = { min = root.medium_gold_value max = root.major_gold_value } }
			}
		}
	}

	option = { # Expose the rival merchant faction's deceit.
		name = republican_trade.1011.a
		
		duel = {
			skill = intrigue
			target = scope:merchant_character
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3
				}
				
				send_interface_toast = {
					type = event_trade_scheme
					title = republican_trade_negotiation_toast
					scope:scheme = {
						add_scheme_modifier = {
							type = extra_success_chance_modifier
						}
					}
				}
			}
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3
				}
				send_interface_toast = {
					type = event_trade_scheme
					title = republican_trade_negotiation_toast
					scope:scheme = { end_scheme = yes }
				}
			}
		}
		stress_impact = {
			deceitful = major_stress_impact_loss
			lazy = minor_stress_impact_gain
			stubborn = medium_stress_impact_loss
		}
		ai_chance = {
			base = 10
			modifier = {
				factor = 0
				OR = {
					has_trait = honest
				}
			}
		}
	}

	option = { # Bypass the court and strike a private deal with the ruler.
		name = republican_trade.1011.b
		
		duel = {
			skill = diplomacy
			target = scope:merchant_character
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3
				}
				send_interface_toast = {
					type = event_trade_scheme
					title = republican_trade_negotiation_toast
					scope:scheme = {
						add_scheme_modifier = {
							type = extra_success_chance_modifier
						}
					}
				}
			}
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3
				}
				send_interface_toast = {
					type = event_trade_scheme
					title = republican_trade_negotiation_toast
					scope:scheme = { end_scheme = yes }
				}
			}
		}
		stress_impact = {
			shy = major_stress_impact_loss
			lazy = minor_stress_impact_gain
			honest = medium_stress_impact_loss
			stubborn = medium_stress_impact_loss
		}
		ai_chance = {
			base = 10
			modifier = {
				factor = 0
				OR = {
					has_trait = shy
					has_trait = deceitful
				}
			}
		}
	}

	option = { # Buy out the rival merchants to remove their influence.
		name = republican_trade.1011.c
		
		remove_short_term_gold = minor_gold_value
		stress_impact = {
			greedy = major_stress_impact_gain
			arrogant = medium_stress_impact_gain
			lazy = medium_stress_impact_loss
		}
		ai_chance = {
			base = 10
			modifier = {
				factor = 0
				OR = {
					has_trait = greedy
					has_trait = arrogant
				}
			}
		}
	}

	option = { # Abandon the deal and focus elsewhere.
		name = republican_trade.1011.d
		scope:scheme = { end_scheme = yes }
		stress_impact = {
			lazy = major_stress_impact_loss
			greedy = minor_stress_impact_gain
		}
		ai_chance = {
			base = 10
		}
	}
